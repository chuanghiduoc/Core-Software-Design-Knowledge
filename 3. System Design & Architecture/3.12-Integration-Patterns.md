## Anti-Corruption Layer

Khi integrate với external system (legacy, third-party), **đừng để model/API của họ "nhiễm" vào domain của bạn**.

```typescript
// ❌ Stripe types leak vào business logic
class OrderService {
  async processPayment(order: Order): Promise<void> {
    // Stripe-specific types spread throughout codebase
    const intent: Stripe.PaymentIntent = await stripe.paymentIntents.create({
      amount: order.total,
      currency: 'usd',
      metadata: { orderId: order.id },
    });
    // Business logic now coupled to Stripe's data model
    if (intent.status === 'requires_action') { /* Stripe-specific status */ }
  }
}

// ✅ Anti-Corruption Layer: isolate external models behind YOUR interfaces

// YOUR domain interface
interface PaymentGateway {
  charge(amount: Money, customerId: string, metadata?: Record<string, string>): Promise<PaymentResult>;
  refund(paymentId: string, amount?: Money): Promise<RefundResult>;
}

interface PaymentResult {
  id: string;
  status: 'succeeded' | 'pending' | 'failed' | 'requires_action';
  actionUrl?: string; // For 3D Secure redirects
  error?: string;
}

// Stripe adapter — ALL Stripe-specific code lives HERE
class StripePaymentGateway implements PaymentGateway {
  constructor(private readonly stripe: Stripe) {}

  async charge(amount: Money, customerId: string, metadata?: Record<string, string>): Promise<PaymentResult> {
    try {
      const intent = await this.stripe.paymentIntents.create({
        amount: amount.cents,
        currency: amount.currency.toLowerCase(),
        customer: customerId,
        metadata,
      });
      return this.mapResult(intent);
    } catch (error) {
      return this.mapError(error);
    }
  }

  // Translation layer: Stripe → YOUR domain
  private mapResult(intent: Stripe.PaymentIntent): PaymentResult {
    return {
      id: intent.id,
      status: this.mapStatus(intent.status),
      actionUrl: intent.status === 'requires_action'
        ? intent.next_action?.redirect_to_url?.url
        : undefined,
    };
  }

  private mapStatus(stripeStatus: Stripe.PaymentIntent.Status): PaymentResult['status'] {
    const mapping: Record<string, PaymentResult['status']> = {
      'succeeded': 'succeeded',
      'processing': 'pending',
      'requires_payment_method': 'failed',
      'requires_confirmation': 'pending',
      'requires_action': 'requires_action',
      'canceled': 'failed',
    };
    return mapping[stripeStatus] ?? 'failed';
  }

  private mapError(error: unknown): PaymentResult {
    if (error instanceof Stripe.errors.StripeCardError) {
      return { id: '', status: 'failed', error: error.message };
    }
    throw error; // Re-throw unexpected errors
  }
}

// Business logic: ZERO knowledge of Stripe
class OrderService {
  constructor(private readonly payment: PaymentGateway) {}

  async processPayment(order: Order): Promise<void> {
    const result = await this.payment.charge(order.total, order.customerId);
    // Uses YOUR domain types, not Stripe types
    if (result.status === 'succeeded') {
      order.markPaid(result.id);
    } else if (result.status === 'requires_action') {
      order.markPendingAction(result.actionUrl!);
    } else {
      order.markPaymentFailed(result.error ?? 'Unknown error');
    }
  }
}

// Đổi từ Stripe sang Adyen? Tạo AdyenPaymentGateway, swap in DI container.
// OrderService KHÔNG thay đổi.
```

### Backend for Frontend (BFF)

```typescript
// Problem: Mobile, Web, Admin cần data khác nhau từ cùng services
// Solution: BFF — tailored API per client type

// Web BFF: full data, server-rendered
class WebBFF {
  async getProductPage(productId: string): Promise<WebProductPage> {
    const [product, reviews, recommendations, inventory] = await Promise.all([
      this.catalogService.getProduct(productId),
      this.reviewService.getReviews(productId, { limit: 20 }),
      this.recommendationService.getRelated(productId, { limit: 8 }),
      this.inventoryService.getStock(productId),
    ]);

    return {
      product: this.enrichWithSEO(product),
      reviews: reviews.items,
      reviewsSummary: reviews.summary,
      recommendations,
      inStock: inventory.available > 0,
      estimatedDelivery: inventory.estimatedDelivery,
      breadcrumbs: await this.buildBreadcrumbs(product.categoryId),
    };
  }
}

// Mobile BFF: minimal data, bandwidth-optimized
class MobileBFF {
  async getProductPage(productId: string): Promise<MobileProductPage> {
    const [product, inventory] = await Promise.all([
      this.catalogService.getProduct(productId),
      this.inventoryService.getStock(productId),
    ]);

    return {
      product: {
        id: product.id,
        name: product.name,
        price: product.price,
        imageUrl: product.images[0]?.thumbnailUrl, // Smaller image
      },
      inStock: inventory.available > 0,
      // No reviews, no recommendations (lazy load on scroll)
    };
  }
}
```

## Webhook Patterns

```typescript
// Receiving webhooks from external services (Stripe, GitHub, etc.)

// 1. VERIFY webhook signature (CRITICAL — prevent spoofing)
function verifyStripeWebhook(req: Request, secret: string): Stripe.Event {
  const signature = req.headers['stripe-signature'] as string;
  try {
    return stripe.webhooks.constructEvent(req.rawBody, signature, secret);
  } catch {
    throw new UnauthorizedError('Invalid webhook signature');
  }
}

// 2. Respond FAST (200 OK), process ASYNC
app.post('/webhooks/stripe', express.raw({ type: 'application/json' }), async (req, res) => {
  const event = verifyStripeWebhook(req, config.stripeWebhookSecret);

  // Respond immediately — Stripe retries if no 200 within 30s
  res.status(200).json({ received: true });

  // Process asynchronously
  await webhookQueue.add('stripe', {
    eventId: event.id,
    type: event.type,
    data: event.data,
  });
});

// 3. Process idempotently (webhooks may be delivered multiple times)
class WebhookProcessor {
  async processStripeEvent(event: { eventId: string; type: string; data: unknown }): Promise<void> {
    // Idempotency check
    const processed = await this.db('processed_webhooks')
      .where('event_id', event.eventId)
      .first();
    if (processed) return; // Already handled

    // Process
    switch (event.type) {
      case 'payment_intent.succeeded':
        await this.handlePaymentSuccess(event.data);
        break;
      case 'customer.subscription.updated':
        await this.handleSubscriptionUpdate(event.data);
        break;
    }

    // Mark as processed
    await this.db('processed_webhooks').insert({
      event_id: event.eventId,
      event_type: event.type,
      processed_at: new Date(),
    });
  }
}

// 4. Sending webhooks to YOUR customers
class WebhookSender {
  async send(subscription: WebhookSubscription, event: WebhookEvent): Promise<void> {
    const payload = JSON.stringify(event);
    const signature = this.sign(payload, subscription.secret);

    const response = await fetch(subscription.url, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'X-Webhook-Signature': signature,
        'X-Webhook-Id': event.id,
        'X-Webhook-Timestamp': event.timestamp,
      },
      body: payload,
      signal: AbortSignal.timeout(10_000), // 10s timeout
    });

    if (!response.ok) {
      // Schedule retry with exponential backoff
      await this.scheduleRetry(subscription, event, response.status);
    }
  }

  private sign(payload: string, secret: string): string {
    return crypto.createHmac('sha256', secret).update(payload).digest('hex');
  }
}
```

## Third-Party Integration Best Practices

```typescript
// Every external dependency = potential failure point

class ExternalServiceClient {
  constructor(
    private readonly httpClient: HttpClient,
    private readonly circuitBreaker: CircuitBreaker,
    private readonly cache: Cache,
    private readonly logger: Logger,
  ) {}

  async getExchangeRate(from: string, to: string): Promise<number> {
    const cacheKey = `exchange:${from}:${to}`;

    // 1. Try cache first (stale data better than no data)
    const cached = await this.cache.get<number>(cacheKey);
    if (cached !== null) return cached;

    // 2. Call external API with circuit breaker + timeout
    try {
      const rate = await this.circuitBreaker.execute(() =>
        this.httpClient.get<ExchangeRateResponse>(
          `https://api.exchangerate.com/v1/rates/${from}/${to}`,
          { timeout: 3000 },
        ),
      );

      // 3. Cache successful response
      await this.cache.set(cacheKey, rate.data.rate, 300); // 5 min cache
      return rate.data.rate;
    } catch (error) {
      this.logger.warn('Exchange rate API failed, using fallback', { from, to, error: String(error) });

      // 4. Fallback: stale cache (longer TTL)
      const staleRate = await this.cache.get<number>(`exchange:stale:${from}:${to}`);
      if (staleRate !== null) return staleRate;

      // 5. Ultimate fallback: hardcoded rates (updated monthly)
      return this.getHardcodedRate(from, to);
    }
  }
}

// Checklist cho mỗi third-party integration:
// □ Circuit breaker wrapped?
// □ Timeout configured?
// □ Retry policy defined?
// □ Fallback behavior defined?
// □ Rate limits respected?
// □ API key rotatable?
// □ Webhook signature verified?
// □ Error responses mapped to your domain errors?
// □ Anti-corruption layer in place?
// □ Monitoring/alerting for API failures?
```

## Data Synchronization

```typescript
// Keeping data in sync across systems

// === Change Data Capture (CDC) ===
// Database changes → stream → other systems

// Debezium (PostgreSQL → Kafka)
// PostgreSQL WAL → Debezium → Kafka topic → Consumers

// Consumer: sync to Elasticsearch
class SearchIndexSync {
  async handleChange(change: CDCEvent): Promise<void> {
    switch (change.operation) {
      case 'INSERT':
      case 'UPDATE':
        await this.elasticsearch.index({
          index: 'products',
          id: change.after.id,
          body: this.mapToSearchDocument(change.after),
        });
        break;
      case 'DELETE':
        await this.elasticsearch.delete({
          index: 'products',
          id: change.before.id,
        });
        break;
    }
  }
}

// === Periodic Sync (simpler, for non-real-time) ===
class PeriodicSync {
  async syncUsersToMailingList(): Promise<SyncResult> {
    const lastSync = await this.getLastSyncTimestamp('mailing-list');
    const changedUsers = await this.db('users')
      .where('updated_at', '>', lastSync)
      .select('id', 'email', 'name', 'preferences');

    let synced = 0;
    let failed = 0;

    for (const user of changedUsers) {
      try {
        await this.mailingListApi.upsertSubscriber({
          email: user.email,
          name: user.name,
          tags: user.preferences.emailTags,
        });
        synced++;
      } catch {
        failed++;
      }
    }

    await this.setLastSyncTimestamp('mailing-list', new Date());
    return { synced, failed, total: changedUsers.length };
  }
}
```

## ETL & Data Pipelines

```typescript
// Extract → Transform → Load pipeline

interface PipelineStep<TIn, TOut> {
  name: string;
  process(input: TIn): Promise<TOut>;
}

class DataPipeline<TSource, TTarget> {
  private steps: PipelineStep<any, any>[] = [];

  addStep<TIn, TOut>(step: PipelineStep<TIn, TOut>): DataPipeline<TSource, TOut> {
    this.steps.push(step);
    return this as any;
  }

  async run(source: AsyncIterable<TSource>): Promise<PipelineResult> {
    const result = { processed: 0, failed: 0, errors: [] as Error[] };

    for await (const item of source) {
      try {
        let current: any = item;
        for (const step of this.steps) {
          current = await step.process(current);
          if (current === null) break; // Filtered out
        }
        if (current !== null) result.processed++;
      } catch (error) {
        result.failed++;
        result.errors.push(error as Error);
      }
    }

    return result;
  }
}

// Usage: CRM data sync pipeline
const pipeline = new DataPipeline()
  .addStep({
    name: 'validate',
    async process(record: CrmRecord) {
      if (!record.email) return null; // Skip invalid
      return record;
    },
  })
  .addStep({
    name: 'transform',
    async process(record: CrmRecord) {
      return {
        externalId: record.id,
        email: record.email.toLowerCase(),
        name: `${record.firstName} ${record.lastName}`.trim(),
        company: record.organization,
        lastContactedAt: new Date(record.lastActivity),
      };
    },
  })
  .addStep({
    name: 'load',
    async process(contact: TransformedContact) {
      await db('contacts')
        .insert(contact)
        .onConflict('external_id')
        .merge();
      return contact;
    },
  });

const result = await pipeline.run(crmApi.streamContacts());
console.log(`Synced: ${result.processed}, Failed: ${result.failed}`);
```
