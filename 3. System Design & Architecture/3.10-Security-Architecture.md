## Security Mindset

### Defense in Depth

```
Không có 1 layer nào "đủ an toàn". Mỗi layer là 1 lớp bảo vệ.

  Internet
  ├── WAF / DDoS protection (Cloudflare)
  ├── API Gateway (rate limiting, IP filtering)
  ├── Authentication (JWT, session)
  ├── Authorization (RBAC, ABAC)
  ├── Input validation (Zod, joi)
  ├── Business logic checks (domain rules)
  ├── Database constraints (CHECK, FK, UNIQUE)
  ├── Encryption at rest (disk encryption, column encryption)
  └── Audit logging (who did what when)

Mỗi layer assume layers trên ĐÃ BỊ BYPASS.
```

## Authentication

### JWT Best Practices

```typescript
// JWT: stateless authentication token

interface JwtPayload {
  sub: string;        // User ID
  role: string;       // User role
  iat: number;        // Issued at
  exp: number;        // Expiration
  jti: string;        // JWT ID (for revocation)
}

class TokenService {
  constructor(
    private readonly accessSecret: string,
    private readonly refreshSecret: string,
  ) {
    // Validate secrets at startup
    if (accessSecret.length < 32) throw new Error('Access secret too short (min 32 chars)');
    if (refreshSecret.length < 32) throw new Error('Refresh secret too short (min 32 chars)');
  }

  generateTokenPair(user: User): TokenPair {
    const jti = crypto.randomUUID();

    const accessToken = jwt.sign(
      { sub: user.id, role: user.role, jti },
      this.accessSecret,
      {
        expiresIn: '15m',       // Short-lived: 15 minutes
        algorithm: 'HS256',
      },
    );

    const refreshToken = jwt.sign(
      { sub: user.id, jti },
      this.refreshSecret,
      {
        expiresIn: '7d',        // Longer-lived: 7 days
        algorithm: 'HS256',
      },
    );

    return { accessToken, refreshToken, expiresIn: 900 };
  }

  verifyAccessToken(token: string): JwtPayload {
    try {
      return jwt.verify(token, this.accessSecret, {
        algorithms: ['HS256'],  // Whitelist algorithms (prevent "none" attack)
      }) as JwtPayload;
    } catch (error) {
      if (error instanceof jwt.TokenExpiredError) {
        throw new UnauthorizedError('Token expired');
      }
      throw new UnauthorizedError('Invalid token');
    }
  }

  async refresh(refreshToken: string, tokenStore: TokenStore): Promise<TokenPair> {
    const payload = jwt.verify(refreshToken, this.refreshSecret, {
      algorithms: ['HS256'],
    }) as JwtPayload;

    // Check if refresh token has been revoked
    const isRevoked = await tokenStore.isRevoked(payload.jti);
    if (isRevoked) throw new UnauthorizedError('Token has been revoked');

    // Revoke old refresh token (rotation)
    await tokenStore.revoke(payload.jti);

    // Issue new pair
    const user = await this.userRepo.findById(payload.sub);
    if (!user) throw new UnauthorizedError('User not found');

    return this.generateTokenPair(user);
  }
}

// Token revocation store (Redis)
class RedisTokenStore implements TokenStore {
  constructor(private readonly redis: Redis) {}

  async revoke(jti: string, ttlSeconds = 7 * 86400): Promise<void> {
    await this.redis.set(`revoked:${jti}`, '1', 'EX', ttlSeconds);
  }

  async isRevoked(jti: string): Promise<boolean> {
    return (await this.redis.exists(`revoked:${jti}`)) === 1;
  }
}
```

### Auth Middleware

```typescript
function authMiddleware(tokenService: TokenService): Middleware {
  return async (req, res, next) => {
    const header = req.headers.authorization;
    if (!header?.startsWith('Bearer ')) {
      throw new UnauthorizedError('Missing Bearer token');
    }

    const token = header.slice(7);
    const payload = tokenService.verifyAccessToken(token);

    req.user = {
      id: payload.sub,
      role: payload.role,
    };

    next();
  };
}

// ❌ email, phone, address, permissions list
// ✅ user ID, role (minimal claims, lookup details from DB if needed)
```

### Password Hashing

```typescript
import { hash, verify } from '@node-rs/argon2';

class PasswordService {
  // Argon2id: recommended by OWASP
  // bcrypt: acceptable alternative
  // ❌ NEVER: MD5, SHA-1, SHA-256 (not designed for passwords)

  async hash(plaintext: string): Promise<string> {
    return hash(plaintext, {
      memoryCost: 65536,    // 64 MB
      timeCost: 3,          // 3 iterations
      parallelism: 4,       // 4 parallel threads
    });
  }

  async verify(plaintext: string, hashed: string): Promise<boolean> {
    return verify(hashed, plaintext);
  }
}

// Password policy
const PasswordPolicy = {
  minLength: 8,
  maxLength: 128,         // Prevent DoS via extremely long passwords
  requireUppercase: true,
  requireLowercase: true,
  requireDigit: true,
  requireSpecial: false,  // NIST 800-63B: special chars NOT required
  // NIST recommends: check against breached password lists instead
};

// Check against breached passwords (Have I Been Pwned API)
async function isBreachedPassword(password: string): Promise<boolean> {
  const sha1 = crypto.createHash('sha1').update(password).digest('hex').toUpperCase();
  const prefix = sha1.slice(0, 5);
  const suffix = sha1.slice(5);

  const response = await fetch(`https://api.pwnedpasswords.com/range/${prefix}`);
  const text = await response.text();

  return text.split('\n').some(line => line.startsWith(suffix));
}
```

## Authorization

### Role-Based Access Control (RBAC)

```typescript
// Simple RBAC cho phần lớn apps

type Role = 'viewer' | 'editor' | 'admin' | 'super_admin';
type Permission = 'read' | 'create' | 'update' | 'delete' | 'manage_users' | 'manage_billing';

const ROLE_PERMISSIONS: Record<Role, ReadonlySet<Permission>> = {
  viewer: new Set(['read']),
  editor: new Set(['read', 'create', 'update']),
  admin: new Set(['read', 'create', 'update', 'delete', 'manage_users']),
  super_admin: new Set(['read', 'create', 'update', 'delete', 'manage_users', 'manage_billing']),
};

function hasPermission(role: Role, permission: Permission): boolean {
  return ROLE_PERMISSIONS[role]?.has(permission) ?? false;
}

// Middleware
function requirePermission(...permissions: Permission[]): Middleware {
  return (req, _res, next) => {
    const userRole = req.user?.role as Role;
    if (!userRole) throw new UnauthorizedError('Not authenticated');

    const hasAll = permissions.every(p => hasPermission(userRole, p));
    if (!hasAll) {
      throw new ForbiddenError(`Missing permissions: ${permissions.join(', ')}`);
    }

    next();
  };
}

// Usage
router.delete('/users/:id',
  authMiddleware(tokenService),
  requirePermission('delete', 'manage_users'),
  userController.delete,
);
```

### Attribute-Based Access Control (ABAC)

```typescript
// ABAC: fine-grained, khi RBAC không đủ

interface PolicyContext {
  user: { id: string; role: Role; organizationId: string; department: string };
  resource: { id: string; type: string; ownerId: string; organizationId: string };
  action: string;
  environment: { time: Date; ipAddress: string };
}

type PolicyRule = (ctx: PolicyContext) => 'allow' | 'deny' | 'abstain';

const policies: PolicyRule[] = [
  // Users can only access resources in their own organization
  (ctx) => ctx.user.organizationId !== ctx.resource.organizationId ? 'deny' : 'abstain',

  // Resource owners can always edit their own resources
  (ctx) => ctx.user.id === ctx.resource.ownerId && ctx.action === 'update' ? 'allow' : 'abstain',

  // Admins can do anything in their org
  (ctx) => ctx.user.role === 'admin' ? 'allow' : 'abstain',

  // No access outside business hours for non-admins (example)
  (ctx) => {
    const hour = ctx.environment.time.getHours();
    if (ctx.user.role !== 'admin' && (hour < 6 || hour > 22)) return 'deny';
    return 'abstain';
  },
];

function evaluatePolicy(ctx: PolicyContext): boolean {
  for (const rule of policies) {
    const result = rule(ctx);
    if (result === 'deny') return false;
    if (result === 'allow') return true;
  }
  return false; // Default deny
}
```

## Input Validation & Sanitization

```typescript
// VALIDATE everything at system boundaries. Trust NOTHING from outside.

import { z } from 'zod';

// ✅ Strict schema validation with Zod
const CreateUserSchema = z.object({
  name: z.string()
    .trim()
    .min(2, 'Name must be at least 2 characters')
    .max(100, 'Name must be at most 100 characters')
    .regex(/^[\p{L}\p{M}\s'-]+$/u, 'Name contains invalid characters'), // Unicode letters only

  email: z.string()
    .trim()
    .email('Invalid email')
    .max(255)
    .transform(v => v.toLowerCase()),

  password: z.string()
    .min(8)
    .max(128), // MAX length prevents DoS via hashing extremely long strings

  role: z.enum(['viewer', 'editor']).default('viewer'),
  // ❌ NEVER allow client to set 'admin' role
});

// SQL Injection prevention
// ✅ ALWAYS use parameterized queries
await db.query('SELECT * FROM users WHERE email = $1', [email]);

// ❌ NEVER interpolate user input into SQL
await db.query(`SELECT * FROM users WHERE email = '${email}'`); // SQL INJECTION!

// XSS prevention
// ✅ When rendering user content in HTML:
import DOMPurify from 'isomorphic-dompurify';

function sanitizeHtml(dirty: string): string {
  return DOMPurify.sanitize(dirty, {
    ALLOWED_TAGS: ['b', 'i', 'em', 'strong', 'a', 'p', 'br'],
    ALLOWED_ATTR: ['href', 'title'],
  });
}

// Path Traversal prevention
function safePath(userInput: string, baseDir: string): string {
  const resolved = path.resolve(baseDir, userInput);
  if (!resolved.startsWith(path.resolve(baseDir))) {
    throw new Error('Path traversal attempt detected');
  }
  return resolved;
}
```

## Data Protection

### Encryption

```typescript
// Encryption at rest: database-level (RDS encryption, disk encryption)
// Encryption in transit: TLS everywhere (HTTPS, encrypted DB connections)
// Encryption at application level: sensitive columns

import { createCipheriv, createDecipheriv, randomBytes, scryptSync } from 'crypto';

class FieldEncryption {
  private readonly algorithm = 'aes-256-gcm';
  private readonly key: Buffer;

  constructor(encryptionKey: string) {
    // Derive key from password using scrypt
    this.key = scryptSync(encryptionKey, 'salt-should-be-unique', 32);
  }

  encrypt(plaintext: string): string {
    const iv = randomBytes(16);
    const cipher = createCipheriv(this.algorithm, this.key, iv);

    let encrypted = cipher.update(plaintext, 'utf8', 'hex');
    encrypted += cipher.final('hex');

    const authTag = cipher.getAuthTag();

    // Store: iv:authTag:encrypted
    return `${iv.toString('hex')}:${authTag.toString('hex')}:${encrypted}`;
  }

  decrypt(ciphertext: string): string {
    const [ivHex, authTagHex, encrypted] = ciphertext.split(':');

    const iv = Buffer.from(ivHex, 'hex');
    const authTag = Buffer.from(authTagHex, 'hex');
    const decipher = createDecipheriv(this.algorithm, this.key, iv);
    decipher.setAuthTag(authTag);

    let decrypted = decipher.update(encrypted, 'hex', 'utf8');
    decrypted += decipher.final('utf8');

    return decrypted;
  }
}

// Usage: encrypt PII in database
const encryption = new FieldEncryption(config.FIELD_ENCRYPTION_KEY);

// Save
await db.query(
  'INSERT INTO users (id, name, email_encrypted, ssn_encrypted) VALUES ($1, $2, $3, $4)',
  [user.id, user.name, encryption.encrypt(user.email), encryption.encrypt(user.ssn)],
);

// Read
const row = await db.query('SELECT * FROM users WHERE id = $1', [userId]);
const email = encryption.decrypt(row.email_encrypted);
```

## Common Vulnerabilities

### OWASP Top 10 Checklist cho TypeScript

| # | Vulnerability | Prevention |
|---|--------------|-----------|
| 1 | **Broken Access Control** | AuthZ checks on EVERY endpoint, deny by default |
| 2 | **Cryptographic Failures** | TLS everywhere, encrypt PII, strong hashing (argon2/bcrypt) |
| 3 | **Injection** | Parameterized queries, Zod validation, no eval() |
| 4 | **Insecure Design** | Threat modeling, abuse cases, rate limiting |
| 5 | **Security Misconfiguration** | Helmet headers, no default credentials, minimal permissions |
| 6 | **Vulnerable Components** | `npm audit`, Dependabot, pin versions |
| 7 | **Auth Failures** | MFA, account lockout, breached password check |
| 8 | **Data Integrity Failures** | Input validation, SSRF protection, SRI for CDN scripts |
| 9 | **Logging Failures** | Audit logs for auth events, anomaly detection |
| 10 | **SSRF** | Allowlist outbound URLs, block internal IPs |

## Audit & Compliance

```typescript
// Audit log: immutable record of WHO did WHAT to WHICH resource WHEN

interface AuditEntry {
  readonly id: string;
  readonly timestamp: Date;
  readonly actor: {
    readonly userId: string;
    readonly role: string;
    readonly ipAddress: string;
    readonly userAgent: string;
  };
  readonly action: string;           // 'user.created', 'order.cancelled'
  readonly resource: {
    readonly type: string;
    readonly id: string;
  };
  readonly changes?: {
    readonly before: Record<string, unknown>;
    readonly after: Record<string, unknown>;
  };
  readonly metadata?: Record<string, unknown>;
}

class AuditLogger {
  constructor(private readonly db: Knex) {}

  async log(entry: Omit<AuditEntry, 'id' | 'timestamp'>): Promise<void> {
    await this.db('audit_log').insert({
      id: crypto.randomUUID(),
      timestamp: new Date(),
      actor_user_id: entry.actor.userId,
      actor_role: entry.actor.role,
      actor_ip: entry.actor.ipAddress,
      actor_user_agent: entry.actor.userAgent,
      action: entry.action,
      resource_type: entry.resource.type,
      resource_id: entry.resource.id,
      changes: entry.changes ? JSON.stringify(entry.changes) : null,
      metadata: entry.metadata ? JSON.stringify(entry.metadata) : null,
    });
  }
}

// Middleware: auto-audit for sensitive operations
function auditMiddleware(auditLogger: AuditLogger, action: string, resourceType: string): Middleware {
  return async (req, res, next) => {
    const originalJson = res.json.bind(res);

    res.json = (body: any) => {
      // Log after successful response
      if (res.statusCode < 400) {
        auditLogger.log({
          actor: {
            userId: req.user?.id ?? 'anonymous',
            role: req.user?.role ?? 'none',
            ipAddress: req.ip ?? 'unknown',
            userAgent: req.headers['user-agent'] ?? 'unknown',
          },
          action,
          resource: {
            type: resourceType,
            id: req.params.id ?? body?.id ?? 'unknown',
          },
        }).catch(console.error); // Don't fail request if audit fails
      }

      return originalJson(body);
    };

    next();
  };
}

// Usage
router.delete('/users/:id',
  authMiddleware(tokenService),
  requirePermission('delete', 'manage_users'),
  auditMiddleware(auditLogger, 'user.deleted', 'user'),
  userController.delete,
);

// Audit log table: APPEND-ONLY, no updates, no deletes
// CREATE TABLE audit_log (
//   id UUID PRIMARY KEY,
//   timestamp TIMESTAMPTZ NOT NULL DEFAULT NOW(),
//   ...
// );
// REVOKE UPDATE, DELETE ON audit_log FROM app_user;
```
