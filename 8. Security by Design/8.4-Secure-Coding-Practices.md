### 2. A02: Cryptographic Failures

```typescript
import crypto from 'crypto';
import bcrypt from 'bcrypt';

// VULNERABLE: MD5/SHA1 for passwords, hardcoded key, ECB mode
const bad1 = crypto.createHash('md5').update(password).digest('hex');
const bad2 = crypto.createCipheriv('aes-128-ecb', 'hardcoded!!!!!!!', null);

// MITIGATED: bcrypt for passwords
async function hashPassword(pw: string): Promise<string> { return bcrypt.hash(pw, 12); }
async function verifyPassword(pw: string, hash: string): Promise<boolean> { return bcrypt.compare(pw, hash); }

// MITIGATED: AES-256-GCM with proper key management
class EncryptionService {
  constructor(private key: Buffer) {
    if (key.length !== 32) throw new Error('Key must be 256 bits');
  }

  encrypt(plaintext: string): { ciphertext: string; iv: string; tag: string } {
    const iv = crypto.randomBytes(16);
    const cipher = crypto.createCipheriv('aes-256-gcm', this.key, iv);
    let enc = cipher.update(plaintext, 'utf8', 'base64');
    enc += cipher.final('base64');
    return { ciphertext: enc, iv: iv.toString('base64'), tag: cipher.getAuthTag().toString('base64') };
  }

  decrypt(data: { ciphertext: string; iv: string; tag: string }): string {
    const decipher = crypto.createDecipheriv('aes-256-gcm', this.key, Buffer.from(data.iv, 'base64'));
    decipher.setAuthTag(Buffer.from(data.tag, 'base64'));
    let dec = decipher.update(data.ciphertext, 'base64', 'utf8');
    return dec + decipher.final('utf8');
  }
}
```

```typescript
// Brute force protection with exponential backoff
class LoginRateLimiter {
  constructor(private redis: Redis) {}

  async checkAndIncrement(identifier: string): Promise<{ allowed: boolean; retryAfter?: number }> {
    const key = `login_attempts:${identifier}`;
    const attempts = parseInt(await this.redis.get(key) ?? '0', 10);

    if (attempts >= 5) {
      const lockout = Math.min(Math.pow(2, attempts - 5), 3600);
      const ttl = await this.redis.ttl(key);
      if (ttl > 0) return { allowed: false, retryAfter: ttl };
    }
    await this.redis.multi().incr(key).expire(key, 3600).exec();
    return { allowed: true };
  }

  async resetOnSuccess(identifier: string): Promise<void> {
    await this.redis.del(`login_attempts:${identifier}`);
  }
}
```

### 9. A09: Security Logging and Monitoring Failures

```typescript
type SecurityEvent = 'auth.login.success' | 'auth.login.failure' | 'auth.logout' |
  'auth.password_change' | 'auth.mfa_enabled' | 'auth.token_revoked' |
  'authz.access_denied' | 'authz.privilege_escalation_attempt' |
  'data.export' | 'data.bulk_delete' |
  'admin.user_created' | 'admin.role_changed' |
  'api.rate_limit_exceeded';

interface SecurityLogEntry {
  event: SecurityEvent;
  severity: 'info' | 'warn' | 'error' | 'critical';
  timestamp: Date;
  actor: { id: string; type: 'user' | 'service'; ip?: string };
  target?: { type: string; id: string };
  details: Record<string, unknown>;
  requestId: string;
}

class SecurityLogger {
  constructor(private logger: Logger, private alertService: AlertService) {}

  async log(entry: SecurityLogEntry): Promise<void> {
    this.logger.log({ level: entry.severity, message: `[SECURITY] ${entry.event}`,
      ...entry, details: this.redact(entry.details) });
    if (entry.severity === 'critical') {
      await this.alertService.sendAlert({ channel: 'security', title: entry.event, details: entry });
    }
  }

  private redact(details: Record<string, unknown>): Record<string, unknown> {
    const sensitive = ['password', 'token', 'secret', 'apiKey', 'ssn', 'creditCard'];
    const result = { ...details };
    for (const key of Object.keys(result)) {
      if (sensitive.some(s => key.toLowerCase().includes(s))) result[key] = '[REDACTED]';
    }
    return result;
  }
}
```

### 10. A10: Server-Side Request Forgery (SSRF)

```typescript
import dns from 'dns/promises';

async function validateUrl(urlStr: string): Promise<{ safe: boolean; reason?: string }> {
  let parsed: URL;
  try { parsed = new URL(urlStr); } catch { return { safe: false, reason: 'Invalid URL' }; }

  if (parsed.protocol !== 'https:') return { safe: false, reason: 'Only HTTPS' };

  const blocked = ['localhost', '127.0.0.1', '0.0.0.0', '169.254.169.254', 'metadata.google.internal'];
  if (blocked.includes(parsed.hostname)) return { safe: false, reason: 'Blocked hostname' };

  // Resolve DNS — check for internal IPs
  try {
    const addrs = await dns.resolve4(parsed.hostname);
    for (const ip of addrs) {
      const p = ip.split('.').map(Number);
      const isInternal = p[0] === 10 || (p[0] === 172 && p[1] >= 16 && p[1] <= 31) ||
        (p[0] === 192 && p[1] === 168) || p[0] === 127 || (p[0] === 169 && p[1] === 254);
      if (isInternal) return { safe: false, reason: 'Resolves to internal IP' };
    }
  } catch { return { safe: false, reason: 'DNS resolution failed' }; }

  return { safe: true };
}

app.post('/api/fetch-url', authenticate, async (req, res) => {
  const validation = await validateUrl(req.body.url);
  if (!validation.safe) return res.status(400).json({ error: validation.reason });

  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), 10_000);
  try {
    const response = await fetch(req.body.url, {
      signal: controller.signal, redirect: 'error',
    });
    const data = await response.text();
    res.json({ data: data.slice(0, 1024 * 1024) }); // 1MB limit
  } catch { res.status(500).json({ error: 'Fetch failed' }); }
  finally { clearTimeout(timeout); }
});
```

## File Upload Security

```typescript
import multer from 'multer';

const upload = multer({
  storage: multer.memoryStorage(), // Don't write to disk before validation
  limits: { fileSize: 10 * 1024 * 1024, files: 5 },
  fileFilter: (req, file, cb) => {
    const allowed = ['image/jpeg', 'image/png', 'image/webp', 'application/pdf'];
    cb(null, allowed.includes(file.mimetype));
  },
});

app.post('/api/upload', authenticate, upload.single('file'), async (req, res) => {
  if (!req.file) return res.status(400).json({ error: 'No file' });

  // Verify actual type via magic bytes (not Content-Type header)
  const { fileTypeFromBuffer } = await import('file-type');
  const detected = await fileTypeFromBuffer(req.file.buffer);
  if (!detected || !['image/jpeg', 'image/png', 'application/pdf'].includes(detected.mime)) {
    return res.status(400).json({ error: 'File content mismatch' });
  }

  // Safe filename — never use user-provided name on disk
  const safeFilename = `${crypto.randomUUID()}.${detected.ext}`;

  // Upload to S3, NOT local filesystem
  await s3.putObject({
    Bucket: process.env.S3_BUCKET!, Key: `uploads/${req.user.orgId}/${safeFilename}`,
    Body: req.file.buffer, ContentType: detected.mime,
    ContentDisposition: 'attachment', // Force download, prevent XSS
    ServerSideEncryption: 'AES256',
  });

  res.json({ id: safeFilename });
});
```

## Secrets Management

```typescript
// Rules: NEVER hardcode | NEVER commit to git | NEVER log

// AWS Secrets Manager
import { SecretsManagerClient, GetSecretValueCommand } from '@aws-sdk/client-secrets-manager';

class SecretsService {
  private client: SecretsManagerClient;
  private cache = new Map<string, { value: string; expiresAt: number }>();

  constructor(region: string) { this.client = new SecretsManagerClient({ region }); }

  async getSecret(secretId: string): Promise<string> {
    const cached = this.cache.get(secretId);
    if (cached && cached.expiresAt > Date.now()) return cached.value;

    const resp = await this.client.send(new GetSecretValueCommand({ SecretId: secretId }));
    const value = resp.SecretString!;
    this.cache.set(secretId, { value, expiresAt: Date.now() + 300_000 }); // 5min cache
    return value;
  }
}
```

## Security Headers Comprehensive Guide

```typescript
app.use((req, res, next) => {
  // Force HTTPS (1 year, include subdomains, preload)
```

## Production Security Checklist

```
INPUT VALIDATION
  [ ] All input validated with schema (Zod)
  [ ] SQL: parameterized queries / ORM
  [ ] File uploads verified via magic bytes
  [ ] URLs validated for SSRF

AUTHENTICATION
  [ ] Passwords: bcrypt/argon2 (cost >= 12)
  [ ] JWT: verified properly (algorithm, issuer, audience)
  [ ] Rate limiting on auth endpoints
  [ ] MFA available for users

AUTHORIZATION
  [ ] Every endpoint has auth check
  [ ] IDOR prevention (verify ownership)
  [ ] Tenant isolation enforced
  [ ] Admin endpoints protected

DATA PROTECTION
  [ ] Encrypted at rest and in transit
  [ ] Sensitive fields redacted in logs
  [ ] PII handled per compliance

INFRASTRUCTURE
  [ ] Security headers (CSP, HSTS, etc.)
  [ ] CORS restricted to allowlist
  [ ] Dependencies audited (npm audit, Snyk)
  [ ] Secrets in secrets manager
  [ ] Pre-commit hooks prevent secret commits

MONITORING
  [ ] Security events logged
  [ ] Failed auth tracked + alerted
  [ ] Incident response plan documented
```
