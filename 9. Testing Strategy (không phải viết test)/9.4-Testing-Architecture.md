## Designing Code for Testability

### Nguyên tắc #1: Dependency Injection

```typescript
// ❌ UNTESTABLE — hardcoded dependencies
class OrderService {
  async createOrder(input: CreateOrderInput): Promise<Order> {
    const db = new PostgresClient(process.env.DATABASE_URL!);  // hardcoded
    const stripe = new Stripe(process.env.STRIPE_KEY!);        // hardcoded
    const product = await db.query('SELECT * FROM products WHERE id = $1', [input.productId]);
    const charge = await stripe.charges.create({ amount: product.price });
    return { id: randomUUID(), productId: input.productId, chargeId: charge.id };
  }
}

// ✅ TESTABLE — dependencies injected via interfaces
interface ProductRepository {
  findById(id: string): Promise<Product | null>;
}
interface PaymentGateway {
  charge(amount: Money, customerId: string): Promise<PaymentResult>;
}

class OrderService {
  constructor(
    private readonly productRepo: ProductRepository,
    private readonly paymentGateway: PaymentGateway,
    private readonly logger: Logger,
  ) {}
  async createOrder(input: CreateOrderInput): Promise<Order> {
    const product = await this.productRepo.findById(input.productId);
    if (!product) throw new ProductNotFoundError(input.productId);
    const payment = await this.paymentGateway.charge(product.price, input.customerId);
    if (!payment.success) throw new PaymentFailedError(payment.reason);
    return Order.create({ productId: product.id, paymentId: payment.id });
  }
}

// Test dễ dàng với bất kỳ implementation nào
const service = new OrderService(
  new FakeProductRepository([testProduct]),
  { charge: jest.fn().mockResolvedValue({ success: true, id: 'pay-1' }) },
  { info: jest.fn(), error: jest.fn() },
);
```

### Nguyên tắc #2: Ports & Adapters

```typescript
// PORTS — interfaces defined by domain (domain KHÔNG biết implementation)
interface OrderPersistence {
  save(order: Order): Promise<void>;
  findById(id: OrderId): Promise<Order | null>;
}

// PRODUCTION ADAPTER
class PostgresOrderPersistence implements OrderPersistence {
  constructor(private readonly pool: Pool) {}
  async save(order: Order): Promise<void> {
    await this.pool.query('INSERT INTO orders ...', [order.id, order.total.toCents()]);
  }
  // ...
}

// TEST ADAPTER — swap at test time, same interface
class InMemoryOrderPersistence implements OrderPersistence {
  private orders = new Map<string, Order>();
  async save(order: Order): Promise<void> { this.orders.set(order.id, structuredClone(order)); }
  async findById(id: OrderId): Promise<Order | null> { return this.orders.get(id) ?? null; }
  // Test helpers
  getAll(): Order[] { return [...this.orders.values()]; }
  count(): number { return this.orders.size; }
}
```

### Nguyên tắc #3: Functional Core, Imperative Shell

```typescript
// CORE — pure function, tất cả business decisions, dễ test vô cùng
function decideOrderApproval(
  order: OrderData, customer: CustomerData, inventory: InventorySnapshot,
): OrderDecision {
  for (const item of order.items) {
    if (inventory.getStock(item.productId) < item.quantity) {
      return { approved: false, reason: `Insufficient stock for ${item.productId}` };
    }
  }
  if (customer.creditUsed + order.total > customer.creditLimit) {
    return { approved: false, reason: 'Credit limit exceeded' };
  }
  if (order.total > 10000 && customer.accountAgeDays < 30) {
    return { approved: false, reason: 'High-value from new account', requiresReview: true };
  }
  return { approved: true };
}

// SHELL — thin orchestrator, delegates decisions to pure core
class OrderApprovalHandler {
  async handle(orderId: string): Promise<void> {
    const order = await this.orderRepo.findById(orderId);      // impure: fetch
    const customer = await this.customerRepo.findById(order.customerId);
    const inventory = await this.inventoryService.getSnapshot(order.items.map(i => i.productId));

    const decision = decideOrderApproval(order, customer, inventory); // PURE

    if (decision.approved) {                                    // impure: act
      await this.orderRepo.approve(orderId);
    } else {
      await this.orderRepo.reject(orderId, decision.reason);
    }
  }
}
// 20 unit tests cho decideOrderApproval (pure, fast)
// 2-3 integration tests cho OrderApprovalHandler (wiring)
```

## CI/CD Test Pipeline Design

```yaml
# .github/workflows/ci.yml
jobs:
  # Layer 1: < 30s — mỗi push
  static-analysis:
    steps:
      - run: npm run typecheck && npm run lint && npm run format:check

  # Layer 2: < 2min — mỗi PR
  unit-tests:
    needs: static-analysis
    steps:
      - run: npm run test:unit -- --coverage

  # Layer 3: < 5min — mỗi PR (parallel với unit)
  integration-tests:
    needs: static-analysis
    services:
      postgres: { image: 'postgres:16-alpine', ports: ['5432:5432'] }
      redis: { image: 'redis:7-alpine', ports: ['6379:6379'] }
    steps:
      - run: npm run db:migrate && npm run test:integration

  # Layer 4: < 3min — mỗi PR
  contract-tests:
    needs: unit-tests
    steps:
      - run: npm run test:contract && npm run pact:publish

  # Layer 5: < 10min — chỉ merge vào main
  e2e-tests:
    needs: [unit-tests, integration-tests]
    if: github.ref == 'refs/heads/main'
    steps:
      - run: npx playwright install --with-deps
      - run: npm run test:e2e

  # Layer 6: nightly only
  performance-tests:
    if: github.event_name == 'schedule'
    steps:
      - run: npm run test:performance
```

### Pipeline principles

```
FAIL FAST: Layer 1 fail → đừng chạy Layer 2+
PARALLEL:  Layer 2, 3, 4 chạy song song
ARTIFACTS: Upload traces/screenshots khi E2E fail
CACHE:     node_modules, Docker layers
FLAKY:     Retry 1 lần cho E2E, KHÔNG retry unit/integration
```

```typescript
  thresholds: {
    'src/domain/**':         { branches: 90, functions: 95, lines: 90 },
    'src/services/**':       { branches: 70, functions: 80, lines: 75 },
    'src/api/**':            { branches: 30, functions: 50, lines: 40 },
    'src/infrastructure/**': { branches: 0 }, // tested via integration

  exclude: ['**/*.test.ts', 'src/generated/**', 'src/migrations/**'],
}

// Metrics QUAN TRỌNG hơn line coverage:
// 1. Branch coverage > Line coverage (if/else paths)
// 2. Mutation score > Coverage percentage
// 3. Flaky test rate (should be 0%)
```

```typescript
class FakeEventBus implements EventBus {
  readonly publishedEvents: DomainEvent[] = [];
  async publish(event: DomainEvent): Promise<void> { this.publishedEvents.push(event); }

    return this.publishedEvents.filter(e => e.type === type) as T[];
```

## Security Testing Integration

```typescript
describe('Security Tests', () => {
  // SQL Injection
  it('rejects SQL injection attempts', async () => {
    const payloads = ["'; DROP TABLE users;--", "1' OR '1'='1", "' UNION SELECT password FROM users--"];
    for (const input of payloads) {
      const res = await request(app).get(`/api/products?search=${encodeURIComponent(input)}`)
        .set('Authorization', `Bearer ${token}`);
      expect(res.status).not.toBe(500);
      expect(JSON.stringify(res.body)).not.toContain('password');
    }
  });

  // XSS
  it('sanitizes HTML in user input', async () => {
    const res = await request(app).post('/api/comments').set('Authorization', `Bearer ${token}`)
      .send({ text: '<script>alert("xss")</script>Hello' });
    expect(res.body.text).not.toContain('<script>');
  });

  // Authentication
  it('rejects expired JWT', async () => {
    const expired = generateToken({ userId: 'u1', expiresIn: '-1h' });
    expect((await request(app).get('/api/profile')
      .set('Authorization', `Bearer ${expired}`)).status).toBe(401);
  });

  // Authorization — horizontal privilege escalation
  it('prevents accessing another user resources', async () => {
    const alice = await factory.createUser();
    const bob = await factory.createUser();
    const bobOrder = await factory.createOrder({ customerId: bob.id });
    const res = await request(app).get(`/api/orders/${bobOrder.id}`)
      .set('Authorization', `Bearer ${alice.token}`);
    expect([403, 404]).toContain(res.status);
  });

  // Rate Limiting
  it('blocks excessive requests', async () => {
    const responses = await Promise.all(
      Array.from({ length: 101 }, () => request(app).get('/api/products')
        .set('Authorization', `Bearer ${token}`)),
    );
    expect(responses.filter(r => r.status === 429).length).toBeGreaterThan(0);
  });
});
```

## Tổng kết

```
1. DESIGN FOR TESTABILITY: DI, Ports/Adapters, Pure Functions, No Global State
   → Code khó test = code cần refactor

2. TEST ENVIRONMENT: Docker Compose + Testcontainers + centralized env config

3. TEST DATA: Factories cho dynamic, fixtures cho reference, mỗi test tạo riêng

4. CI/CD PIPELINE: Layered — static → unit → integration → contract → E2E
   → Fast feedback first, expensive tests later

5. COVERAGE: Domain 90%+, Services 70%+, Infrastructure via integration
   → Branch coverage > Line coverage. Mutation testing cho critical logic.

6. MICROSERVICES: Contract tests BETWEEN, component tests FOR each service

7. EVENT-DRIVEN: Test publishers (correct events), handlers (correct actions),
   idempotency (same event twice = safe)

8. SECURITY & A11Y: SAST + DAST in CI, axe-core for accessibility

Goal: "Deploy bất kỳ lúc nào, confident rằng tests pass = production hoạt động đúng."
```
