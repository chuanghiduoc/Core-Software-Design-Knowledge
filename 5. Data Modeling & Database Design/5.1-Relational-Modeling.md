## 1. Normalization

Normalization là quá trình tổ chức data để **giảm redundancy** và **đảm bảo data integrity**.

### 1NF — First Normal Form

**Quy tắc:** Mỗi column chỉ chứa atomic values. Không có repeating groups.

```sql
-- ❌ Column chứa multiple values
CREATE TABLE orders (
  id SERIAL PRIMARY KEY,
  customer_name TEXT,
  products TEXT  -- "iPhone,MacBook,AirPods" ← KHÔNG atomic
);

-- ✅ Tách thành bảng riêng, mỗi row là 1 giá trị atomic
CREATE TABLE orders (
  id SERIAL PRIMARY KEY,
  customer_name TEXT NOT NULL
);
CREATE TABLE order_items (
  id SERIAL PRIMARY KEY,
  order_id INTEGER NOT NULL REFERENCES orders(id),
  product_name TEXT NOT NULL,
  quantity INTEGER NOT NULL DEFAULT 1,
  unit_price NUMERIC(12,2) NOT NULL
);
```

### 2NF — Second Normal Form

**Quy tắc:** Đã thỏa 1NF + mọi non-key column phụ thuộc **toàn bộ** primary key (không phụ thuộc một phần). Chỉ relevant khi PK là composite key.

```sql
-- ❌ student_name phụ thuộc chỉ vào student_id, KHÔNG phụ thuộc course_id
CREATE TABLE enrollments (
  student_id INTEGER, course_id INTEGER,
  student_name TEXT,       -- ← phụ thuộc partial key
  course_title TEXT,       -- ← phụ thuộc partial key
  PRIMARY KEY (student_id, course_id)
);

-- ✅ Tách thành 3 bảng — mỗi non-key column phụ thuộc toàn bộ PK
CREATE TABLE students (id SERIAL PRIMARY KEY, name TEXT NOT NULL);
CREATE TABLE courses (id SERIAL PRIMARY KEY, title TEXT NOT NULL);
CREATE TABLE enrollments (
  student_id INTEGER REFERENCES students(id),
  course_id INTEGER REFERENCES courses(id),
  enrollment_date DATE NOT NULL DEFAULT CURRENT_DATE,
  PRIMARY KEY (student_id, course_id)
);
```

### 3NF — Third Normal Form

**Quy tắc:** Đã thỏa 2NF + không có **transitive dependencies** (non-key → non-key).

```sql
-- ❌ city, state phụ thuộc vào zip_code (transitive qua non-key)
CREATE TABLE customers (
  id SERIAL PRIMARY KEY, name TEXT NOT NULL,
  zip_code TEXT NOT NULL,
  city TEXT NOT NULL,       -- ← phụ thuộc zip_code
  state TEXT NOT NULL       -- ← phụ thuộc zip_code
);

-- ✅ Tách bảng zip_codes
CREATE TABLE zip_codes (code TEXT PRIMARY KEY, city TEXT NOT NULL, state TEXT NOT NULL);
CREATE TABLE customers (
  id SERIAL PRIMARY KEY, name TEXT NOT NULL,
  zip_code TEXT NOT NULL REFERENCES zip_codes(code)
);
```

### BCNF — Boyce-Codd Normal Form

**Quy tắc:** Đã thỏa 3NF + mọi determinant phải là candidate key. Xử lý edge case khi 3NF vẫn có anomaly do overlapping candidate keys.

```sql
-- ❌ advisor_id → subject, nhưng advisor_id KHÔNG phải candidate key
CREATE TABLE student_advisors (
  student_id INTEGER, subject TEXT, advisor_id INTEGER,
  PRIMARY KEY (student_id, subject)
);

-- ✅ Tách để mỗi determinant là candidate key
CREATE TABLE advisors (id SERIAL PRIMARY KEY, subject TEXT NOT NULL);
CREATE TABLE student_advisors (
  student_id INTEGER,
  advisor_id INTEGER REFERENCES advisors(id),
  PRIMARY KEY (student_id, advisor_id)
);
```

> **Rule of thumb:** Normalize tới 3NF là đủ cho 99% use cases.

## 2. Denormalization

Denormalization là **cố ý** thêm redundancy để tối ưu read performance. Trade-off có ý thức, KHÔNG phải lười normalize.

**Dùng khi:** Read:Write ratio rất cao (>100:1), JOIN quá nhiều bảng, aggregation compute quá đắt.
**KHÔNG dùng khi:** Schema giai đoạn đầu, write-heavy workloads, "chắc sẽ nhanh hơn" mà chưa đo.

### Kỹ thuật denormalization

```sql
-- ✅ 1. Computed columns — cache aggregate values
ALTER TABLE orders ADD COLUMN total_amount NUMERIC(12,2);
ALTER TABLE orders ADD COLUMN item_count INTEGER;

-- ✅ 2. Materialized Views — pre-compute dashboard data
CREATE MATERIALIZED VIEW mv_monthly_sales AS
SELECT date_trunc('month', o.created_at) AS month, p.category,
  COUNT(DISTINCT o.id) AS order_count,
  SUM(oi.unit_price * oi.quantity) AS total_revenue
FROM orders o
JOIN order_items oi ON o.id = oi.order_id
JOIN products p ON oi.product_id = p.id
WHERE o.status = 'completed'
GROUP BY 1, 2;

CREATE UNIQUE INDEX idx_mv_monthly_sales ON mv_monthly_sales(month, category);
REFRESH MATERIALIZED VIEW CONCURRENTLY mv_monthly_sales;

-- ✅ 3. Denormalized copy columns (snapshot at write time)
ALTER TABLE orders ADD COLUMN customer_name TEXT;
-- customer đổi tên → orders cũ giữ tên cũ (đúng business logic)
```

```typescript
// ✅ TypeScript: Update cached columns in transaction
class OrderService {
  async addItem(orderId: string, item: OrderItemInput): Promise<void> {
    await this.db.transaction(async (tx) => {
      await tx.insert(orderItems).values({ orderId, ...item });
      const totals = await tx
        .select({ total: sql`SUM(unit_price * quantity)`, count: sql`COUNT(*)` })
        .from(orderItems).where(eq(orderItems.orderId, orderId));
      await tx.update(orders).set({
        totalAmount: totals[0].total, itemCount: totals[0].count,
      }).where(eq(orders.id, orderId));
    });
  }
}
```

## 3. Entity-Relationship Modeling

### Các loại relationship và implementation

```sql
-- 1:N — User → Orders
-- FK ở bảng "many" side
ALTER TABLE orders ADD COLUMN customer_id UUID REFERENCES users(id);

-- M:N — Students ↔ Courses → cần junction table
-- ❌ Sai: Array of IDs — không FK, không index hiệu quả
CREATE TABLE students (id SERIAL PRIMARY KEY, course_ids INTEGER[]);

-- ✅ Junction table với relationship metadata
CREATE TABLE student_courses (
  student_id INTEGER REFERENCES students(id) ON DELETE CASCADE,
  course_id INTEGER REFERENCES courses(id) ON DELETE CASCADE,
  enrolled_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  grade CHAR(2),
  PRIMARY KEY (student_id, course_id)
);

-- Self-referencing: employee → manager (tree structure)
CREATE TABLE employees (
  id SERIAL PRIMARY KEY, name TEXT NOT NULL,
  manager_id INTEGER REFERENCES employees(id)
);
-- Query cây con dùng recursive CTE
WITH RECURSIVE subordinates AS (
  SELECT id, name, manager_id, 0 AS level FROM employees WHERE id = :root
  UNION ALL
  SELECT e.id, e.name, e.manager_id, s.level + 1
  FROM employees e JOIN subordinates s ON e.manager_id = s.id
) SELECT * FROM subordinates ORDER BY level;
```

## 4. Schema Design Patterns

### 4.1 Polymorphic Associations

Nhiều entity types (Post, Photo, Video) đều có comments.

```sql
-- ❌ Nullable FK cho mỗi parent type — thêm entity = ALTER TABLE
CREATE TABLE comments (
  id SERIAL PRIMARY KEY, body TEXT NOT NULL,
  post_id INTEGER REFERENCES posts(id),
  photo_id INTEGER REFERENCES photos(id)
  -- Chỉ 1 FK phải NOT NULL — khó enforce
);

-- ✅ Option 1: Separate tables per parent (khuyến khích)
CREATE TABLE post_comments (
  id SERIAL PRIMARY KEY,
  post_id INTEGER NOT NULL REFERENCES posts(id) ON DELETE CASCADE,
  body TEXT NOT NULL, author_id INTEGER NOT NULL REFERENCES users(id),
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- ✅ Option 2: Discriminator column (query across types)
CREATE TABLE comments (
  id SERIAL PRIMARY KEY,
  commentable_type TEXT NOT NULL CHECK (commentable_type IN ('post', 'photo', 'video')),
  commentable_id INTEGER NOT NULL,
  body TEXT NOT NULL, author_id INTEGER NOT NULL REFERENCES users(id),
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_comments_post ON comments(commentable_id) WHERE commentable_type = 'post';
```

### 4.2 EAV vs JSONB

```sql
-- ❌ EAV: mọi thứ là string, không type-safe, query chậm
CREATE TABLE product_attributes (
  product_id INTEGER REFERENCES products(id),
  attribute_name TEXT, attribute_value TEXT,
  PRIMARY KEY (product_id, attribute_name)
);

-- ✅ JSONB: flexible, queryable, indexable
CREATE TABLE products (
  id SERIAL PRIMARY KEY, name TEXT NOT NULL, category TEXT NOT NULL,
  base_price NUMERIC(12,2) NOT NULL,
  attributes JSONB NOT NULL DEFAULT '{}',
  CONSTRAINT valid_attributes CHECK (
    CASE category
      WHEN 'clothing' THEN attributes ?& ARRAY['size', 'color']
      WHEN 'electronics' THEN attributes ?& ARRAY['brand', 'model']
      ELSE true
    END
  )
);
CREATE INDEX idx_products_attributes ON products USING GIN (attributes);
```

```typescript
// TypeScript: Type-safe JSONB per category
interface ClothingAttributes { size: string; color: string; material: string; }
interface ElectronicsAttributes { brand: string; model: string; }
type ProductAttributes = ClothingAttributes | ElectronicsAttributes;

interface Product<T extends ProductAttributes = ProductAttributes> {
  id: string; name: string; category: string; basePrice: number; attributes: T;
}
```

### 4.3 Temporal Data (Bi-temporal)

```sql
-- ✅ Price history: valid time + transaction time
CREATE TABLE product_prices (
  id SERIAL PRIMARY KEY,
  product_id INTEGER NOT NULL REFERENCES products(id),
  price NUMERIC(12,2) NOT NULL, currency CHAR(3) NOT NULL DEFAULT 'USD',
  valid_from TIMESTAMPTZ NOT NULL, valid_to TIMESTAMPTZ NOT NULL DEFAULT 'infinity',
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  superseded_at TIMESTAMPTZ,
  CONSTRAINT valid_time_range CHECK (valid_from < valid_to),
  CONSTRAINT no_overlap EXCLUDE USING GIST (
    product_id WITH =, tstzrange(valid_from, valid_to) WITH &&
  ) WHERE (superseded_at IS NULL)
);
```

### 4.4 Soft Deletes

```sql
-- ❌ Boolean flag: UNIQUE constraints vẫn count deleted records
ALTER TABLE users ADD COLUMN is_deleted BOOLEAN DEFAULT false;

-- ✅ deleted_at + partial unique index
ALTER TABLE users ADD COLUMN deleted_at TIMESTAMPTZ;
CREATE UNIQUE INDEX idx_users_email_active ON users(email) WHERE deleted_at IS NULL;
CREATE VIEW active_users AS SELECT * FROM users WHERE deleted_at IS NULL;
```

### 4.5 Audit Trails

```sql
-- ✅ Generic audit log với trigger tự động
CREATE TABLE audit_logs (
  id BIGSERIAL PRIMARY KEY, table_name TEXT NOT NULL, record_id TEXT NOT NULL,
  action TEXT NOT NULL CHECK (action IN ('INSERT', 'UPDATE', 'DELETE')),
  old_data JSONB, new_data JSONB, changed_fields TEXT[],
  actor_id TEXT, occurred_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE OR REPLACE FUNCTION audit_trigger_func() RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO audit_logs(table_name, record_id, action, old_data, new_data, actor_id)
  VALUES (TG_TABLE_NAME, COALESCE(NEW.id::text, OLD.id::text), TG_OP,
    CASE WHEN TG_OP IN ('UPDATE','DELETE') THEN to_jsonb(OLD) END,
    CASE WHEN TG_OP IN ('INSERT','UPDATE') THEN to_jsonb(NEW) END,
    current_setting('app.current_user_id', true));
  RETURN COALESCE(NEW, OLD);
END; $$ LANGUAGE plpgsql;

CREATE TRIGGER users_audit AFTER INSERT OR UPDATE OR DELETE ON users
  FOR EACH ROW EXECUTE FUNCTION audit_trigger_func();
```

```typescript
// ✅ Set actor context trước mỗi DB operation
async function withAuditContext<T>(
  pool: Pool, userId: string, fn: (client: PoolClient) => Promise<T>
): Promise<T> {
  const client = await pool.connect();
  try {
    await client.query('BEGIN');
    await client.query(`SET LOCAL app.current_user_id = '${userId}'`);
    const result = await fn(client);
    await client.query('COMMIT');
    return result;
  } catch (error) {
    await client.query('ROLLBACK');
    throw error;
  } finally { client.release(); }
}
```

## 5. Naming Conventions

```sql
-- ❌ Inconsistent: Users, order_items, tblProduct, OrderDetail
-- ✅ Rules:
-- 1. snake_case cho tất cả         4. FK = referenced_table_singular + _id
-- 2. Plural cho table names         5. Boolean: is_ hoặc has_ prefix
-- 3. Singular cho column names      6. Timestamps: _at suffix

CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email TEXT NOT NULL,
  display_name TEXT NOT NULL,
  is_verified BOOLEAN NOT NULL DEFAULT false,
  has_mfa_enabled BOOLEAN NOT NULL DEFAULT false,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Junction table: alphabetical order
CREATE TABLE organization_users (
  organization_id UUID REFERENCES organizations(id) ON DELETE CASCADE,
  user_id UUID REFERENCES users(id) ON DELETE CASCADE,
  role TEXT NOT NULL DEFAULT 'member',
  PRIMARY KEY (organization_id, user_id)
);

-- Index: idx_{table}_{columns}  |  Constraint: chk_{table}_{desc}
CREATE INDEX idx_users_email ON users(email);
```

## 6. Constraints — Business Rules trong Database

```sql
-- ✅ NOT NULL là default mindset — nullable phải có lý do
CREATE TABLE products (
  id SERIAL PRIMARY KEY,
  name TEXT NOT NULL,                  -- Product phải có tên
  price NUMERIC(12,2) NOT NULL,        -- Price bắt buộc
  description TEXT,                    -- Nullable OK: optional
  discontinued_at TIMESTAMPTZ          -- Nullable OK: NULL = chưa discontinued
);

-- ✅ CHECK constraints = business logic trong DB
CREATE TABLE orders (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  status TEXT NOT NULL DEFAULT 'draft'
    CHECK (status IN ('draft','pending','confirmed','shipped','delivered','cancelled')),
  total_amount NUMERIC(12,2) NOT NULL CHECK (total_amount >= 0),
  discount_percent NUMERIC(5,2) CHECK (discount_percent BETWEEN 0 AND 100),
  shipping_address_id UUID,
  CONSTRAINT confirmed_needs_address CHECK (
    status NOT IN ('confirmed','shipped','delivered') OR shipping_address_id IS NOT NULL
  )
);

-- ✅ Composite UNIQUE = business rules
CREATE TABLE product_reviews (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  product_id UUID NOT NULL REFERENCES products(id),
  user_id UUID NOT NULL REFERENCES users(id),
  rating INTEGER NOT NULL CHECK (rating BETWEEN 1 AND 5),
  UNIQUE (product_id, user_id)  -- 1 review per user per product
);

-- ✅ FK actions phù hợp business logic
ALTER TABLE order_items ADD FOREIGN KEY (order_id) REFERENCES orders(id) ON DELETE CASCADE;
ALTER TABLE orders ADD FOREIGN KEY (customer_id) REFERENCES customers(id) ON DELETE RESTRICT;
ALTER TABLE posts ADD FOREIGN KEY (author_id) REFERENCES users(id) ON DELETE SET NULL;
```

## 7. Full Example: Multi-Tenant SaaS Schema

Schema cho SaaS project management (tương tự Linear/Jira). Strategy: shared database, tenant_id column + RLS.

```sql
CREATE TABLE tenants (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name TEXT NOT NULL, slug TEXT NOT NULL UNIQUE,
  plan TEXT NOT NULL DEFAULT 'free' CHECK (plan IN ('free','pro','enterprise')),
  settings JSONB NOT NULL DEFAULT '{}',
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email TEXT NOT NULL UNIQUE, password_hash TEXT NOT NULL,
  display_name TEXT NOT NULL, is_email_verified BOOLEAN NOT NULL DEFAULT false,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE tenant_members (
  tenant_id UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  role TEXT NOT NULL DEFAULT 'member' CHECK (role IN ('owner','admin','member','viewer')),
  PRIMARY KEY (tenant_id, user_id)
);

CREATE TABLE projects (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
  name TEXT NOT NULL, slug TEXT NOT NULL,
  status TEXT NOT NULL DEFAULT 'active' CHECK (status IN ('active','archived')),
  UNIQUE (tenant_id, slug)
);

CREATE TABLE issues (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
  project_id UUID NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
  issue_number INTEGER NOT NULL,
  title TEXT NOT NULL, description TEXT,
  status TEXT NOT NULL DEFAULT 'backlog'
    CHECK (status IN ('backlog','todo','in_progress','in_review','done','cancelled')),
  priority TEXT NOT NULL DEFAULT 'medium'
    CHECK (priority IN ('urgent','high','medium','low','none')),
  assignee_id UUID REFERENCES users(id),
  reporter_id UUID NOT NULL REFERENCES users(id),
  labels TEXT[] NOT NULL DEFAULT '{}',
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  deleted_at TIMESTAMPTZ,
  UNIQUE (project_id, issue_number)
);

-- RLS: tự động filter theo tenant
ALTER TABLE issues ENABLE ROW LEVEL SECURITY;
CREATE POLICY tenant_isolation ON issues
  USING (tenant_id = current_setting('app.current_tenant_id')::uuid);

-- Indexes: tenant_id luôn đứng đầu composite index
CREATE INDEX idx_issues_tenant_project ON issues(tenant_id, project_id);
CREATE INDEX idx_issues_tenant_assignee ON issues(tenant_id, assignee_id) WHERE deleted_at IS NULL;
CREATE INDEX idx_issues_search ON issues
  USING GIN (to_tsvector('english', title || ' ' || COALESCE(description, '')));
```

```typescript
// ✅ TypeScript: Middleware set tenant context cho RLS
async function withTenantContext<T>(
  pool: Pool, ctx: { tenantId: string; userId: string },
  fn: (client: PoolClient) => Promise<T>
): Promise<T> {
  const client = await pool.connect();
  try {
    await client.query('BEGIN');
    await client.query(`SET LOCAL app.current_tenant_id = '${ctx.tenantId}'`);
    await client.query(`SET LOCAL app.current_user_id = '${ctx.userId}'`);
    const result = await fn(client);
    await client.query('COMMIT');
    return result;
  } catch (error) {
    await client.query('ROLLBACK');
    throw error;
  } finally { client.release(); }
}

// ✅ Service: RLS tự động filter — query an toàn
class IssueService {
  async listIssues(ctx: TenantContext, projectId: string, filters: IssueFilters) {
    return withTenantContext(this.pool, ctx, async (client) => {
      const conditions = ['project_id = $1', 'deleted_at IS NULL'];
      const params: unknown[] = [projectId];
      if (filters.status) { params.push(filters.status); conditions.push(`status = $${params.length}`); }
      if (filters.assigneeId) { params.push(filters.assigneeId); conditions.push(`assignee_id = $${params.length}`); }
      const { rows } = await client.query(
        `SELECT * FROM issues WHERE ${conditions.join(' AND ')}
         ORDER BY CASE priority WHEN 'urgent' THEN 1 WHEN 'high' THEN 2 WHEN 'medium' THEN 3 ELSE 4 END,
         created_at DESC LIMIT 50`, params);
      return rows;
    });
  }
}
```

### Multi-tenant strategy comparison

| Strategy | Ưu điểm | Nhược điểm | Dùng khi |
|----------|---------|------------|----------|
| Shared schema + RLS | Đơn giản, chi phí thấp | Isolation yếu hơn | Startup, SaaS nhỏ-vừa |
| Schema per tenant | Isolation tốt, dễ backup riêng | Migrations phức tạp | Compliance requirements |
| Database per tenant | Isolation mạnh nhất | Operational overhead cao | Enterprise, regulated |
