## Containerization

### Production Dockerfile cho Node.js/TypeScript

```dockerfile
# Multi-stage build: separate build and runtime

# Stage 1: Build
FROM node:20-alpine AS builder
WORKDIR /app

# Install dependencies first (cache layer)
COPY package.json package-lock.json ./
RUN npm ci --ignore-scripts

# Copy source and build
COPY tsconfig.json ./
COPY src/ ./src/
RUN npm run build

# Prune dev dependencies
RUN npm ci --omit=dev --ignore-scripts

# Stage 2: Production runtime
FROM node:20-alpine AS runtime
WORKDIR /app

# Security: don't run as root
RUN addgroup --system appgroup && adduser --system appuser --ingroup appgroup

# Copy only what's needed
COPY --from=builder --chown=appuser:appgroup /app/dist ./dist
COPY --from=builder --chown=appuser:appgroup /app/node_modules ./node_modules
COPY --from=builder --chown=appuser:appgroup /app/package.json ./

USER appuser

# Health check
HEALTHCHECK --interval=30s --timeout=5s --start-period=10s --retries=3 \
  CMD wget --no-verbose --tries=1 --spider http://localhost:3000/health/live || exit 1

EXPOSE 3000

CMD ["node", "dist/main.js"]
```

### Docker Compose cho development

```yaml
# docker-compose.yml
services:
  app:
    build:
      context: .
      target: builder  # Use build stage for development
    volumes:
      - ./src:/app/src  # Hot reload
    ports:
      - "3000:3000"
    environment:
      - NODE_ENV=development
      - DATABASE_URL=postgres://dev:dev@postgres:5432/myapp
      - REDIS_URL=redis://redis:6379
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_started

  postgres:
    image: postgres:16-alpine
    environment:
      POSTGRES_USER: dev
      POSTGRES_PASSWORD: dev
      POSTGRES_DB: myapp
    ports:
      - "5432:5432"
    volumes:
      - pgdata:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U dev"]
      interval: 5s
      timeout: 5s
      retries: 5

  redis:
    image: redis:7-alpine
    ports:
      - "6379:6379"

volumes:
  pgdata:
```

## CI/CD Pipeline

### GitHub Actions Pipeline

```yaml
# .github/workflows/ci.yml
name: CI/CD

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  # Stage 1: Lint + Type check
  check:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm
      - run: npm ci
      - run: npm run lint
      - run: npm run typecheck

  # Stage 2: Test (parallel with check)
  test:
    runs-on: ubuntu-latest
    services:
      postgres:
        image: postgres:16
        env:
          POSTGRES_PASSWORD: test
          POSTGRES_DB: test
        ports:
          - 5432:5432
        options: >-
          --health-cmd pg_isready
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm
      - run: npm ci
      - run: npm test
        env:
          DATABASE_URL: postgres://postgres:test@localhost:5432/test

  # Stage 3: Build + Push (only on main, after check+test pass)
  deploy:
    needs: [check, test]
    if: github.ref == 'refs/heads/main'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Build and push Docker image
        run: |
          docker build -t myapp:${{ github.sha }} .
          docker tag myapp:${{ github.sha }} registry.example.com/myapp:latest
          docker push registry.example.com/myapp:${{ github.sha }}
          docker push registry.example.com/myapp:latest

      - name: Deploy to production
        run: |
          # Update Kubernetes deployment
          kubectl set image deployment/myapp \
            myapp=registry.example.com/myapp:${{ github.sha }}
          kubectl rollout status deployment/myapp --timeout=300s
```

### Pipeline Principles

```
1. Fast Feedback: lint/typecheck first (fail trong 30s, không đợi 10min test)
2. Parallel stages: lint + test chạy song song
3. Immutable artifacts: Docker image tagged with commit SHA
4. Same artifact: build ONCE, deploy to staging → production (same image)
5. Automated rollback: nếu deploy fail → tự rollback
6. No manual steps: dev push → production (nếu tests pass)
```

```typescript
// Pulumi (TypeScript-native IaC) — ví dụ deploy to AWS

import * as aws from '@pulumi/aws';
import * as pulumi from '@pulumi/pulumi';

// VPC
const vpc = new aws.ec2.Vpc('main', {
  cidrBlock: '10.0.0.0/16',
  enableDnsHostnames: true,

// RDS PostgreSQL
const db = new aws.rds.Instance('app-db', {
  engine: 'postgres',
  engineVersion: '16',
  instanceClass: 'db.t3.medium',
  allocatedStorage: 50,
  dbName: 'myapp',
  username: 'admin',
  password: dbPasswordSecret.result,
  vpcSecurityGroupIds: [dbSecurityGroup.id],
  backupRetentionPeriod: 7,
  multiAz: true,              // High availability
  storageEncrypted: true,     // Encryption at rest
  deletionProtection: true,   // Prevent accidental deletion
});

// ECS Fargate Service
const service = new aws.ecs.Service('app', {
  cluster: cluster.id,
  taskDefinition: taskDef.arn,
  desiredCount: 2,            // Minimum 2 instances
  launchType: 'FARGATE',
  loadBalancers: [{
    targetGroupArn: targetGroup.arn,
    containerName: 'app',
    containerPort: 3000,
  }],
  deploymentConfiguration: {
    minimumHealthyPercent: 100, // Zero-downtime: keep old running until new is healthy
    maximumPercent: 200,
  },
});

// Auto-scaling
const scaling = new aws.appautoscaling.Target('app-scaling', {
  serviceNamespace: 'ecs',
  resourceId: pulumi.interpolate`service/${cluster.name}/${service.name}`,
  scalableDimension: 'ecs:service:DesiredCount',
  minCapacity: 2,
  maxCapacity: 10,
});

new aws.appautoscaling.Policy('cpu-scaling', {
  serviceNamespace: 'ecs',
  resourceId: scaling.resourceId,
  scalableDimension: scaling.scalableDimension,
  policyType: 'TargetTrackingScaling',
  targetTrackingScalingPolicyConfiguration: {
    predefinedMetricSpecification: {
      predefinedMetricType: 'ECSServiceAverageCPUUtilization',
    },
    targetValue: 70,  // Scale out when CPU > 70%
  },
});
```

## Environments Strategy

```
Development:    Local Docker Compose, hot reload, mock services
                → Developer laptop

Staging:        Production-like, real infrastructure, test data
                → Deploy on every PR merge, automated tests
                → Same Docker image as production

Production:     Real traffic, real data
                → Deploy after staging validation
                → Blue-green or rolling deployment
                → Monitoring + alerting active

Preview:        Per-PR ephemeral environments (optional)
                → Vercel preview deployments, Railway PR environments
                → Auto-destroyed after PR merge
```

## Zero-Downtime Deployment

### Rolling Deployment

```
Kubernetes rolling update:

Time 0: [Pod A v1] [Pod B v1]              ← All running v1
Time 1: [Pod A v1] [Pod B v1] [Pod C v2]   ← New pod starting
Time 2: [Pod A v1] [Pod C v2]              ← Pod B terminated after C healthy
Time 3: [Pod A v1] [Pod C v2] [Pod D v2]   ← Another new pod
Time 4: [Pod C v2] [Pod D v2]              ← All v2, zero downtime
```

### Blue-Green Deployment

```
Blue (current):  [Server A] [Server B] ← Serving traffic
Green (new):     [Server C] [Server D] ← Deployed, tested, not serving

Switch: Load balancer points to Green
Rollback: Load balancer points back to Blue (instant)
Cleanup: Terminate Blue after confidence period
```

### Database Migrations — The Hard Part

```
Deployment order for ZERO downtime:

1. Deploy NEW code that handles BOTH old AND new schema
2. Run migration (add column, create table)
3. Deploy code that uses new schema
4. (Later) Clean up: remove old schema support from code

Example: Adding "phone" column to users

Step 1: Deploy code v2 (reads phone if exists, doesn't require it)
Step 2: ALTER TABLE users ADD COLUMN phone VARCHAR(20);
Step 3: Deploy code v3 (starts writing phone, shows in UI)

NEVER: Deploy code that REQUIRES new column before migration runs
       → Old pods will crash!
```

## Configuration Management

```typescript
// Typed configuration with validation at startup

import { z } from 'zod';

const ConfigSchema = z.object({
  port: z.coerce.number().int().min(1).max(65535).default(3000),
  environment: z.enum(['development', 'staging', 'production']),
  databaseUrl: z.string().url(),
  redisUrl: z.string().url(),
  jwtSecret: z.string().min(32),
  corsOrigins: z.string().transform(s => s.split(',')),
  logLevel: z.enum(['debug', 'info', 'warn', 'error']).default('info'),

  // Feature flags
  enableNewCheckout: z.coerce.boolean().default(false),

  // Limits
  maxUploadSizeMb: z.coerce.number().default(10),
  rateLimitPerMinute: z.coerce.number().default(100),
});

type AppConfig = z.infer<typeof ConfigSchema>;

function loadConfig(): AppConfig {
  const result = ConfigSchema.safeParse({
    port: process.env.PORT,
    environment: process.env.NODE_ENV,
    databaseUrl: process.env.DATABASE_URL,
    redisUrl: process.env.REDIS_URL,
    jwtSecret: process.env.JWT_SECRET,
    corsOrigins: process.env.CORS_ORIGINS,
    logLevel: process.env.LOG_LEVEL,
    enableNewCheckout: process.env.ENABLE_NEW_CHECKOUT,
    maxUploadSizeMb: process.env.MAX_UPLOAD_SIZE_MB,
    rateLimitPerMinute: process.env.RATE_LIMIT_PER_MINUTE,
  });

  if (!result.success) {
    console.error('Invalid configuration:');
    for (const error of result.error.errors) {
      console.error(`  ${error.path.join('.')}: ${error.message}`);
    }
    process.exit(1); // Fail fast: don't start with bad config
  }

  return result.data;
}

// Load ONCE at startup, pass as dependency
const config = loadConfig();
```
