## Integration Tests

### Bugs mà CHỈ integration tests bắt được

```
├── SQL query sai (ORM mapping lỗi, wrong JOIN)
├── Serialization mismatch (JSON ↔ Object)
├── HTTP request/response format sai
├── Transaction boundary sai (partial commits)
├── Middleware chain order sai (auth before validation?)
└── Message queue routing lỗi
```

### Nguyên tắc

```typescript
// REAL database, REAL HTTP server, mock CHỈ external services ngoài control

describe('Order API (integration)', () => {
  let app: Express;
  let db: TestDatabase;

  beforeAll(async () => {
    db = await TestDatabase.start(); // real PostgreSQL
    app = createApp({ database: db.connectionString });
  });
  afterAll(async () => { await db.stop(); });
  beforeEach(async () => { await db.truncateAll(); });

  it('creates order and persists to database', async () => {
    const user = await db.seedUser({ email: 'buyer@example.com' });
    const product = await db.seedProduct({ price: 29.99, stock: 50 });

    const response = await request(app)
      .post('/api/orders')
      .set('Authorization', `Bearer ${user.token}`)
      .send({ items: [{ productId: product.id, quantity: 3 }] });

    expect(response.status).toBe(201);
    expect(response.body).toMatchObject({ status: 'pending', total: 89.97 });

    // Verify database state directly
    const dbOrder = await db.query('SELECT * FROM orders WHERE id = $1', [response.body.id]);
    expect(dbOrder.rows[0].total_cents).toBe(8997);

    const dbProduct = await db.query('SELECT stock FROM products WHERE id = $1', [product.id]);
    expect(dbProduct.rows[0].stock).toBe(47);
  });
});
```

## Database Integration Tests

### Testcontainers — Real database in Docker

```typescript
import { PostgreSqlContainer } from '@testcontainers/postgresql';

class TestDatabase {
  private container!: StartedPostgreSqlContainer;
  private pool!: Pool;

  async start(): Promise<void> {
    this.container = await new PostgreSqlContainer('postgres:16-alpine')
      .withDatabase('testdb').withUsername('test').withPassword('test')
      .start();
    this.pool = new Pool({ connectionString: this.container.getConnectionUri() });
    await runMigrations(this.pool);
  }
  async stop(): Promise<void> { await this.pool.end(); await this.container.stop(); }
  async truncateAll(): Promise<void> {
    await this.pool.query(`
      DO $$ DECLARE r RECORD; BEGIN
        FOR r IN (SELECT tablename FROM pg_tables WHERE schemaname = 'public') LOOP
          EXECUTE 'TRUNCATE TABLE ' || quote_ident(r.tablename) || ' CASCADE';
        END LOOP;
      END $$;
    `);
  }
}
```

### Transaction rollback — nhanh hơn truncate

```typescript
// Mỗi test trong transaction, rollback khi xong → siêu nhanh
class TransactionTestHelper {
  private client!: PoolClient;
  async begin(): Promise<void> {
    this.client = await pool.connect();
    await this.client.query('BEGIN');
    await this.client.query('SAVEPOINT test_start');
  }
  async rollback(): Promise<void> {
    await this.client.query('ROLLBACK TO SAVEPOINT test_start');
    await this.client.query('ROLLBACK');
    this.client.release();
  }
}

beforeEach(async () => await txHelper.begin());
afterEach(async () => await txHelper.rollback());
```

## API Integration Tests

```typescript
describe('REST API Integration', () => {
  // Authentication
  it('returns 401 without token', async () => {
    expect((await request(app).get('/api/orders')).status).toBe(401);
  });
  it('returns 403 for insufficient permissions', async () => {
    const viewer = await getTokenForRole('viewer');
    expect((await request(app).delete('/api/orders/123')
      .set('Authorization', `Bearer ${viewer}`)).status).toBe(403);
  });

  // Validation
  it('returns 400 with detailed errors for invalid input', async () => {
    const response = await request(app)
      .post('/api/orders').set('Authorization', `Bearer ${adminToken}`)
      .send({ items: [], shippingAddress: { city: '' } });
    expect(response.status).toBe(400);
    expect(response.body.errors).toEqual(expect.arrayContaining([
      expect.objectContaining({ field: 'items', message: expect.any(String) }),
    ]));
  });

  // Full lifecycle
  it('create → confirm → ship → complete', async () => {
    const createRes = await request(app).post('/api/orders')
      .set('Authorization', `Bearer ${customerToken}`)
      .send({ items: [{ productId: 'prod-1', quantity: 2 }], shippingAddress: validAddress });
    expect(createRes.status).toBe(201);
    const orderId = createRes.body.id;

    const confirmRes = await request(app).post(`/api/orders/${orderId}/confirm`)
      .set('Authorization', `Bearer ${adminToken}`);
    expect(confirmRes.body.status).toBe('confirmed');

    const shipRes = await request(app).post(`/api/orders/${orderId}/ship`)
      .set('Authorization', `Bearer ${adminToken}`)
      .send({ trackingNumber: 'TRACK-123' });
    expect(shipRes.body.status).toBe('shipped');
  });

  // Pagination
  it('returns paginated results with metadata', async () => {
    await Promise.all(Array.from({ length: 25 }, () => db.seedOrder({ customerId })));
    const page = await request(app).get('/api/orders?page=1&limit=10')
      .set('Authorization', `Bearer ${customerToken}`);
    expect(page.body.data).toHaveLength(10);
    expect(page.body.pagination).toMatchObject({
      total: 25, totalPages: 3, hasNextPage: true,
    });
  });

  // Concurrency
  it('returns 409 for concurrent modification conflict', async () => {
    const order = await db.seedOrder({ status: 'pending', version: 1 });
    await db.query('UPDATE orders SET version = 2 WHERE id = $1', [order.id]);
    const response = await request(app).patch(`/api/orders/${order.id}`)
      .set('Authorization', `Bearer ${adminToken}`)
      .send({ status: 'confirmed', expectedVersion: 1 });
    expect(response.status).toBe(409);
  });
});
```

## External Service Testing

### MSW — Mock HTTP ở network level

```typescript
import { setupServer } from 'msw/node';
import { http, HttpResponse } from 'msw';

const server = setupServer(
  http.post('https://api.stripe.com/v1/charges', async ({ request }) => {
    const body = await request.text();
    const amount = parseInt(new URLSearchParams(body).get('amount') ?? '0');
    if (amount > 999999) {
      return HttpResponse.json({ error: { message: 'Amount too large' } }, { status: 402 });
    }
    return HttpResponse.json({ id: `ch_test_${Date.now()}`, status: 'succeeded', amount });
  }),
);

beforeAll(() => server.listen({ onUnhandledRequest: 'error' }));
afterEach(() => server.resetHandlers());
afterAll(() => server.close());

it('handles payment failure gracefully', async () => {
  server.use(
    http.post('https://api.stripe.com/v1/charges', () =>
      HttpResponse.json({ error: { message: 'Card declined' } }, { status: 402 })),
  );
  const result = await paymentService.charge({ amount: Money.of(100), currency: 'usd' });
  expect(result.success).toBe(false);
  expect(result.error).toBe('Card declined');
});

it('retries on 5xx errors', async () => {
  let attempts = 0;
  server.use(
    http.post('https://api.stripe.com/v1/charges', () => {
      attempts++;
      if (attempts < 3) return HttpResponse.json({}, { status: 500 });
      return HttpResponse.json({ id: 'ch_ok', status: 'succeeded', amount: 1000 });
    }),
  );
  const result = await paymentService.charge({ amount: Money.of(10), currency: 'usd' });
  expect(result.success).toBe(true);
  expect(attempts).toBe(3);
});
```

## Contract Testing với Pact

### Tại sao cần

```
Service A gọi Service B → B thay đổi response format → A's unit tests pass (mock cũ)
→ B's tests pass (B test B) → Production: A CRASH

Contract testing: A (consumer) định nghĩa expectations → B (provider) verify mỗi deploy
```

### Consumer side

```typescript
import { PactV3, MatchersV3 } from '@pact-foundation/pact';
const { string, integer, eachLike } = MatchersV3;

const provider = new PactV3({ consumer: 'OrderService', provider: 'ProductService' });

it('returns product details by ID', async () => {
  await provider
    .given('product with ID prod-123 exists')
    .uponReceiving('a request for product details')
    .withRequest({ method: 'GET', path: '/api/products/prod-123' })
    .willRespondWith({
      status: 200,
      body: {
        id: string('prod-123'),
        name: string('Widget'),
        price: { amount: integer(2999), currency: string('USD') },
        inStock: true,
        categories: eachLike(string('electronics')),
      },
    })
    .executeTest(async (mockServer) => {
      const client = new ProductServiceClient(mockServer.url);
      const product = await client.getProduct('prod-123');
      expect(product.name).toBe('Widget');
    });
});
// → Generates pact file: OrderService-ProductService.json
```

### Provider side

```typescript
import { Verifier } from '@pact-foundation/pact';

it('fulfills OrderService contract', async () => {
  const verifier = new Verifier({
    providerBaseUrl: `http://localhost:${port}`,
    pactUrls: ['./pacts/OrderService-ProductService.json'],
    stateHandlers: {
      'product with ID prod-123 exists': async () => {
        await db.seedProduct({ id: 'prod-123', name: 'Widget', priceCents: 2999 });
      },
    },
  });
  await verifier.verifyProvider();
  // → Nếu Provider thay đổi API format → test FAIL → fix trước khi break consumers
});
```

## E2E Tests

### Nguyên tắc: critical business paths ONLY

```
E2E cần: Registration flow, purchase flow, core business operation
E2E KHÔNG cần: Edge cases, error handling, API validation → unit/integration
Rule of thumb: 5-15 E2E tests cho một application
```

### Page Object Model — tách UI details khỏi test logic

class ProductPage {
  constructor(private page: Page) {}
  async goto() { await this.page.goto('/products'); }
  async search(query: string) {
    await this.page.fill('[data-testid="search"]', query);
    await this.page.keyboard.press('Enter');
    await this.page.waitForSelector('[data-testid="product-card"]');
  }
  async selectProduct(name: string) {
    await this.page.click(`[data-testid="product-card"]:has-text("${name}")`);
  }
}

class CheckoutPage {
  constructor(private page: Page) {}
  async fillShipping(data: ShippingData) { /* ... */ }
  async fillPayment(card: TestCard) { /* ... */ }
  async placeOrder(): Promise<string> {
    await this.page.click('[data-testid="place-order"]');
    await this.page.waitForSelector('[data-testid="order-confirmation"]');

// Test using Page Objects — clean, maintainable

## Performance Testing Strategy

### Các loại

```
Load:   Simulate expected traffic → measure response time, throughput, error rate
Stress: Tăng dần load → tìm breaking point và failure mode
Spike:  Flash sale / viral moment → test auto-scaling
Soak:   Moderate load 24h → detect memory leaks, connection exhaustion
```

### Performance budgets

```typescript
interface PerformanceBudget {
  endpoint: string;
  p95ResponseMs: number;
  maxErrorRate: number;
  minThroughputRps: number;
}

const budgets: PerformanceBudget[] = [
  { endpoint: 'GET /api/products', p95ResponseMs: 200, maxErrorRate: 0.001, minThroughputRps: 1000 },
  { endpoint: 'POST /api/orders', p95ResponseMs: 800, maxErrorRate: 0.01, minThroughputRps: 100 },
];

// CI smoke test — detect major regressions
it('product listing within budget under light load', async () => {
  const results = await runLoadTest({ url: `${baseUrl}/api/products`, concurrency: 10, duration: '10s' });
  expect(results.p95).toBeLessThan(200);
  expect(results.errorRate).toBeLessThan(0.01);
});
```

### Khi nào chạy

```
Mỗi PR:      Smoke test (10s, 10 users) → detect major regression
Nightly:      Light load (5min, 50 users) → trend analysis
Pre-release:  Full load test (30min, expected peak)
Quarterly:    Stress + Soak → find limits and leaks
```

## Chaos Testing Integration

### Test hệ thống TRONG điều kiện thất bại

```typescript
// 1. Circuit breaker — verify graceful degradation
it('degrades gracefully when payment service is down', async () => {
  server.use(http.post('https://api.stripe.com/*', () => HttpResponse.error()));

  const response = await request(app)
    .post('/api/orders').set('Authorization', `Bearer ${token}`)
    .send(validOrderPayload);

  expect(response.status).toBe(202); // Accepted, queued for later
  expect(response.body.status).toBe('pending_payment');
});

// 2. Latency injection — verify timeouts
it('times out gracefully when external service is slow', async () => {
  server.use(
    http.get('https://api.inventory.internal/stock/*', async () => {
      await new Promise(resolve => setTimeout(resolve, 10_000));
      return HttpResponse.json({ stock: 100 });
    }),
  );
  const response = await request(app)
    .get('/api/products/prod-1/availability').set('Authorization', `Bearer ${token}`);
  expect(response.status).toBe(504);
});

// 3. Data corruption — verify validation
it('handles malformed external response', async () => {
  server.use(
    http.get('https://api.exchange-rates.com/latest', () =>
      HttpResponse.json({ rates: { USD: 'not-a-number', EUR: null } })),
  );
  const response = await request(app)
    .get('/api/products/prod-1/price?currency=EUR').set('Authorization', `Bearer ${token}`);
  expect(response.status).toBe(502);
});
```

### Infrastructure chaos (staging/production)

```
Tools: Gremlin, Chaos Monkey, Litmus (Kubernetes)
Experiments: Kill pod, add DB latency, fill disk, block network

Rules:
├── Start in staging, NOT production
├── Small blast radius (1 pod, not cluster)
├── Have kill switch
├── Business hours only (team ready to respond)
└── Document hypothesis BEFORE running
```

## Tổng kết

```
1. Integration tests = highest ROI cho hầu hết services
   → Real DB, Real HTTP, mock CHỈ external services

2. DB tests: Testcontainers cho accuracy, transaction rollback cho speed

3. API tests: Test full HTTP lifecycle (auth, validation, business, errors)

4. Contract tests (Pact): BẮT BUỘC trong microservices
   → Prevent cross-service breaks at deploy time

5. E2E: 5-15 tests cho critical paths, KHÔNG NHIỀU HƠN
   → Page Object Model + API seeding

6. Performance: Budget-driven, layered (smoke → load → stress → soak)

7. Chaos: Test FAILURE HANDLING — circuit breakers, timeouts, fallbacks
```
