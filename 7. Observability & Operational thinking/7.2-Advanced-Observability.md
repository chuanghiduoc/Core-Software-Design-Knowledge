## Structured Logging Best Practices

### Correlation IDs và Context Propagation

Section 3.9 giới thiệu structured logging cơ bản. Ở đây, ta đi sâu vào **context propagation** xuyên suốt toàn bộ hệ thống.

```typescript
import { AsyncLocalStorage } from 'node:async_hooks';

// ===== Request Context: tự động propagate qua async call chain =====

interface RequestContext {
  requestId: string;
  traceId: string;
  spanId: string;
  userId?: string;
  sessionId?: string;
  // Business context
  tenantId?: string;
  orderId?: string;
  // Source info
  sourceService?: string;
  sourceVersion?: string;
}

// AsyncLocalStorage cho phép "thread-local" storage trong Node.js
// Mỗi request có context riêng, không cần truyền qua parameter
const contextStore = new AsyncLocalStorage<RequestContext>();

function getCurrentContext(): RequestContext | undefined {
  return contextStore.getStore();
}

function runWithContext<T>(ctx: RequestContext, fn: () => T): T {
  return contextStore.run(ctx, fn);
}

// ===== Advanced Logger với automatic context injection =====

type LogLevel = 'debug' | 'info' | 'warn' | 'error' | 'fatal';

interface LogConfig {
  service: string;
  version: string;
  environment: string;
  minLevel: LogLevel;
  sensitiveFields: Set<string>;
  maxFieldLength: number;
}

interface LogEntry {
  timestamp: string;
  level: LogLevel;
  message: string;
  service: string;
  version: string;
  environment: string;
  // Context (auto-injected)
  requestId?: string;
  traceId?: string;
  spanId?: string;
  userId?: string;
  tenantId?: string;
  // Error details
  error?: {
    name: string;
    message: string;
    stack?: string;
    code?: string;
    cause?: string;
  };
  // Custom fields
  [key: string]: unknown;
}

const LOG_LEVELS: Record<LogLevel, number> = {
  debug: 0, info: 1, warn: 2, error: 3, fatal: 4,
};

class AdvancedLogger {
  constructor(
    private config: LogConfig,
    private extraContext: Record<string, unknown> = {},
  ) {}

  child(context: Record<string, unknown>): AdvancedLogger {
    return new AdvancedLogger(this.config, {
      ...this.extraContext,
      ...context,
    });
  }

  debug(message: string, data?: Record<string, unknown>): void {
    this.write('debug', message, data);
  }

  info(message: string, data?: Record<string, unknown>): void {
    this.write('info', message, data);
  }

  warn(message: string, data?: Record<string, unknown>): void {
    this.write('warn', message, data);
  }

  error(message: string, error?: Error, data?: Record<string, unknown>): void {
    this.write('error', message, {
      ...data,
      error: error ? this.serializeError(error) : undefined,
    });
  }

  fatal(message: string, error?: Error, data?: Record<string, unknown>): void {
    this.write('fatal', message, {
      ...data,
      error: error ? this.serializeError(error) : undefined,
    });
  }

  private write(level: LogLevel, message: string, data?: Record<string, unknown>): void {
    if (LOG_LEVELS[level] < LOG_LEVELS[this.config.minLevel]) return;

    // Auto-inject request context từ AsyncLocalStorage
    const reqCtx = getCurrentContext();

    const entry: LogEntry = {
      timestamp: new Date().toISOString(),
      level,
      message,
      service: this.config.service,
      version: this.config.version,
      environment: this.config.environment,
      // Auto-injected context
      requestId: reqCtx?.requestId,
      traceId: reqCtx?.traceId,
      spanId: reqCtx?.spanId,
      userId: reqCtx?.userId,
      tenantId: reqCtx?.tenantId,
      // Merged extra context + call-site data
      ...this.extraContext,
      ...this.sanitize(data ?? {}),
    };

    // Dùng stdout cho info/debug, stderr cho warn/error/fatal
    const stream = LOG_LEVELS[level] >= LOG_LEVELS['warn']
      ? process.stderr
      : process.stdout;

    stream.write(JSON.stringify(entry) + '\n');
  }

  private sanitize(data: Record<string, unknown>): Record<string, unknown> {
    const result: Record<string, unknown> = {};
    for (const [key, value] of Object.entries(data)) {
      if (this.config.sensitiveFields.has(key.toLowerCase())) {
        result[key] = '[REDACTED]';
      } else if (typeof value === 'string' && value.length > this.config.maxFieldLength) {
        result[key] = value.substring(0, this.config.maxFieldLength) + '...[truncated]';
      } else {
        result[key] = value;
      }
    }
    return result;
  }

  private serializeError(error: Error): LogEntry['error'] {
    return {
      name: error.name,
      message: error.message,
      stack: this.config.environment === 'production'
        ? error.stack?.split('\n').slice(0, 5).join('\n')  // Limit stack trace in prod
        : error.stack,
      code: (error as NodeJS.ErrnoException).code,
      cause: error.cause ? String(error.cause) : undefined,
    };
  }
}

// ===== Express Middleware: context creation =====

import { randomUUID } from 'node:crypto';
import type { Request, Response, NextFunction } from 'express';

function contextMiddleware(logger: AdvancedLogger) {
  return (req: Request, res: Response, next: NextFunction): void => {
    const context: RequestContext = {
      requestId: (req.headers['x-request-id'] as string) ?? randomUUID(),
      traceId: extractTraceId(req) ?? randomUUID().replace(/-/g, ''),
      spanId: randomUUID().replace(/-/g, '').substring(0, 16),
      userId: (req as any).user?.id,
      tenantId: req.headers['x-tenant-id'] as string,
      sourceService: req.headers['x-source-service'] as string,
    };

    // Propagate request ID back to caller
    res.setHeader('X-Request-Id', context.requestId);

    runWithContext(context, () => {
      const start = performance.now();

      logger.info('request.start', {
        method: req.method,
        path: req.path,
        query: req.query,
        userAgent: req.headers['user-agent'],
        contentLength: req.headers['content-length'],
        ip: req.ip,
      });

      res.on('finish', () => {
        const durationMs = Math.round(performance.now() - start);
        const logLevel = res.statusCode >= 500 ? 'error'
          : res.statusCode >= 400 ? 'warn' : 'info';

        logger[logLevel]('request.finish', {
          method: req.method,
          path: req.path,
          statusCode: res.statusCode,
          durationMs,
          contentLength: res.getHeader('content-length'),
        });
      });

      next();
    });
  };
}

function extractTraceId(req: Request): string | undefined {
  const traceparent = req.headers['traceparent'] as string;
  if (!traceparent) return undefined;
  const parts = traceparent.split('-');
  return parts.length >= 2 ? parts[1] : undefined;
}
```

### Log Levels: Khi nào dùng gì

```
FATAL  → Process sắp crash, không thể recovery
         Ví dụ: Cannot bind to port, out of memory, corrupted data
         Action: Page on-call NGAY LẬP TỨC

ERROR  → Operation thất bại, nhưng process vẫn chạy
         Ví dụ: Payment failed, database query timeout, external API error
         Action: Investigate trong giờ làm việc, hoặc alert nếu rate cao

WARN   → Điều bất thường nhưng được handle
         Ví dụ: Retry succeeded, cache miss, deprecated API called
         Action: Monitor trend, fix nếu tần suất tăng

INFO   → Business events và operational milestones
         Ví dụ: Order placed, user signed up, deployment started
         Action: Dùng cho dashboards và audit

DEBUG  → Chi tiết để troubleshoot, TẮT ở production bình thường
         Ví dụ: Query parameters, response body, internal state
         Action: BẬT khi cần debug specific issue, TẮT khi xong
```

## Metrics Design: RED & USE Methods

### RED Method (cho request-driven services)

```
RED = Rate, Errors, Duration

RATE:     Bao nhiêu requests/giây đang được xử lý?
ERRORS:   Bao nhiêu requests đang thất bại?
DURATION: Request mất bao lâu? (latency distribution)

RED phù hợp cho:
  → API servers, Web services, Microservices
  → Bất kỳ service nào nhận requests từ bên ngoài
```

```typescript
import client from 'prom-client';

// ===== RED Metrics Implementation với prom-client =====

// RATE: Counter đếm requests
const httpRequestsTotal = new client.Counter({
  name: 'http_requests_total',
  help: 'Total number of HTTP requests',
  labelNames: ['method', 'path', 'status_code'] as const,
});

// ERRORS: Counter đếm errors (subset of requests)
const httpErrorsTotal = new client.Counter({
  name: 'http_errors_total',
  help: 'Total number of HTTP errors (4xx and 5xx)',
  labelNames: ['method', 'path', 'status_code', 'error_type'] as const,
});

// DURATION: Histogram cho latency distribution
const httpRequestDuration = new client.Histogram({
  name: 'http_request_duration_seconds',
  help: 'HTTP request duration in seconds',
  labelNames: ['method', 'path', 'status_code'] as const,
  // Buckets thiết kế cho web request latency
  buckets: [0.005, 0.01, 0.025, 0.05, 0.1, 0.25, 0.5, 1, 2.5, 5, 10],
});

// Middleware tự động collect RED metrics
function redMetricsMiddleware(req: Request, res: Response, next: NextFunction): void {
  const endTimer = httpRequestDuration.startTimer();
  const path = req.route?.path ?? req.path; // Dùng route pattern, không phải actual path

  res.on('finish', () => {
    const labels = {
      method: req.method,
      path,
      status_code: String(res.statusCode),
    };

    httpRequestsTotal.inc(labels);
    endTimer(labels);

    if (res.statusCode >= 400) {
      httpErrorsTotal.inc({
        ...labels,
        error_type: res.statusCode >= 500 ? 'server_error' : 'client_error',
      });
    }
  });

  next();
}

// PromQL queries cho RED dashboard:
// Rate:     rate(http_requests_total[5m])
// Errors:   rate(http_errors_total{status_code=~"5.."}[5m])
//           / rate(http_requests_total[5m])
// Duration: histogram_quantile(0.99, rate(http_request_duration_seconds_bucket[5m]))
```

### USE Method (cho infrastructure resources)

```
USE = Utilization, Saturation, Errors

UTILIZATION: Resource đang được dùng bao nhiêu %?
SATURATION:  Bao nhiêu work đang phải chờ (queued)?
ERRORS:      Bao nhiêu errors liên quan đến resource?

USE phù hợp cho:
  → CPU, Memory, Disk, Network
  → Connection pools, Thread pools, Queue depths
  → Bất kỳ resource nào có capacity limit
```

```typescript
// ===== USE Metrics cho Node.js runtime =====

// --- CPU ---
const cpuUtilization = new client.Gauge({
  name: 'nodejs_cpu_utilization_ratio',
  help: 'CPU utilization as a ratio (0-1)',
});

// --- Memory ---
const memoryUtilization = new client.Gauge({
  name: 'nodejs_memory_utilization_bytes',
  help: 'Memory utilization in bytes',
  labelNames: ['type'] as const, // rss, heapTotal, heapUsed, external
});

// --- Event Loop ---
const eventLoopLag = new client.Gauge({
  name: 'nodejs_eventloop_lag_seconds',
  help: 'Event loop lag in seconds',
});

const eventLoopUtilization = new client.Gauge({
  name: 'nodejs_eventloop_utilization_ratio',
  help: 'Event loop utilization ratio (0-1)',
});

// --- Connection Pool (Database) ---
const dbPoolUtilization = new client.Gauge({
  name: 'db_pool_utilization',
  help: 'Database connection pool utilization',
  labelNames: ['state'] as const, // active, idle, waiting
});

const dbPoolSaturation = new client.Gauge({
  name: 'db_pool_waiting_count',
  help: 'Number of requests waiting for a DB connection',
});

// ===== Collector: update USE metrics periodically =====

import { performance, monitorEventLoopDelay } from 'node:perf_hooks';
import * as v8 from 'node:v8';

class NodejsMetricsCollector {
  private histogram = monitorEventLoopDelay({ resolution: 20 });
  private previousCpuUsage = process.cpuUsage();
  private previousTimestamp = process.hrtime.bigint();

  constructor(private collectIntervalMs: number = 5000) {}

  start(): void {
    this.histogram.enable();

    setInterval(() => {
      this.collectCpu();
      this.collectMemory();
      this.collectEventLoop();
    }, this.collectIntervalMs);
  }

  private collectCpu(): void {
    const currentUsage = process.cpuUsage();
    const currentTimestamp = process.hrtime.bigint();

    const userDiff = currentUsage.user - this.previousCpuUsage.user;
    const systemDiff = currentUsage.system - this.previousCpuUsage.system;
    const totalCpuMicros = userDiff + systemDiff;

    const elapsedMicros = Number(currentTimestamp - this.previousTimestamp) / 1000;
    const utilization = totalCpuMicros / elapsedMicros;

    cpuUtilization.set(Math.min(1, utilization));

    this.previousCpuUsage = currentUsage;
    this.previousTimestamp = currentTimestamp;
  }

  private collectMemory(): void {
    const mem = process.memoryUsage();
    memoryUtilization.set({ type: 'rss' }, mem.rss);
    memoryUtilization.set({ type: 'heapTotal' }, mem.heapTotal);
    memoryUtilization.set({ type: 'heapUsed' }, mem.heapUsed);
    memoryUtilization.set({ type: 'external' }, mem.external);
  }

  private collectEventLoop(): void {
    // Lag = thời gian chờ trước khi callback được execute
    const lagMs = this.histogram.percentile(99) / 1e6; // ns to ms
    eventLoopLag.set(lagMs / 1000); // ms to seconds

    // Utilization = % thời gian event loop đang bận
    const elu = performance.eventLoopUtilization();
    eventLoopUtilization.set(elu.utilization);
  }

  stop(): void {
    this.histogram.disable();
  }
}

// --- Database Pool USE metrics ---
function collectDbPoolMetrics(pool: any): void {
  // Giả sử pool có getStats() method (pg, knex, typeorm)
  setInterval(() => {
    const stats = pool.getStats?.() ?? {
      totalCount: pool.totalCount ?? 0,
      idleCount: pool.idleCount ?? 0,
      waitingCount: pool.waitingCount ?? 0,
    };
    const activeCount = stats.totalCount - stats.idleCount;

    dbPoolUtilization.set({ state: 'active' }, activeCount);
    dbPoolUtilization.set({ state: 'idle' }, stats.idleCount);
    dbPoolSaturation.set(stats.waitingCount);
  }, 5000);
}
```

## Custom Business Metrics

### Tại sao cần business metrics?

```
Technical metrics cho bạn biết: "API đang chạy, latency ổn"
Business metrics cho bạn biết: "Users đang MUA HÀNG và HÀI LÒNG"

Hệ thống có thể "healthy" về mặt technical nhưng "broken" về business:
  → API 200 OK nhưng trả về empty cart (bug trong logic)
  → Latency tốt nhưng search results sai (relevance issue)
  → Zero errors nhưng zero orders (checkout flow broken)
```

```typescript
const searchQuality = new client.Histogram({

// ===== Instrumentation trong business logic =====

class OrderService {
  constructor(
    private readonly paymentService: PaymentService,
    private readonly inventoryService: InventoryService,
    private readonly logger: AdvancedLogger,
  ) {}

  async placeOrder(command: PlaceOrderCommand): Promise<Order> {
    funnelSteps.inc({ step: 'payment_submitted' });

    const timer = orderProcessingDuration.startTimer({
      payment_method: command.paymentMethod,
    });

    try {
      // Validate inventory
      const inventoryResult = await this.inventoryService.reserve(command.items);
      if (!inventoryResult.success) {
        ordersTotal.inc({ status: 'failed', payment_method: command.paymentMethod, plan: command.plan });
        cartAbandonment.inc({ reason: 'out_of_stock' });
        throw new OutOfStockError(inventoryResult.unavailableItems);
      }

      // Process payment
      const paymentResult = await this.paymentService.charge({
        amount: command.totalAmount,
        currency: command.currency,
        method: command.paymentMethod,
      });

      if (!paymentResult.success) {
        ordersTotal.inc({ status: 'failed', payment_method: command.paymentMethod, plan: command.plan });
        cartAbandonment.inc({ reason: 'payment_failed' });
        await this.inventoryService.release(command.items);
        throw new PaymentFailedError(paymentResult.error);
      }

      // Create order
      const order = await this.createOrder(command, paymentResult);

      // Record business metrics
      ordersTotal.inc({ status: 'success', payment_method: command.paymentMethod, plan: command.plan });
      orderValue.observe({ currency: command.currency }, command.totalAmount);
      revenueTotal.inc({ currency: command.currency, payment_method: command.paymentMethod }, command.totalAmount);
      funnelSteps.inc({ step: 'order_confirmed' });

      timer({ status: 'success' });

      this.logger.info('order.placed', {
        orderId: order.id,
        amount: command.totalAmount,
        currency: command.currency,
        itemCount: command.items.length,
      });

      return order;
    } catch (error) {
      timer({ status: 'failed' });
      throw error;
    }
  }
}

  help: 'Time to process an order end-to-end',
  labelNames: ['payment_method', 'status'] as const,
  buckets: [0.1, 0.5, 1, 2, 5, 10, 30],
});
```

### OpenTelemetry Setup đầy đủ

Section 3.9 giới thiệu manual trace propagation. Ở đây ta setup **OpenTelemetry SDK hoàn chỉnh**.

```typescript
import { NodeSDK } from '@opentelemetry/sdk-node';
import { OTLPTraceExporter } from '@opentelemetry/exporter-trace-otlp-http';
import { OTLPMetricExporter } from '@opentelemetry/exporter-metrics-otlp-http';
import { Resource } from '@opentelemetry/resources';
import {
  ATTR_SERVICE_NAME,
  ATTR_SERVICE_VERSION,
  ATTR_DEPLOYMENT_ENVIRONMENT_NAME,
} from '@opentelemetry/semantic-conventions';
import { HttpInstrumentation } from '@opentelemetry/instrumentation-http';
import { ExpressInstrumentation } from '@opentelemetry/instrumentation-express';
import { PgInstrumentation } from '@opentelemetry/instrumentation-pg';
import { RedisInstrumentation } from '@opentelemetry/instrumentation-redis-4';
import { PeriodicExportingMetricReader } from '@opentelemetry/sdk-metrics';
import {
  ParentBasedSampler,
  TraceIdRatioBasedSampler,
} from '@opentelemetry/sdk-trace-node';

function initTelemetry(config: {
  serviceName: string;
  version: string;
  environment: string;
  otlpEndpoint: string;
  samplingRatio: number;
}): NodeSDK {
  const sdk = new NodeSDK({
    resource: new Resource({
      [ATTR_SERVICE_NAME]: config.serviceName,
      [ATTR_SERVICE_VERSION]: config.version,
      [ATTR_DEPLOYMENT_ENVIRONMENT_NAME]: config.environment,
    }),

    // Trace exporter: gửi traces tới OTLP collector (Jaeger, Tempo, etc.)
    traceExporter: new OTLPTraceExporter({
      url: `${config.otlpEndpoint}/v1/traces`,
    }),

    // Metric reader: gửi metrics tới OTLP collector
    metricReader: new PeriodicExportingMetricReader({
      exporter: new OTLPMetricExporter({
        url: `${config.otlpEndpoint}/v1/metrics`,
      }),
      exportIntervalMillis: 15000,
    }),

    // Sampling strategy
    sampler: new ParentBasedSampler({
      root: new TraceIdRatioBasedSampler(config.samplingRatio),
    }),

    // Auto-instrumentation: tự động tạo spans cho HTTP, DB, Redis
    instrumentations: [
      new HttpInstrumentation({
        // Ignore health check endpoints
        ignoreIncomingRequestHook: (req) =>
          req.url === '/health' || req.url === '/metrics',
      }),
      new ExpressInstrumentation(),
      new PgInstrumentation({
        enhancedDatabaseReporting: true,
      }),
      new RedisInstrumentation(),
    ],
  });

  sdk.start();
  return sdk;
}

// Khởi tạo TRƯỚC KHI import bất kỳ module nào khác
// (instrumentation cần hook vào modules trước khi chúng được loaded)
```

### Custom Spans và Attributes

```typescript
import { trace, SpanStatusCode, SpanKind } from '@opentelemetry/api';

const tracer = trace.getTracer('order-service', '1.5.0');

// ===== Tạo custom span cho business operations =====

async function processPayment(
  orderId: string,
  amount: number,
  currency: string,
): Promise<PaymentResult> {
  return tracer.startActiveSpan('payment.process', {
    kind: SpanKind.CLIENT,
    attributes: {
      'order.id': orderId,
      'payment.amount': amount,
      'payment.currency': currency,
    },
  }, async (span) => {
    try {
      const result = await paymentGateway.charge(amount, currency);

      span.setAttributes({
        'payment.provider': result.provider,
        'payment.transaction_id': result.transactionId,
        'payment.status': result.status,
      });

      if (result.status === 'declined') {
        span.setStatus({
          code: SpanStatusCode.ERROR,
          message: `Payment declined: ${result.declineReason}`,
        });
        // Thêm event cho declined payment
        span.addEvent('payment.declined', {
          reason: result.declineReason,
          retryable: result.retryable,
        });
      } else {
        span.setStatus({ code: SpanStatusCode.OK });
        span.addEvent('payment.captured', {
          transactionId: result.transactionId,
        });
      }

      return result;
    } catch (error) {
      span.setStatus({
        code: SpanStatusCode.ERROR,
        message: (error as Error).message,
      });
      span.recordException(error as Error);
      throw error;
    } finally {
      span.end();
    }
  });
}

// ===== Utility: wrap any function with tracing =====

function traced<TArgs extends unknown[], TReturn>(
  spanName: string,
  fn: (...args: TArgs) => Promise<TReturn>,
  options?: {
    kind?: SpanKind;
    attributes?: Record<string, string | number | boolean>;
  },
): (...args: TArgs) => Promise<TReturn> {
  return (...args: TArgs) => {
    return tracer.startActiveSpan(spanName, {
      kind: options?.kind ?? SpanKind.INTERNAL,
      attributes: options?.attributes,
    }, async (span) => {
      try {
        const result = await fn(...args);
        span.setStatus({ code: SpanStatusCode.OK });
        return result;
      } catch (error) {
        span.setStatus({
          code: SpanStatusCode.ERROR,
          message: (error as Error).message,
        });
        span.recordException(error as Error);
        throw error;
      } finally {
        span.end();
      }
    });
  };
}

// Usage: wrap functions declaratively
const findUserById = traced('user.findById', async (id: string) => {
  return db.query('SELECT * FROM users WHERE id = $1', [id]);
}, { kind: SpanKind.CLIENT });
```

### Trace Sampling Strategies

```
Sampling quyết định TRACES NÀO được thu thập. Ở production, 100% sampling
không khả thi (chi phí lưu trữ, bandwidth, CPU overhead).

Strategies:

1. PROBABILITY SAMPLING
   → Sample X% tất cả traces
   → Đơn giản, nhưng có thể miss rare errors
   → Phù hợp cho: services có traffic đều

2. PARENT-BASED SAMPLING
   → Nếu parent span được sample → sample child cũng
   → Đảm bảo trace hoàn chỉnh (không bị mất span giữa chừng)
   → Phù hợp cho: distributed systems

3. TAIL-BASED SAMPLING (advanced)
   → Thu thập tất cả, nhưng chỉ GIỮ LẠI traces "thú vị"
   → Giữ: errors, high latency, specific user
   → Cần collector buffer → phức tạp hơn
   → Phù hợp cho: high-traffic systems cần debug rare issues

4. RULE-BASED SAMPLING
   → Rules khác nhau cho endpoints khác nhau
   → /health → 0%, /api/checkout → 100%, /api/search → 5%
   → Phù hợp cho: mixed-criticality endpoints

Khuyến nghị:
  Development:  100% (sample everything)
  Staging:      100%
  Production:   5-10% probability + 100% for errors (tail-based)
```

### ELK Stack (Elasticsearch, Logstash, Kibana)

```
Architecture:

  App → stdout → Filebeat → Logstash → Elasticsearch → Kibana
          ↑                     ↑             ↑            ↑
     JSON logs          Parse, enrich     Store, index   Visualize
                        filter, transform  full-text      dashboards

  + Full-text search cực mạnh
  + Kibana UI mature, nhiều features
  + Ecosystem lớn, nhiều plugins
  + Xử lý unstructured logs tốt

Nhược điểm:
  - Resource-heavy (Elasticsearch cần nhiều RAM)
  - Operational overhead cao
  - Chi phí lưu trữ lớn
  - Index management phức tạp
```

### Grafana Loki (lightweight alternative)

```
Architecture:

  App → stdout → Promtail → Loki → Grafana
          ↑          ↑         ↑       ↑
     JSON logs   Ship logs   Index   Query
                              labels  & visualize
                              only

Ưu điểm:
  + Nhẹ hơn ELK rất nhiều (chỉ index labels, không full-text)
  + Chi phí lưu trữ thấp (dùng object storage: S3, GCS)
  + Tích hợp native với Grafana
  + LogQL query language tương tự PromQL

Nhược điểm:
  - Không có full-text search (phải filter bằng labels trước)
  - Không phù hợp cho unstructured logs
  - Query chậm hơn Elasticsearch cho ad-hoc searches
```

## Trace Analysis

### Jaeger / Grafana Tempo

```
Jaeger: full-featured distributed tracing backend
  + UI cho trace visualization
  + Dependency graph
  + Trace comparison
  + Open source, mature

Tempo: Grafana's tracing backend
  + Chỉ cần object storage (S3, GCS) — rẻ
  + Tích hợp native với Grafana
  + TraceQL query language
  + Không cần index — search bằng trace ID hoặc qua Loki/Prometheus

Typical workflow:
  1. Alert fires (từ Prometheus)
  2. Mở Grafana dashboard, thấy latency spike
  3. Click vào data point → link tới exemplar trace
  4. Mở trace trong Jaeger/Tempo → thấy span tree
  5. Tìm bottleneck span → click → xem logs liên quan
```

### Trace Analysis Patterns

```
Pattern 1: CRITICAL PATH ANALYSIS
  Tìm span dài nhất trên critical path
  → Đó là bottleneck, optimize span đó trước

Pattern 2: FAN-OUT ANALYSIS
  Một span có quá nhiều child spans?
  → Có thể N+1 query problem
  → Batch requests thay vì sequential

Pattern 3: ERROR PROPAGATION
  Error xuất phát từ span nào?
  → Root cause analysis: trace từ leaf span có error lên

Pattern 4: COMPARISON
  So sánh trace "nhanh" vs trace "chậm" cùng endpoint
  → Tìm ra khác biệt: missing cache hit? extra DB call?

Pattern 5: DEPENDENCY MAPPING
  Từ traces, tạo dependency graph tự động
  → Service A gọi B, B gọi C
  → Phát hiện unexpected dependencies
```

## Correlation Across Three Pillars

### Tại sao cần correlation?

```
Logs cho bạn biết "CÁI GÌ xảy ra" nhưng không biết "Ở ĐÂU trong hệ thống"
Metrics cho bạn biết "CÓ VẤN ĐỀ" nhưng không biết "VÌ SAO"
Traces cho bạn biết "LUỒNG REQUEST" nhưng thiếu chi tiết

Kết hợp cả 3 → full picture:
  Metric spike → tìm trace liên quan → đọc logs chi tiết
```

### Implementation: Linking Logs ↔ Metrics ↔ Traces

```typescript
// ===== Exemplars: Link Metrics → Traces =====
// Exemplar = sample data point trong metric kèm trace ID
// Khi thấy spike trên graph, click → nhảy thẳng tới trace

const requestDuration = new client.Histogram({
  name: 'http_request_duration_seconds',
  help: 'Request duration',
  labelNames: ['method', 'path'] as const,
  enableExemplars: true, // Enable exemplar support
});

// Khi record metric, attach trace ID
function recordWithExemplar(
  histogram: client.Histogram,
  value: number,
  labels: Record<string, string>,
): void {
  const span = trace.getActiveSpan();
  if (span) {
    const spanContext = span.spanContext();
    histogram.observe(
      { ...labels },
      value,
      {
        traceID: spanContext.traceId,
        spanID: spanContext.spanId,
      },
    );
  } else {
    histogram.observe(labels, value);
  }
}

// ===== Logs ↔ Traces: Automatic correlation =====
// Mọi log entry phải có traceId và spanId

function createCorrelatedLogger(baseLogger: AdvancedLogger): AdvancedLogger {
  // Logger tự động inject trace context từ OpenTelemetry
  return new Proxy(baseLogger, {
    get(target, prop) {
      if (['debug', 'info', 'warn', 'error', 'fatal'].includes(prop as string)) {
        return (message: string, ...args: unknown[]) => {
          const span = trace.getActiveSpan();
          const traceContext = span ? {
            traceId: span.spanContext().traceId,
            spanId: span.spanContext().spanId,
          } : {};

          // Inject trace context vào log data
          const originalMethod = (target as any)[prop];
          if (prop === 'error' || prop === 'fatal') {
            const [error, data] = args as [Error?, Record<string, unknown>?];
            return originalMethod.call(target, message, error, {
              ...data,
              ...traceContext,
            });
          }
          const [data] = args as [Record<string, unknown>?];
          return originalMethod.call(target, message, {
            ...data,
            ...traceContext,
          });
        };
      }
      return (target as any)[prop];
    },
  });
}
```
