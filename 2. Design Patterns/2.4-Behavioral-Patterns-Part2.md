## 19. State

> Cho phép object **thay đổi behavior** khi internal state thay đổi. Object trông như thay đổi class.

### Vấn đề

Object có behavior phụ thuộc vào state hiện tại. Khi state thay đổi, behavior thay đổi theo. Nếu dùng if/else cho mọi method → code trở nên unmanageable.

### ❌ Không có State pattern — if/else nightmare

```typescript
class Order {
  private status: 'draft' | 'pending' | 'paid' | 'shipped' | 'delivered' | 'cancelled';

  submit(): void {
    if (this.status === 'draft') {
      this.status = 'pending';
      this.notifyWarehouse();
    } else {
      throw new Error(`Cannot submit order in ${this.status} state`);
    }
  }

  pay(): void {
    if (this.status === 'pending') {
      this.status = 'paid';
      this.processPayment();
    } else if (this.status === 'draft') {
      throw new Error('Must submit before paying');
    } else {
      throw new Error(`Cannot pay in ${this.status} state`);
    }
  }

  ship(): void {
    if (this.status === 'paid') {
      this.status = 'shipped';
      this.createShipment();
    } else {
      throw new Error(`Cannot ship in ${this.status} state`);
    }
  }

  cancel(): void {
    if (this.status === 'draft' || this.status === 'pending') {
      this.status = 'cancelled';
    } else if (this.status === 'paid') {
      this.status = 'cancelled';
      this.processRefund();
    } else {
      throw new Error(`Cannot cancel in ${this.status} state`);
    }
  }

  // Mỗi method: if/else cho TỪNG state
  // Thêm state mới → sửa TOÀN BỘ methods
  // 6 states × 5 methods = 30 branches to maintain
}
```

### ✅ State Pattern

```typescript
// State interface
interface OrderState {
  readonly name: string;
  submit(order: OrderContext): void;
  pay(order: OrderContext): void;
  ship(order: OrderContext): void;
  deliver(order: OrderContext): void;
  cancel(order: OrderContext): void;
}

// Context — delegates to current state
class OrderContext {
  private state: OrderState;
  private history: Array<{ state: string; timestamp: Date }> = [];

  constructor(
    readonly id: string,
    readonly items: ReadonlyArray<OrderItem>,
    private readonly eventBus: EventBus,
  ) {
    this.state = new DraftState();
    this.recordTransition();
  }

  get currentState(): string { return this.state.name; }
  get stateHistory(): ReadonlyArray<{ state: string; timestamp: Date }> { return this.history; }

  transitionTo(newState: OrderState): void {
    const oldName = this.state.name;
    this.state = newState;
    this.recordTransition();
    this.eventBus.emit('order:stateChanged', {
      orderId: this.id,
      from: oldName,
      to: newState.name,
    });
  }

  // Public methods delegate to state
  submit(): void { this.state.submit(this); }
  pay(): void { this.state.pay(this); }
  ship(): void { this.state.ship(this); }
  deliver(): void { this.state.deliver(this); }
  cancel(): void { this.state.cancel(this); }

  private recordTransition(): void {
    this.history.push({ state: this.state.name, timestamp: new Date() });
  }
}

// Base state — default behavior is throw
abstract class BaseOrderState implements OrderState {
  abstract readonly name: string;

  submit(order: OrderContext): void {
    throw new InvalidTransitionError(this.name, 'submit');
  }
  pay(order: OrderContext): void {
    throw new InvalidTransitionError(this.name, 'pay');
  }
  ship(order: OrderContext): void {
    throw new InvalidTransitionError(this.name, 'ship');
  }
  deliver(order: OrderContext): void {
    throw new InvalidTransitionError(this.name, 'deliver');
  }
  cancel(order: OrderContext): void {
    throw new InvalidTransitionError(this.name, 'cancel');
  }
}

// Concrete States — each only defines VALID transitions
class DraftState extends BaseOrderState {
  readonly name = 'draft';

  submit(order: OrderContext): void {
    // Validate: must have items
    if (order.items.length === 0) {
      throw new EmptyOrderError();
    }
    order.transitionTo(new PendingState());
  }

  cancel(order: OrderContext): void {
    order.transitionTo(new CancelledState());
  }
}

class PendingState extends BaseOrderState {
  readonly name = 'pending';

  pay(order: OrderContext): void {
    order.transitionTo(new PaidState());
  }

  cancel(order: OrderContext): void {
    order.transitionTo(new CancelledState());
  }
}

class PaidState extends BaseOrderState {
  readonly name = 'paid';

  ship(order: OrderContext): void {
    order.transitionTo(new ShippedState());
  }

  cancel(order: OrderContext): void {
    // Paid → cancel requires refund
    order.transitionTo(new RefundingState());
  }
}

class ShippedState extends BaseOrderState {
  readonly name = 'shipped';

  deliver(order: OrderContext): void {
    order.transitionTo(new DeliveredState());
  }

  // Cannot cancel shipped orders
}

class DeliveredState extends BaseOrderState {
  readonly name = 'delivered';
  // Terminal state — no transitions
}

class CancelledState extends BaseOrderState {
  readonly name = 'cancelled';
  // Terminal state
}

class RefundingState extends BaseOrderState {
  readonly name = 'refunding';

  // After refund completes, transition to cancelled
  cancel(order: OrderContext): void {
    order.transitionTo(new CancelledState());
  }
}

// Usage
const order = new OrderContext('ord_1', items, eventBus);
order.submit();    // draft → pending ✅
order.pay();       // pending → paid ✅
order.ship();      // paid → shipped ✅
order.cancel();    // shipped → ❌ InvalidTransitionError
order.deliver();   // shipped → delivered ✅

// Thêm state mới (e.g., "on-hold") = thêm 1 class
// KHÔNG sửa states hiện tại
```

### ✅ Lightweight Alternative: State Machine

```typescript
// Khi không cần full OOP State pattern, dùng state machine definition

type OrderStatus = 'draft' | 'pending' | 'paid' | 'shipped' | 'delivered' | 'cancelled';
type OrderAction = 'submit' | 'pay' | 'ship' | 'deliver' | 'cancel';

// State machine definition — declarative, dễ đọc
const ORDER_TRANSITIONS: Record<OrderStatus, Partial<Record<OrderAction, OrderStatus>>> = {
  draft:     { submit: 'pending', cancel: 'cancelled' },
  pending:   { pay: 'paid', cancel: 'cancelled' },
  paid:      { ship: 'shipped', cancel: 'cancelled' },
  shipped:   { deliver: 'delivered' },
  delivered: {},
  cancelled: {},
};

function transition(currentState: OrderStatus, action: OrderAction): OrderStatus {
  const nextState = ORDER_TRANSITIONS[currentState]?.[action];
  if (!nextState) {
    throw new InvalidTransitionError(currentState, action);
  }
  return nextState;
}

// Tiny, testable, easy to visualize
// Nhưng không support complex side effects per transition
```

### Khi nào dùng

- Object có nhiều states với behavior khác nhau per state
- State transitions có rules phức tạp
- Muốn tránh if/else chain trên state
- Workflows, order processing, game entities, UI components

## 20. Template Method

> Định nghĩa **skeleton** của algorithm trong base class, để subclasses **override specific steps** mà không thay đổi structure.

### Vấn đề

Có algorithm chung nhưng một số steps khác nhau tùy context. Muốn fix algorithm structure, vary individual steps.

### ✅ Real-world: Data Import Pipeline

```typescript
// Base class định nghĩa pipeline, subclass customize steps

abstract class DataImporter<TRaw, TValidated, TDomain> {
  // Template Method — algorithm skeleton, KHÔNG override
  async import(source: DataSource): Promise<ImportReport> {
    const report = new ImportReport(source.name);

    try {
      // Step 1: Extract
      report.startPhase('extract');
      const rawData = await this.extract(source);
      report.completePhase('extract', rawData.length);

      // Step 2: Validate
      report.startPhase('validate');
      const { valid, invalid } = await this.validateAll(rawData);
      report.completePhase('validate', valid.length, invalid.length);

      // Step 3: Transform
      report.startPhase('transform');
      const transformed = await this.transformAll(valid);
      report.completePhase('transform', transformed.length);

      // Step 4: Load
      report.startPhase('load');
      const loaded = await this.loadAll(transformed);
      report.completePhase('load', loaded);

      // Step 5: Post-process (optional hook)
      await this.onComplete(report);

      return report;
    } catch (error) {
      await this.onError(error as Error, report);
      throw error;
    }
  }

  // Abstract steps — MUST override
  protected abstract extract(source: DataSource): Promise<TRaw[]>;
  protected abstract validate(item: TRaw): ValidationResult<TValidated>;
  protected abstract transform(item: TValidated): TDomain;
  protected abstract load(items: TDomain[]): Promise<number>;

  // Hook methods — CAN override (default implementations)
  protected async onComplete(report: ImportReport): Promise<void> {
    // Default: do nothing. Override for notifications, cleanup, etc.
  }

  protected async onError(error: Error, report: ImportReport): Promise<void> {
    console.error(`Import failed at phase: ${report.currentPhase}`, error);
  }

  // Fixed helper methods
  private async validateAll(items: TRaw[]): Promise<{ valid: TValidated[]; invalid: TRaw[] }> {
    const valid: TValidated[] = [];
    const invalid: TRaw[] = [];
    for (const item of items) {
      const result = this.validate(item);
      if (result.isValid) {
        valid.push(result.value);
      } else {
        invalid.push(item);
      }
    }
    return { valid, invalid };
  }

  private async transformAll(items: TValidated[]): Promise<TDomain[]> {
    return items.map(item => this.transform(item));
  }

  private async loadAll(items: TDomain[]): Promise<number> {
    const batchSize = 100;
    let total = 0;
    for (let i = 0; i < items.length; i += batchSize) {
      const batch = items.slice(i, i + batchSize);
      total += await this.load(batch);
    }
    return total;
  }
}

// Concrete: CSV User Import
class CsvUserImporter extends DataImporter<CsvRow, ValidatedUserRow, User> {
  constructor(private readonly userRepo: UserRepository) {
    super();
  }

  protected async extract(source: DataSource): Promise<CsvRow[]> {
    const content = await source.read();
    return parseCsv(content);
  }

  protected validate(row: CsvRow): ValidationResult<ValidatedUserRow> {
    if (!row.email?.includes('@')) return ValidationResult.invalid('Invalid email');
    if (!row.name?.trim()) return ValidationResult.invalid('Name required');
    return ValidationResult.valid({ name: row.name.trim(), email: row.email.toLowerCase() });
  }

  protected transform(row: ValidatedUserRow): User {
    return User.create(row.name, row.email);
  }

  protected async load(users: User[]): Promise<number> {
    return this.userRepo.bulkInsert(users);
  }

  // Override hook: send notification after import
  protected async onComplete(report: ImportReport): Promise<void> {
    await sendSlackNotification(`User import completed: ${report.summary()}`);
  }
}

// Concrete: JSON Product Import
class JsonProductImporter extends DataImporter<JsonProduct, ValidatedProduct, Product> {
  protected async extract(source: DataSource): Promise<JsonProduct[]> {
    const content = await source.read();
    return JSON.parse(content);
  }

  protected validate(item: JsonProduct): ValidationResult<ValidatedProduct> {
    if (item.price <= 0) return ValidationResult.invalid('Price must be positive');
    if (!item.sku) return ValidationResult.invalid('SKU required');
    return ValidationResult.valid(item as ValidatedProduct);
  }

  protected transform(item: ValidatedProduct): Product {
    return Product.create(item.sku, item.name, Money.fromCents(item.price));
  }

  protected async load(products: Product[]): Promise<number> {
    return this.productRepo.bulkUpsert(products);
  }
}

// Usage — algorithm structure fixed, steps customized
const userImporter = new CsvUserImporter(userRepo);
const report = await userImporter.import(new FileDataSource('users.csv'));
```

### Template Method vs Strategy

```
Template Method: Fix algorithm structure, vary STEPS (inheritance)
Strategy: Fix interface, vary ENTIRE algorithm (composition)

Template Method: "Implement step 3 differently"
Strategy: "Replace the whole algorithm"

Prefer Strategy khi có thể — composition > inheritance
Use Template Method khi algorithm structure must be fixed
```

### Khi nào dùng

- Algorithm có fixed structure, variable steps
- Multiple implementations share 90% logic, differ 10%
- Framework hooks (lifecycle methods)
- ETL pipelines, report generators, test fixtures

## 21. Memento

> Capture và externalize object's internal **state** để có thể **restore** về state đó sau này, mà không violate encapsulation.

### Vấn đề

Cần save/restore object state (undo, checkpoints, time travel debugging) mà không expose internal details.

### ✅ Real-world: Form State Snapshots

```typescript
// Memento: immutable snapshot of state
interface FormMemento {
  readonly id: string;
  readonly timestamp: Date;
  readonly label: string;
  readonly data: Readonly<Record<string, unknown>>; // Opaque to outside
}

// Originator: object whose state we want to save/restore
class FormState {
  private fields: Record<string, unknown> = {};
  private validationErrors: Record<string, string[]> = {};
  private isDirty = false;

  setField(name: string, value: unknown): void {
    this.fields[name] = value;
    this.isDirty = true;
    this.validateField(name);
  }

  getField(name: string): unknown {
    return this.fields[name];
  }

  // Save state — create memento
  save(label: string): FormMemento {
    return {
      id: crypto.randomUUID(),
      timestamp: new Date(),
      label,
      data: structuredClone({
        fields: this.fields,
        validationErrors: this.validationErrors,
        isDirty: this.isDirty,
      }),
    };
  }

  // Restore state from memento
  restore(memento: FormMemento): void {
    const data = memento.data as any;
    this.fields = structuredClone(data.fields);
    this.validationErrors = structuredClone(data.validationErrors);
    this.isDirty = data.isDirty;
  }

  private validateField(name: string): void { /* ... */ }
}

// Caretaker: manages mementos (doesn't look inside them)
class FormHistory {
  private snapshots: FormMemento[] = [];
  private currentIndex = -1;
  private maxSnapshots: number;

  constructor(maxSnapshots = 50) {
    this.maxSnapshots = maxSnapshots;
  }

  save(memento: FormMemento): void {
    // Remove future snapshots if we've undone
    this.snapshots = this.snapshots.slice(0, this.currentIndex + 1);
    this.snapshots.push(memento);
    this.currentIndex = this.snapshots.length - 1;

    // Limit memory
    if (this.snapshots.length > this.maxSnapshots) {
      this.snapshots.shift();
      this.currentIndex--;
    }
  }

  undo(): FormMemento | null {
    if (this.currentIndex <= 0) return null;
    this.currentIndex--;
    return this.snapshots[this.currentIndex];
  }

  redo(): FormMemento | null {
    if (this.currentIndex >= this.snapshots.length - 1) return null;
    this.currentIndex++;
    return this.snapshots[this.currentIndex];
  }

  get canUndo(): boolean { return this.currentIndex > 0; }
  get canRedo(): boolean { return this.currentIndex < this.snapshots.length - 1; }

  getSnapshots(): ReadonlyArray<{ id: string; label: string; timestamp: Date }> {
    return this.snapshots.map(s => ({ id: s.id, label: s.label, timestamp: s.timestamp }));
  }
}

// Usage
const form = new FormState();
const history = new FormHistory();

// Auto-save on changes
form.setField('name', 'Alice');
history.save(form.save('Changed name'));

form.setField('email', 'alice@example.com');
history.save(form.save('Changed email'));

form.setField('name', 'Bob'); // Oops, wrong name
history.save(form.save('Changed name to Bob'));

// Undo!
const previous = history.undo();
if (previous) form.restore(previous);
// name is back to 'Alice', email still 'alice@example.com'
```

## 22. Visitor

> Cho phép thêm **operations mới** cho objects mà **không sửa** classes của chúng.

### Vấn đề

Có hierarchy objects ổn định (hiếm thêm type mới), nhưng thường xuyên thêm operations mới. Visitor tách operation khỏi object structure.

### ✅ Real-world: AST Processing

```typescript
// Abstract Syntax Tree nodes — structure ổn định

// Node types
interface NumberLiteral { type: 'number'; value: number }
interface StringLiteral { type: 'string'; value: string }
interface BinaryExpression { type: 'binary'; operator: '+' | '-' | '*' | '/'; left: AstNode; right: AstNode }
interface UnaryExpression { type: 'unary'; operator: '-' | '!'; operand: AstNode }
interface FunctionCall { type: 'call'; name: string; args: AstNode[] }
interface Identifier { type: 'identifier'; name: string }

type AstNode = NumberLiteral | StringLiteral | BinaryExpression | UnaryExpression | FunctionCall | Identifier;

// Visitor interface — mỗi method xử lý 1 node type
interface AstVisitor<TResult> {
  visitNumber(node: NumberLiteral): TResult;
  visitString(node: StringLiteral): TResult;
  visitBinary(node: BinaryExpression): TResult;
  visitUnary(node: UnaryExpression): TResult;
  visitCall(node: FunctionCall): TResult;
  visitIdentifier(node: Identifier): TResult;
}

// Accept function — dispatch to correct visitor method
function accept<T>(node: AstNode, visitor: AstVisitor<T>): T {
  switch (node.type) {
    case 'number': return visitor.visitNumber(node);
    case 'string': return visitor.visitString(node);
    case 'binary': return visitor.visitBinary(node);
    case 'unary': return visitor.visitUnary(node);
    case 'call': return visitor.visitCall(node);
    case 'identifier': return visitor.visitIdentifier(node);
  }
}

// Visitor 1: Evaluate expression
class Evaluator implements AstVisitor<number> {
  constructor(private readonly env: Record<string, number> = {}) {}

  visitNumber(node: NumberLiteral): number {
    return node.value;
  }

  visitString(_node: StringLiteral): number {
    throw new Error('Cannot evaluate string as number');
  }

  visitBinary(node: BinaryExpression): number {
    const left = accept(node.left, this);
    const right = accept(node.right, this);
    switch (node.operator) {
      case '+': return left + right;
      case '-': return left - right;
      case '*': return left * right;
      case '/':
        if (right === 0) throw new Error('Division by zero');
        return left / right;
    }
  }

  visitUnary(node: UnaryExpression): number {
    const operand = accept(node.operand, this);
    switch (node.operator) {
      case '-': return -operand;
      case '!': return operand === 0 ? 1 : 0;
    }
  }

  visitCall(node: FunctionCall): number {
    const args = node.args.map(arg => accept(arg, this));
    switch (node.name) {
      case 'max': return Math.max(...args);
      case 'min': return Math.min(...args);
      case 'abs': return Math.abs(args[0]);
      default: throw new Error(`Unknown function: ${node.name}`);
    }
  }

  visitIdentifier(node: Identifier): number {
    if (!(node.name in this.env)) throw new Error(`Undefined variable: ${node.name}`);
    return this.env[node.name];
  }
}

// Visitor 2: Pretty Printer — thêm operation mới KHÔNG sửa AST nodes
class PrettyPrinter implements AstVisitor<string> {
  visitNumber(node: NumberLiteral): string { return String(node.value); }
  visitString(node: StringLiteral): string { return `"${node.value}"`; }

  visitBinary(node: BinaryExpression): string {
    return `(${accept(node.left, this)} ${node.operator} ${accept(node.right, this)})`;
  }

  visitUnary(node: UnaryExpression): string {
    return `${node.operator}${accept(node.operand, this)}`;
  }

  visitCall(node: FunctionCall): string {
    const args = node.args.map(a => accept(a, this)).join(', ');
    return `${node.name}(${args})`;
  }

  visitIdentifier(node: Identifier): string { return node.name; }
}

// Visitor 3: Type Checker — lại thêm operation, KHÔNG sửa AST
class TypeChecker implements AstVisitor<'number' | 'string' | 'unknown'> {
  visitNumber(): 'number' { return 'number'; }
  visitString(): 'string' { return 'string'; }

  visitBinary(node: BinaryExpression): 'number' | 'string' | 'unknown' {
    const left = accept(node.left, this);
    const right = accept(node.right, this);
    if (node.operator === '+' && (left === 'string' || right === 'string')) return 'string';
    return 'number';
  }

  visitUnary(): 'number' { return 'number'; }
  visitCall(): 'number' { return 'number'; }
  visitIdentifier(): 'unknown' { return 'unknown'; }
}

// Usage
const ast: AstNode = {
  type: 'binary',
  operator: '+',
  left: { type: 'number', value: 3 },
  right: {
    type: 'binary',
    operator: '*',
    left: { type: 'identifier', name: 'x' },
    right: { type: 'number', value: 5 },
  },
};

const evaluator = new Evaluator({ x: 10 });
const printer = new PrettyPrinter();

console.log(accept(ast, printer));    // "(3 + (x * 5))"
console.log(accept(ast, evaluator));  // 53
```

### TypeScript Alternative: Discriminated Union + Functions

```typescript
// Trong TypeScript, discriminated unions + switch thường đơn giản hơn visitor

type Shape =
  | { type: 'circle'; radius: number }
  | { type: 'rectangle'; width: number; height: number }
  | { type: 'triangle'; base: number; height: number };

// "Visitor" as plain functions
function area(shape: Shape): number {
  switch (shape.type) {
    case 'circle': return Math.PI * shape.radius ** 2;
    case 'rectangle': return shape.width * shape.height;
    case 'triangle': return 0.5 * shape.base * shape.height;
  }
}

function perimeter(shape: Shape): number {
  switch (shape.type) {
    case 'circle': return 2 * Math.PI * shape.radius;
    case 'rectangle': return 2 * (shape.width + shape.height);
    case 'triangle': return shape.base + shape.height + Math.sqrt(shape.base ** 2 + shape.height ** 2);
  }
}

// Thêm operation mới (describe, serialize...) = thêm function mới
// Thêm shape mới = sửa tất cả functions (compiler enforces via exhaustive check)
```

### Visitor vs Discriminated Union — Khi nào dùng gì

| | Visitor (OOP) | Discriminated Union (FP) |
|--|--------------|--------------------------|
| **Thêm operation** | Dễ (thêm visitor class) | Dễ (thêm function) |
| **Thêm type** | Khó (sửa visitor interface) | Khó (sửa type + tất cả functions) |
| **State** | Visitor có thể accumulate state | Functions stateless (truyền qua params) |
| **Khi nào** | Types ổn định, ops thay đổi thường | Cả hai thay đổi, structure đơn giản |
| **TypeScript** | Complex AST, compiler internals | Phần lớn use cases |

## 23. Interpreter

> Định nghĩa **grammar** cho language đơn giản và **interpreter** để evaluate sentences trong language đó.

### Vấn đề

Cần xử lý domain-specific language (DSL), query language, rule expressions, template language.

> **Note:** Interpreter pattern ít khi implement đầy đủ trong production. Thường thay bằng parser libraries (PEG.js, nearley, Chevrotain) hoặc đơn giản hơn dùng JSON/function-based rules. Tuy nhiên, hiểu pattern giúp bạn thiết kế rule engines, expression evaluators.

### ✅ Real-world: Business Rule Engine

```typescript
// DSL cho filtering rules, ví dụ: "age > 18 AND (country = 'US' OR country = 'UK')"

// Expression types
type Expression =
  | { type: 'comparison'; field: string; operator: '=' | '!=' | '>' | '<' | '>=' | '<='; value: unknown }
  | { type: 'and'; left: Expression; right: Expression }
  | { type: 'or'; left: Expression; right: Expression }
  | { type: 'not'; operand: Expression }
  | { type: 'in'; field: string; values: unknown[] }
  | { type: 'contains'; field: string; substring: string }
  | { type: 'exists'; field: string }
  | { type: 'true' }
  | { type: 'false' };

// Interpreter
function evaluate(expr: Expression, context: Record<string, unknown>): boolean {
  switch (expr.type) {
    case 'true': return true;
    case 'false': return false;

    case 'comparison': {
      const fieldValue = getFieldValue(context, expr.field);
      switch (expr.operator) {
        case '=': return fieldValue === expr.value;
        case '!=': return fieldValue !== expr.value;
        case '>': return (fieldValue as number) > (expr.value as number);
        case '<': return (fieldValue as number) < (expr.value as number);
        case '>=': return (fieldValue as number) >= (expr.value as number);
        case '<=': return (fieldValue as number) <= (expr.value as number);
      }
    }

    case 'and': return evaluate(expr.left, context) && evaluate(expr.right, context);
    case 'or': return evaluate(expr.left, context) || evaluate(expr.right, context);
    case 'not': return !evaluate(expr.operand, context);

    case 'in': {
      const value = getFieldValue(context, expr.field);
      return expr.values.includes(value);
    }

    case 'contains': {
      const value = String(getFieldValue(context, expr.field));
      return value.includes(expr.substring);
    }

    case 'exists': {
      return getFieldValue(context, expr.field) !== undefined;
    }
  }
}

function getFieldValue(context: Record<string, unknown>, path: string): unknown {
  return path.split('.').reduce<unknown>(
    (obj, key) => (obj as any)?.[key],
    context,
  );
}

// Builder API cho rule construction (thay vì parse string)
const Rule = {
  field: (name: string) => ({
    equals: (value: unknown): Expression => ({ type: 'comparison', field: name, operator: '=', value }),
    notEquals: (value: unknown): Expression => ({ type: 'comparison', field: name, operator: '!=', value }),
    greaterThan: (value: number): Expression => ({ type: 'comparison', field: name, operator: '>', value }),
    lessThan: (value: number): Expression => ({ type: 'comparison', field: name, operator: '<', value }),
    in: (...values: unknown[]): Expression => ({ type: 'in', field: name, values }),
    contains: (substring: string): Expression => ({ type: 'contains', field: name, substring }),
    exists: (): Expression => ({ type: 'exists', field: name }),
  }),
  and: (left: Expression, right: Expression): Expression => ({ type: 'and', left, right }),
  or: (left: Expression, right: Expression): Expression => ({ type: 'or', left, right }),
  not: (operand: Expression): Expression => ({ type: 'not', operand }),
  allOf: (...exprs: Expression[]): Expression => exprs.reduce((acc, e) => Rule.and(acc, e)),
  anyOf: (...exprs: Expression[]): Expression => exprs.reduce((acc, e) => Rule.or(acc, e)),
} as const;

// Usage — type-safe, readable rule construction
const eligibilityRule = Rule.allOf(
  Rule.field('age').greaterThan(18),
  Rule.anyOf(
    Rule.field('country').in('US', 'UK', 'CA'),
    Rule.field('isPremium').equals(true),
  ),
  Rule.field('email').exists(),
  Rule.not(Rule.field('status').equals('banned')),
);

// Evaluate against user data
const user = { age: 25, country: 'US', isPremium: false, email: 'a@b.com', status: 'active' };
const isEligible = evaluate(eligibilityRule, user); // true

// Rules có thể persist (JSON serializable), share, version control
const ruleJson = JSON.stringify(eligibilityRule);
// Store in DB, send to frontend, etc.
```

### ✅ Simple Template Interpreter

```typescript
// Mini template language: "Hello {{name}}, you have {{count}} messages"

interface TemplateToken =
  | { type: 'text'; value: string }
  | { type: 'variable'; name: string }
  | { type: 'conditional'; variable: string; trueBranch: TemplateToken[]; falseBranch: TemplateToken[] };

// Parser (simplified)
function parseTemplate(template: string): TemplateToken[] {
  const tokens: TemplateToken[] = [];
  let remaining = template;

  while (remaining.length > 0) {
    const varStart = remaining.indexOf('{{');
    if (varStart === -1) {
      tokens.push({ type: 'text', value: remaining });
      break;
    }

    if (varStart > 0) {
      tokens.push({ type: 'text', value: remaining.slice(0, varStart) });
    }

    const varEnd = remaining.indexOf('}}', varStart);
    if (varEnd === -1) throw new Error('Unclosed template variable');

    const varName = remaining.slice(varStart + 2, varEnd).trim();
    tokens.push({ type: 'variable', name: varName });

    remaining = remaining.slice(varEnd + 2);
  }

  return tokens;
}

// Interpreter
function interpret(tokens: TemplateToken[], context: Record<string, unknown>): string {
  return tokens.map(token => {
    switch (token.type) {
      case 'text':
        return token.value;
      case 'variable':
        return String(context[token.name] ?? '');
      case 'conditional':
        const branchTokens = context[token.variable] ? token.trueBranch : token.falseBranch;
        return interpret(branchTokens, context);
    }
  }).join('');
}

// Usage
const template = parseTemplate('Hello {{name}}, you have {{count}} new messages.');
const result = interpret(template, { name: 'Alice', count: 5 });
// "Hello Alice, you have 5 new messages."
```

### Khi nào dùng

- Domain-specific languages (DSL) đơn giản
- Rule engines (business rules, feature flags, filters)
- Template languages
- Expression evaluation (calculator, formula parser)

### Khi nào KHÔNG dùng

- Complex grammars → dùng parser generator (ANTLR, PEG.js)
- Performance-critical → compiled rules nhanh hơn interpreted
- Khi JSON config + code đủ → đừng invent language mới
