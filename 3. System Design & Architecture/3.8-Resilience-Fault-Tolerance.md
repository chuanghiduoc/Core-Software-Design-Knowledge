## Circuit Breaker

Ngăn cascade failure bằng cách **stop calling** service đang fail, cho nó thời gian recover.

```
States:
  CLOSED → requests flow normally, track failures
  OPEN → all requests fail immediately (fast fail), don't hit broken service
  HALF-OPEN → allow limited requests to test if service recovered

       failure threshold reached
CLOSED ──────────────────────────► OPEN
  ▲                                  │
  │    success threshold reached     │ timeout expires
  │                                  ▼
  └─────────────────────────── HALF-OPEN
```

```typescript
interface CircuitBreakerOptions {
  failureThreshold: number;       // Failures before opening (default: 5)
  successThreshold: number;       // Successes in half-open before closing (default: 3)
  timeout: number;                // Time in OPEN state before trying half-open (ms)
  monitorWindowMs: number;        // Window to track failure rate
}

class CircuitBreaker {
  private state: 'closed' | 'open' | 'half-open' = 'closed';
  private failureCount = 0;
  private successCount = 0;
  private lastFailureTime = 0;
  private readonly options: CircuitBreakerOptions;

  constructor(
    private readonly name: string,
    options?: Partial<CircuitBreakerOptions>,
  ) {
    this.options = {
      failureThreshold: options?.failureThreshold ?? 5,
      successThreshold: options?.successThreshold ?? 3,
      timeout: options?.timeout ?? 30_000,
      monitorWindowMs: options?.monitorWindowMs ?? 60_000,
    };
  }

  async execute<T>(fn: () => Promise<T>): Promise<T> {
    if (this.state === 'open') {
      if (Date.now() - this.lastFailureTime >= this.options.timeout) {
        this.state = 'half-open';
        this.successCount = 0;
      } else {
        throw new CircuitOpenError(
          `Circuit "${this.name}" is OPEN. Service unavailable.`
        );
      }
    }

    try {
      const result = await fn();
      this.onSuccess();
      return result;
    } catch (error) {
      this.onFailure();
      throw error;
    }
  }

  private onSuccess(): void {
    if (this.state === 'half-open') {
      this.successCount++;
      if (this.successCount >= this.options.successThreshold) {
        this.state = 'closed';
        this.failureCount = 0;
        console.log(`Circuit "${this.name}" CLOSED — service recovered`);
      }
    } else {
      this.failureCount = 0;
    }
  }

  private onFailure(): void {
    this.failureCount++;
    this.lastFailureTime = Date.now();

    if (this.state === 'half-open') {
      this.state = 'open';
      console.log(`Circuit "${this.name}" OPEN — service still failing`);
    } else if (this.failureCount >= this.options.failureThreshold) {
      this.state = 'open';
      console.log(`Circuit "${this.name}" OPEN — failure threshold reached`);
    }
  }

  getState(): string { return this.state; }
}

// Usage
const paymentCircuit = new CircuitBreaker('payment-service', {
  failureThreshold: 3,
  timeout: 30_000,
});

class PaymentClient {
  async charge(amount: Money): Promise<ChargeResult> {
    return paymentCircuit.execute(async () => {
      const response = await fetch('https://payment-service/charge', {
        method: 'POST',
        body: JSON.stringify({ amount: amount.cents }),
        signal: AbortSignal.timeout(5000),
      });
      if (!response.ok) throw new Error(`Payment failed: ${response.status}`);
      return response.json();
    });
  }
}

// After 3 failures: circuit OPENS
// → All subsequent calls fail IMMEDIATELY (no network call)
// → Payment service gets time to recover
// → After 30s: circuit goes HALF-OPEN, tries 1 request
// → If success: circuit CLOSES
// → If fail: circuit stays OPEN for another 30s
```

## Retry with Backoff

```typescript
interface RetryOptions {
  maxRetries: number;
  baseDelayMs: number;
  maxDelayMs: number;
  retryableErrors?: Array<new (...args: any[]) => Error>;
  retryableStatusCodes?: Set<number>;
}

async function withRetry<T>(
  fn: () => Promise<T>,
  options: RetryOptions,
): Promise<T> {
  let lastError: Error | undefined;

  for (let attempt = 0; attempt <= options.maxRetries; attempt++) {
    try {
      return await fn();
    } catch (error) {
      lastError = error as Error;

      // Check if retryable
      if (!isRetryable(error, options)) throw error;

      // Last attempt — don't delay, just throw
      if (attempt === options.maxRetries) break;

      // Exponential backoff with jitter
      const baseDelay = options.baseDelayMs * 2 ** attempt;
      const jitter = Math.random() * baseDelay * 0.1;
      const delay = Math.min(baseDelay + jitter, options.maxDelayMs);

      console.warn(
        `Retry ${attempt + 1}/${options.maxRetries} after ${Math.round(delay)}ms`,
        { error: String(error) },
      );

      await new Promise(resolve => setTimeout(resolve, delay));
    }
  }

  throw lastError;
}

function isRetryable(error: unknown, options: RetryOptions): boolean {
  // Network errors
  if (error instanceof TypeError && String(error).includes('fetch')) return true;

  // HTTP status codes
  if (error instanceof HttpError && options.retryableStatusCodes?.has(error.status)) return true;

  // Specific error types
  if (options.retryableErrors?.some(E => error instanceof E)) return true;

  return false;
}

// Usage
const result = await withRetry(
  () => fetch('https://api.external.com/data').then(r => {
    if (!r.ok) throw new HttpError(r.status, r.statusText);
    return r.json();
  }),
  {
    maxRetries: 3,
    baseDelayMs: 1000,
    maxDelayMs: 10_000,
    retryableStatusCodes: new Set([408, 429, 500, 502, 503, 504]),
  },
);
// Attempt 1: fails with 503
// Wait 1000ms + jitter
// Attempt 2: fails with 503
// Wait 2000ms + jitter
// Attempt 3: succeeds → return result
```

## Timeout

```typescript
// EVERY external call MUST have a timeout. No exceptions.

// ❌ No timeout — request hangs forever if service is stuck
const response = await fetch('https://slow-service.com/api');

// ✅ Always set timeouts
const response = await fetch('https://slow-service.com/api', {
  signal: AbortSignal.timeout(5000), // 5 second timeout
});

// Generic timeout wrapper
async function withTimeout<T>(
  promise: Promise<T>,
  timeoutMs: number,
  message = 'Operation timed out',
): Promise<T> {
  const timeout = new Promise<never>((_, reject) => {
    const timer = setTimeout(() => reject(new TimeoutError(message)), timeoutMs);
    // Prevent timer from keeping process alive
    if (timer.unref) timer.unref();
  });

  return Promise.race([promise, timeout]);
}

// Usage
const user = await withTimeout(
  userService.findById(userId),
  3000,
  `User lookup timed out (${userId})`,
);

// Timeout budget across call chain
// API has 10s budget
// → DB query: 3s timeout
// → Payment service: 5s timeout
// → Notification: 2s timeout (fire-and-forget if slow)
```

## Bulkhead

Isolate failures — 1 component failing shouldn't consume all resources.

```typescript
// Semaphore-based bulkhead: limit concurrent operations

class Bulkhead {
  private running = 0;
  private queue: Array<{
    resolve: (value: void) => void;
    reject: (error: Error) => void;
  }> = [];

  constructor(
    private readonly name: string,
    private readonly maxConcurrency: number,
    private readonly maxQueueSize: number = 100,
  ) {}

  async execute<T>(fn: () => Promise<T>): Promise<T> {
    await this.acquire();
    try {
      return await fn();
    } finally {
      this.release();
    }
  }

  private acquire(): Promise<void> {
    if (this.running < this.maxConcurrency) {
      this.running++;
      return Promise.resolve();
    }

    if (this.queue.length >= this.maxQueueSize) {
      return Promise.reject(
        new BulkheadFullError(`Bulkhead "${this.name}" is full (${this.running} running, ${this.queue.length} queued)`)
      );
    }

    return new Promise<void>((resolve, reject) => {
      this.queue.push({ resolve, reject });
    });
  }

  private release(): void {
    this.running--;
    const next = this.queue.shift();
    if (next) {
      this.running++;
      next.resolve();
    }
  }
}

// Usage: Isolate different service calls
const paymentBulkhead = new Bulkhead('payment', 10, 50);    // Max 10 concurrent payment calls
const emailBulkhead = new Bulkhead('email', 20, 100);       // Max 20 concurrent email calls
const searchBulkhead = new Bulkhead('search', 5, 20);       // Max 5 concurrent search calls

// Payment service being slow won't consume all connections
// Email service down won't block payment processing
async function processOrder(order: Order): Promise<void> {
  await paymentBulkhead.execute(() => paymentService.charge(order.total));
  await emailBulkhead.execute(() => emailService.sendConfirmation(order));
}
```

## Graceful Degradation

```typescript
// Khi dependency down → degrade gracefully, don't crash

class ProductService {
  constructor(
    private readonly db: Pool,
    private readonly searchClient: ElasticsearchClient,
    private readonly recommendationClient: RecommendationClient,
    private readonly cache: Redis,
  ) {}

  async getProductPage(productId: string): Promise<ProductPageResponse> {
    // Core data: MUST succeed (throw if fails)
    const product = await this.db.query('SELECT * FROM products WHERE id = $1', [productId]);
    if (!product.rows[0]) throw new NotFoundError('Product', productId);

    // Recommendations: NICE TO HAVE (fallback to empty)
    let recommendations: Product[] = [];
    try {
      recommendations = await withTimeout(
        this.recommendationClient.getRelated(productId),
        2000,
      );
    } catch {
      // Recommendation service down → show page without recommendations
      console.warn(`Recommendations unavailable for product ${productId}`);
    }

    // Reviews: NICE TO HAVE (fallback to cached version)
    let reviews: Review[] = [];
    try {
      reviews = await this.getReviews(productId);
    } catch {
      // Try stale cache
      const cached = await this.cache.get(`reviews:${productId}`);
      if (cached) {
        reviews = JSON.parse(cached);
        console.warn(`Serving stale reviews for product ${productId}`);
      }
    }

    return {
      product: product.rows[0],
      recommendations,
      reviews,
      // Tell frontend what's degraded
      degraded: {
        recommendations: recommendations.length === 0,
        reviews: false,
      },
    };
  }
}
```

## Rate Limiting (Server-side)

```typescript
// Protect services from being overwhelmed

// Sliding Window Rate Limiter (Redis-based)
class RateLimiter {
  constructor(
    private readonly redis: Redis,
    private readonly limit: number,
    private readonly windowMs: number,
  ) {}

  async isAllowed(key: string): Promise<{ allowed: boolean; remaining: number; retryAfterMs: number | null }> {
    const now = Date.now();
    const windowStart = now - this.windowMs;
    const redisKey = `ratelimit:${key}`;

    // Atomic operation using Redis pipeline
    const pipeline = this.redis.pipeline();
    pipeline.zremrangebyscore(redisKey, 0, windowStart);   // Remove old entries
    pipeline.zadd(redisKey, now, `${now}:${Math.random()}`); // Add current
    pipeline.zcard(redisKey);                                 // Count in window
    pipeline.expire(redisKey, Math.ceil(this.windowMs / 1000)); // TTL

    const results = await pipeline.exec();
    const count = (results?.[2]?.[1] as number) ?? 0;

    if (count > this.limit) {
      // Get oldest entry to calculate retry-after
      const oldest = await this.redis.zrange(redisKey, 0, 0, 'WITHSCORES');
      const retryAfterMs = oldest.length >= 2
        ? Number(oldest[1]) + this.windowMs - now
        : this.windowMs;

      return { allowed: false, remaining: 0, retryAfterMs };
    }

    return { allowed: true, remaining: this.limit - count, retryAfterMs: null };
  }
}

// Middleware
function rateLimitMiddleware(limiter: RateLimiter, keyExtractor: (req: Request) => string) {
  return async (req: Request, res: Response, next: NextFunction) => {
    const key = keyExtractor(req);
    const result = await limiter.isAllowed(key);

    res.setHeader('X-RateLimit-Limit', limiter.limit);
    res.setHeader('X-RateLimit-Remaining', result.remaining);

    if (!result.allowed) {
      res.setHeader('Retry-After', Math.ceil((result.retryAfterMs ?? 1000) / 1000));
      return res.status(429).json({
        error: { code: 'RATE_LIMITED', message: 'Too many requests', retryAfterMs: result.retryAfterMs },
      });
    }

    next();
  };
}
```

## Health Checks & Self-Healing

### Graceful Shutdown

```typescript
// Handle SIGTERM properly — finish in-flight requests, close connections

function setupGracefulShutdown(
  server: http.Server,
  cleanup: () => Promise<void>,
  timeoutMs = 30_000,
): void {
  let isShuttingDown = false;

  async function shutdown(signal: string): Promise<void> {
    if (isShuttingDown) return;
    isShuttingDown = true;

    console.log(`${signal} received. Starting graceful shutdown...`);

    // Stop accepting new connections
    server.close();

    // Force shutdown after timeout
    const forceTimeout = setTimeout(() => {
      console.error('Graceful shutdown timed out. Forcing exit.');
      process.exit(1);
    }, timeoutMs);

    try {
      // Wait for in-flight requests to complete
      await new Promise<void>((resolve) => {
        server.close(() => resolve());
      });

      // Cleanup: close DB pools, Redis connections, etc.
      await cleanup();

      console.log('Graceful shutdown complete.');
      clearTimeout(forceTimeout);
      process.exit(0);
    } catch (error) {
      console.error('Error during shutdown:', error);
      clearTimeout(forceTimeout);
      process.exit(1);
    }
  }

  process.on('SIGTERM', () => shutdown('SIGTERM'));
  process.on('SIGINT', () => shutdown('SIGINT'));

  // Reject new requests during shutdown
  server.on('request', (req: Request, res: Response) => {
    if (isShuttingDown) {
      res.setHeader('Connection', 'close');
    }
  });
}

// Usage
const server = app.listen(3000);

setupGracefulShutdown(server, async () => {
  await dbPool.end();
  await redis.quit();
  await rabbitConnection.close();
  console.log('All connections closed');
});
```

### Resilience Patterns Summary

```
                    ┌─────────────┐
  Request ─────────►│  Timeout    │ Fail fast if too slow
                    └──────┬──────┘
                           │
                    ┌──────▼──────┐
                    │   Retry     │ Retry transient failures
                    └──────┬──────┘
                           │
                    ┌──────▼──────┐
                    │  Circuit    │ Stop calling dead services
                    │  Breaker    │
                    └──────┬──────┘
                           │
                    ┌──────▼──────┐
                    │  Bulkhead   │ Isolate resource usage
                    └──────┬──────┘
                           │
                    ┌──────▼──────┐
                    │  Fallback   │ Degrade gracefully
                    └──────┬──────┘
                           │
                       Response

Every external dependency should be wrapped with:
Timeout → Retry → Circuit Breaker → Fallback
```
