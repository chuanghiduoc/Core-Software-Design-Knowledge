## Tại sao Inheritance là vấn đề

### Fragile Base Class Problem

```typescript
// Ngày 1: Base class hoạt động tốt
class Collection<T> {
  protected items: T[] = [];

  add(item: T): void {
    this.items.push(item);
  }

  addAll(items: T[]): void {
    items.forEach(item => this.add(item)); // Gọi this.add → subclass có thể override
  }

  count(): number {
    return this.items.length;
  }
}

// Ngày 2: Cần đếm số lần add
class CountingCollection<T> extends Collection<T> {
  private addCount = 0;

  add(item: T): void {
    this.addCount++;
    super.add(item);
  }

  // BUG: addAll() gọi this.add() → mỗi item bị count 2 lần!
  // addAll([a, b, c]) → addCount = 6 thay vì 3
  // Vì addAll trong parent gọi this.add, mà this.add đã bị override

  getAddCount(): number {
    return this.addCount;
  }
}

// Ngày 30: Ai đó optimize base class
class Collection<T> {
  addAll(items: T[]): void {
    // Tối ưu: push trực tiếp thay vì gọi add() từng cái
    this.items.push(...items);
    // → CountingCollection.addAll KHÔNG còn trigger count!
    // → Bug ngược: addCount bây giờ THIẾU
  }
}

// Đây là Fragile Base Class: internal changes ở parent phá subclass
```

## Composition: Cách tiếp cận đúng

### Basic Composition — Has-a thay vì Is-a

```typescript
// ❌ Inheritance: "A Logger IS a FileWriter"
class FileWriter {
  write(path: string, content: string): void { /* ... */ }
}

class Logger extends FileWriter {
  log(message: string): void {
    this.write('/var/log/app.log', `[${new Date().toISOString()}] ${message}`);
  }
}

// ✅ Composition: "A Logger HAS a Writer"
interface Writer {
  write(content: string): Promise<void>;
}

class FileWriter implements Writer {
  constructor(private readonly filePath: string) {}

  async write(content: string): Promise<void> {
    await fs.appendFile(this.filePath, content + '\n');
  }
}

class ConsoleWriter implements Writer {
  async write(content: string): Promise<void> {
    console.log(content);
  }
}

class MultiWriter implements Writer {
  constructor(private readonly writers: ReadonlyArray<Writer>) {}

  async write(content: string): Promise<void> {
    await Promise.all(this.writers.map(w => w.write(content)));
  }
}

class Logger {
  constructor(
    private readonly writer: Writer,
    private readonly formatter: LogFormatter = defaultFormatter,
  ) {}

  log(level: LogLevel, message: string, context?: Record<string, unknown>): void {
    const formatted = this.formatter.format(level, message, context);
    // Fire-and-forget hoặc buffer — tùy strategy
    this.writer.write(formatted).catch(err => {
      console.error('Logger write failed:', err);
    });
  }

  info(message: string, context?: Record<string, unknown>): void {
    this.log('info', message, context);
  }

  error(message: string, context?: Record<string, unknown>): void {
    this.log('error', message, context);
  }
}

// Flexible composition tại runtime
const devLogger = new Logger(new ConsoleWriter());
const prodLogger = new Logger(
  new MultiWriter([
    new FileWriter('/var/log/app.log'),
    new CloudWatchWriter(config.cloudwatch),
  ]),
  new JsonLogFormatter(),
);
```

### Strategy Pattern — Swap behaviors at runtime

```typescript
// Ví dụ: Pricing engine cho e-commerce

interface PricingStrategy {
  calculatePrice(basePrice: Money, context: PricingContext): Money;
}

class RegularPricing implements PricingStrategy {
  calculatePrice(basePrice: Money): Money {
    return basePrice;
  }
}

class MemberDiscountPricing implements PricingStrategy {
  constructor(private readonly discountPercent: number) {}

  calculatePrice(basePrice: Money): Money {
    return basePrice.multiply(1 - this.discountPercent / 100);
  }
}

class SeasonalPricing implements PricingStrategy {
  constructor(
    private readonly rules: ReadonlyArray<SeasonalRule>,
    private readonly fallback: PricingStrategy,
  ) {}

  calculatePrice(basePrice: Money, context: PricingContext): Money {
    const activeRule = this.rules.find(r => r.isActive(context.date));
    if (activeRule) {
      return basePrice.multiply(activeRule.multiplier);
    }
    return this.fallback.calculatePrice(basePrice, context);
  }
}

// Composable: Combine multiple strategies
class CompositePricing implements PricingStrategy {
  constructor(private readonly strategies: ReadonlyArray<PricingStrategy>) {}

  calculatePrice(basePrice: Money, context: PricingContext): Money {
    return this.strategies.reduce(
      (price, strategy) => strategy.calculatePrice(price, context),
      basePrice,
    );
  }
}

// Usage — compose at application startup
const pricingStrategy = new CompositePricing([
  new SeasonalPricing(seasonalRules, new RegularPricing()),
  new MemberDiscountPricing(user.membershipLevel),
  new BulkDiscountPricing(cart.totalQuantity),
]);

const finalPrice = pricingStrategy.calculatePrice(product.basePrice, context);
```

## Patterns trong TypeScript

### Mixins — Composition with type safety

```typescript
// TypeScript mixins: composition mà giữ được type inference

type Constructor<T = {}> = new (...args: any[]) => T;

// Mixin: Timestamped
function Timestamped<TBase extends Constructor>(Base: TBase) {
  return class extends Base {
    createdAt = new Date();
    updatedAt = new Date();

    touch(): void {
      this.updatedAt = new Date();
    }
  };
}

// Mixin: SoftDeletable
function SoftDeletable<TBase extends Constructor>(Base: TBase) {
  return class extends Base {
    deletedAt: Date | null = null;

    softDelete(): void {
      this.deletedAt = new Date();
    }

    restore(): void {
      this.deletedAt = null;
    }

    get isDeleted(): boolean {
      return this.deletedAt !== null;
    }
  };
}

// Mixin: Validatable
function Validatable<TBase extends Constructor>(Base: TBase) {
  return class extends Base {
    validate(): ValidationResult {
      // Default implementation — override in concrete class
      return { valid: true, errors: [] };
    }

    assertValid(): void {
      const result = this.validate();
      if (!result.valid) {
        throw new ValidationError(result.errors);
      }
    }
  };
}

// Compose mixins
class BaseEntity {
  constructor(public readonly id: string) {}
}

class User extends Timestamped(SoftDeletable(Validatable(BaseEntity))) {
  constructor(
    id: string,
    public name: string,
    public email: string,
  ) {
    super(id);
  }

  // Override validate with specific rules
  validate(): ValidationResult {
    const errors: string[] = [];
    if (!this.name.trim()) errors.push('Name is required');
    if (!this.email.includes('@')) errors.push('Invalid email');
    return { valid: errors.length === 0, errors };
  }
}

// TypeScript infers ALL mixed-in methods
const user = new User('1', 'Alice', 'alice@example.com');
user.touch();        // from Timestamped
user.softDelete();   // from SoftDeletable
user.assertValid();  // from Validatable
```

### Decorator Pattern — Enhance behavior without modifying

```typescript
// Real-world: HTTP client with composable middleware

interface HttpClient {
  request<T>(config: RequestConfig): Promise<HttpResponse<T>>;
}

// Base implementation
class FetchHttpClient implements HttpClient {
  async request<T>(config: RequestConfig): Promise<HttpResponse<T>> {
    const response = await fetch(config.url, {
      method: config.method,
      headers: config.headers,
      body: config.body ? JSON.stringify(config.body) : undefined,
    });
    const data = await response.json();
    return { status: response.status, data, headers: response.headers };
  }
}

// Decorator: Logging
class LoggingHttpClient implements HttpClient {
  constructor(
    private readonly inner: HttpClient,
    private readonly logger: Logger,
  ) {}

  async request<T>(config: RequestConfig): Promise<HttpResponse<T>> {
    const start = performance.now();
    this.logger.info('HTTP Request', { method: config.method, url: config.url });

    try {
      const response = await this.inner.request<T>(config);
      const duration = performance.now() - start;
      this.logger.info('HTTP Response', {
        method: config.method,
        url: config.url,
        status: response.status,
        durationMs: Math.round(duration),
      });
      return response;
    } catch (error) {
      const duration = performance.now() - start;
      this.logger.error('HTTP Error', {
        method: config.method,
        url: config.url,
        error: String(error),
        durationMs: Math.round(duration),
      });
      throw error;
    }
  }
}

// Decorator: Retry
class RetryHttpClient implements HttpClient {
  constructor(
    private readonly inner: HttpClient,
    private readonly maxRetries: number = 3,
    private readonly retryableStatuses = new Set([408, 429, 500, 502, 503, 504]),
  ) {}

  async request<T>(config: RequestConfig): Promise<HttpResponse<T>> {
    let lastError: unknown;

    for (let attempt = 0; attempt <= this.maxRetries; attempt++) {
      try {
        const response = await this.inner.request<T>(config);
        if (this.retryableStatuses.has(response.status) && attempt < this.maxRetries) {
          await this.backoff(attempt);
          continue;
        }
        return response;
      } catch (error) {
        lastError = error;
        if (attempt < this.maxRetries) {
          await this.backoff(attempt);
        }
      }
    }

    throw lastError;
  }

  private backoff(attempt: number): Promise<void> {
    const delay = Math.min(1000 * 2 ** attempt, 10000);
    const jitter = Math.random() * delay * 0.1;
    return new Promise(resolve => setTimeout(resolve, delay + jitter));
  }
}

// Decorator: Auth
class AuthHttpClient implements HttpClient {
  constructor(
    private readonly inner: HttpClient,
    private readonly tokenProvider: TokenProvider,
  ) {}

  async request<T>(config: RequestConfig): Promise<HttpResponse<T>> {
    const token = await this.tokenProvider.getToken();
    const authedConfig: RequestConfig = {
      ...config,
      headers: { ...config.headers, Authorization: `Bearer ${token}` },
    };
    return this.inner.request<T>(authedConfig);
  }
}

// Decorator: Caching
class CachingHttpClient implements HttpClient {
  constructor(
    private readonly inner: HttpClient,
    private readonly cache: Cache,
    private readonly ttlMs: number = 60_000,
  ) {}

  async request<T>(config: RequestConfig): Promise<HttpResponse<T>> {
    if (config.method !== 'GET') {
      return this.inner.request<T>(config);
    }

    const cacheKey = `http:${config.url}`;
    const cached = await this.cache.get<HttpResponse<T>>(cacheKey);
    if (cached) return cached;

    const response = await this.inner.request<T>(config);
    if (response.status === 200) {
      await this.cache.set(cacheKey, response, this.ttlMs);
    }
    return response;
  }
}

// Compose decorators — order matters!
const httpClient: HttpClient = new LoggingHttpClient(
  new RetryHttpClient(
    new CachingHttpClient(
      new AuthHttpClient(
        new FetchHttpClient(),
        tokenProvider,
      ),
      cache,
    ),
    3,
  ),
  logger,
);

// Request flow: Log → Retry → Cache → Auth → Fetch
// Thêm/bỏ behavior = thêm/bỏ wrapper. KHÔNG sửa code hiện tại.
```

### Function Composition — Lightweight alternative

```typescript
// Không phải lúc nào cũng cần class. Functions compose tự nhiên trong TypeScript.

// Middleware pattern (Express/Koa style)
type Middleware<TContext> = (
  ctx: TContext,
  next: () => Promise<void>,
) => Promise<void>;

function compose<TContext>(
  middlewares: ReadonlyArray<Middleware<TContext>>,
): Middleware<TContext> {
  return async (ctx, next) => {
    let index = -1;

    async function dispatch(i: number): Promise<void> {
      if (i <= index) throw new Error('next() called multiple times');
      index = i;

      const fn = i < middlewares.length ? middlewares[i] : next;
      await fn(ctx, () => dispatch(i + 1));
    }

    await dispatch(0);
  };
}

// Usage
const pipeline = compose<RequestContext>([
  // Timing
  async (ctx, next) => {
    const start = Date.now();
    await next();
    ctx.responseTime = Date.now() - start;
  },

  // Error handling
  async (ctx, next) => {
    try {
      await next();
    } catch (error) {
      ctx.status = error instanceof AppError ? error.statusCode : 500;
      ctx.body = { error: error instanceof AppError ? error.message : 'Internal error' };
    }
  },

  // Auth
  async (ctx, next) => {
    const token = ctx.headers.authorization?.replace('Bearer ', '');
    if (!token) { ctx.status = 401; return; }
    ctx.user = await verifyToken(token);
    await next();
  },

  // Business logic
  async (ctx, _next) => {
    const result = await processRequest(ctx);
    ctx.body = result;
  },
]);
```

## Khi nào Inheritance hợp lý

Inheritance KHÔNG phải luôn sai. Nó hợp lý khi:

### 1. True "is-a" relationship với shared behavior

```typescript
// ✅ Error hierarchy — tất cả errors THỰC SỰ là Error
// Và cần shared behavior (stack trace, message formatting)

class AppError extends Error {
  constructor(
    message: string,
    readonly code: string,
    readonly statusCode: number,
    readonly isOperational: boolean = true,
  ) {
    super(message);
    this.name = this.constructor.name;
    Error.captureStackTrace(this, this.constructor);
  }

  toJSON(): Record<string, unknown> {
    return {
      name: this.name,
      code: this.code,
      message: this.message,
      statusCode: this.statusCode,
    };
  }
}

class NotFoundError extends AppError {
  constructor(resource: string, id: string) {
    super(`${resource} with id '${id}' not found`, 'NOT_FOUND', 404);
  }
}

class ValidationError extends AppError {
  constructor(
    readonly errors: ReadonlyArray<FieldError>,
  ) {
    super('Validation failed', 'VALIDATION_ERROR', 422);
  }

  toJSON() {
    return { ...super.toJSON(), errors: this.errors };
  }
}

class UnauthorizedError extends AppError {
  constructor(reason: string = 'Authentication required') {
    super(reason, 'UNAUTHORIZED', 401);
  }
}
```

### 2. Framework yêu cầu (React class components, TypeORM entities)

```typescript

// TypeORM entity
```

### 3. Template Method Pattern — shared algorithm, variable steps

```typescript
// ✅ Shared algorithm structure, subclass customize specific steps

abstract class DataImporter<TRaw, TDomain> {
  // Template method — algorithm structure is FIXED
  async import(source: DataSource): Promise<ImportResult> {
    const rawData = await this.extract(source);
    const validated = this.validate(rawData);
    const transformed = validated.map(item => this.transform(item));
    const result = await this.load(transformed);
    return result;
  }

  // Steps that vary by data type
  protected abstract extract(source: DataSource): Promise<TRaw[]>;
  protected abstract validate(data: TRaw[]): TRaw[];
  protected abstract transform(item: TRaw): TDomain;
  protected abstract load(items: TDomain[]): Promise<ImportResult>;
}

class CsvUserImporter extends DataImporter<CsvRow, User> {
  protected async extract(source: DataSource): Promise<CsvRow[]> {
    return parseCsv(await source.read());
  }

  protected validate(data: CsvRow[]): CsvRow[] {
    return data.filter(row => row.email && row.name);
  }

  protected transform(item: CsvRow): User {
    return { name: item.name, email: item.email.toLowerCase() };
  }

  protected async load(items: User[]): Promise<ImportResult> {
    const result = await this.userRepo.bulkInsert(items);
    return { imported: result.count, errors: result.errors };
  }
}
```

### Decision Matrix: Inheritance vs Composition

| Tiêu chí | Inheritance | Composition |
|----------|-------------|-------------|
| Relationship | True "is-a" | "has-a" hoặc "uses-a" |
| Coupling | Tight (compile-time) | Loose (runtime) |
| Flexibility | Fixed at class definition | Swappable at runtime |
| Shared behavior | Shared via base class | Shared via delegation |
| Testing | Phải test cả hierarchy | Mock từng component |
| Multiple behaviors | Single inheritance only | Combine nhiều behaviors |
| **Default choice** | ❌ Avoid | ✅ Prefer |

### Rule of Thumb

> **Composition first, inheritance only when:**
> 1. Có true "is-a" relationship (Error types, DOM nodes)
> 2. Base class là stable và bạn control nó (không phải third-party lib có thể thay đổi)
> 3. Bạn cần shared IMPLEMENTATION, không chỉ shared interface
> 4. Hierarchy tối đa 2-3 levels
>
> Nếu nghi ngờ → chọn composition.
