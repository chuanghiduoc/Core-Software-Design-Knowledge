## Event Types

### 1. Domain Events — "Something happened in the domain"

```typescript
// Mô tả business facts, past tense, immutable

interface DomainEvent {
  readonly eventId: string;       // Unique per event
  readonly eventType: string;     // Discriminator
  readonly aggregateId: string;   // Which entity
  readonly occurredAt: Date;      // When
  readonly version: number;       // Schema version
}

class OrderPlacedEvent implements DomainEvent {
  readonly eventType = 'order.placed';
  readonly eventId = crypto.randomUUID();
  readonly occurredAt = new Date();
  readonly version = 1;

  constructor(
    readonly aggregateId: string,  // orderId
    readonly customerId: string,
    readonly items: ReadonlyArray<{
      productId: string;
      productName: string;
      quantity: number;
      unitPrice: number;
    }>,
    readonly totalCents: number,
    readonly currency: string,
  ) {}
}

class PaymentCompletedEvent implements DomainEvent {
  readonly eventType = 'payment.completed';
  readonly eventId = crypto.randomUUID();
  readonly occurredAt = new Date();
  readonly version = 1;

  constructor(
    readonly aggregateId: string,  // paymentId
    readonly orderId: string,
    readonly amountCents: number,
    readonly paymentMethod: string,
    readonly transactionId: string,
```

```typescript
interface IntegrationEvent {

  type: string,
  data: T,

    source,
    type,
    time: new Date().toISOString(),

    data,
    subject,
```

## Event Bus & Message Brokers

### In-Process Event Bus (Monolith / Modular Monolith)

```typescript
// Typed, async event bus cho single process

type EventHandler<T> = (event: T) => Promise<void>;

class EventBus {
  private handlers = new Map<string, Set<EventHandler<any>>>();
  private deadLetterHandlers: Array<(event: unknown, error: Error) => void> = [];

  on<T extends DomainEvent>(
    eventType: string,
    handler: EventHandler<T>,
  ): () => void {
    if (!this.handlers.has(eventType)) {
      this.handlers.set(eventType, new Set());
    }
    this.handlers.get(eventType)!.add(handler);

    return () => this.handlers.get(eventType)?.delete(handler);
  }

  async publish(event: DomainEvent): Promise<void> {
    const handlers = this.handlers.get(event.eventType);
    if (!handlers || handlers.size === 0) return;

    const results = await Promise.allSettled(
      Array.from(handlers).map(handler => handler(event)),
    );

    // Log failures, don't crash publisher
    for (const result of results) {
      if (result.status === 'rejected') {
        console.error(`Event handler failed for ${event.eventType}:`, result.reason);
        this.deadLetterHandlers.forEach(h => h(event, result.reason));
      }
    }
  }

  async publishAll(events: ReadonlyArray<DomainEvent>): Promise<void> {
    for (const event of events) {
      await this.publish(event);
    }
  }

  onDeadLetter(handler: (event: unknown, error: Error) => void): void {
    this.deadLetterHandlers.push(handler);
  }
}
```

### Distributed: Kafka / RabbitMQ / Redis Streams

```typescript
// === Kafka Producer ===
class KafkaEventPublisher {
  constructor(private readonly producer: Producer) {}

  async publish(event: DomainEvent, topic?: string): Promise<void> {
    const targetTopic = topic ?? this.topicFor(event);

    await this.producer.send({
      topic: targetTopic,
      messages: [{
        key: event.aggregateId,  // Partition key → ordering per aggregate
        value: JSON.stringify({
          id: event.eventId,
          type: event.eventType,
          data: event,
          timestamp: event.occurredAt.toISOString(),
        }),
        headers: {
          'event-type': event.eventType,
          'content-type': 'application/json',
          'schema-version': String(event.version),
        },
      }],
    });
  }

  private topicFor(event: DomainEvent): string {
    // Convention: domain.aggregate.events
    const [domain] = event.eventType.split('.');
    return `${domain}-events`;
  }
}

// === Kafka Consumer ===
class KafkaEventConsumer {
  private handlers = new Map<string, EventHandler<any>>();

  constructor(
    private readonly consumer: Consumer,
    private readonly deadLetterProducer: Producer,
  ) {}

  register<T>(eventType: string, handler: EventHandler<T>): void {
    this.handlers.set(eventType, handler);
  }

  async start(topics: string[]): Promise<void> {
    await this.consumer.subscribe({ topics, fromBeginning: false });

    await this.consumer.run({
      eachMessage: async ({ topic, partition, message }) => {
        const envelope = JSON.parse(message.value!.toString());
        const handler = this.handlers.get(envelope.type);

        if (!handler) {
          console.warn(`No handler for event type: ${envelope.type}`);
          return;
        }

        try {
          await handler(envelope.data);
        } catch (error) {
          console.error(`Failed to handle ${envelope.type}:`, error);

          // Dead letter queue for failed messages
          await this.deadLetterProducer.send({
            topic: `${topic}.dlq`,
            messages: [{
              key: message.key,
              value: message.value,
              headers: {
                ...message.headers,
                'error-message': String(error),
                'original-topic': topic,
                'original-partition': String(partition),
                'failure-timestamp': new Date().toISOString(),
              },
            }],
          });
        }
      },
    });
  }
}
```

## Event Design

### Event Schema Best Practices

```typescript
// 1. Events are FACTS, not requests
// ✅ OrderPlaced (fact — happened)
// ❌ PlaceOrder (command — request)

// 2. Past tense naming
// ✅ UserRegistered, PaymentCompleted, InventoryReserved
// ❌ RegisterUser, CompletePayment, ReserveInventory

// 3. Include enough context (avoid chatty callbacks)
// ❌ Minimal event — consumers must call back for details
interface OrderPlacedEventBad {
  orderId: string;
  // Consumer: "What items? What total? I need to call Order Service!"
}

// ✅ Rich event — consumers self-sufficient
interface OrderPlacedEventGood {
  orderId: string;
  customerId: string;
  customerEmail: string;         // Notification service needs this
  items: ReadonlyArray<{
    productId: string;
    productName: string;         // Notification needs product names
    quantity: number;
    unitPriceCents: number;
  }>;
  totalCents: number;
  currency: string;
  shippingAddress: {
    street: string;
    city: string;
    country: string;
    postalCode: string;
  };
}

// 4. Schema versioning
interface EventEnvelope<T> {
  id: string;
  type: string;
  version: number;       // Schema version: 1, 2, 3...
  data: T;
  timestamp: string;
}

// Version 1
interface OrderPlacedV1 { orderId: string; total: number; }
// Version 2 — added currency (backward compatible)
interface OrderPlacedV2 { orderId: string; totalCents: number; currency: string; }

// Consumer handles multiple versions
function handleOrderPlaced(envelope: EventEnvelope<unknown>): void {
  switch (envelope.version) {
    case 1: {
      const data = envelope.data as OrderPlacedV1;
      processOrder(data.orderId, data.total * 100, 'USD'); // Assume USD for v1
      break;
    }
    case 2: {
      const data = envelope.data as OrderPlacedV2;
      processOrder(data.orderId, data.totalCents, data.currency);
      break;
    }
    default:
      console.warn(`Unknown version ${envelope.version} for ${envelope.type}`);
  }
}
```

### Outbox Pattern — Reliable event publishing

```typescript
// Vấn đề: Save entity + publish event phải atomic
// Nếu save succeeds nhưng publish fails → data inconsistency

// ❌ Không atomic
async function placeOrder(command: PlaceOrderCommand): Promise<void> {
  await orderRepo.save(order);           // ✅ Saved
  await eventPublisher.publish(event);    // ❌ Fails → event lost!
}

// ✅ Outbox Pattern: save event vào DB cùng transaction
async function placeOrder(command: PlaceOrderCommand): Promise<void> {
  await db.transaction(async (trx) => {
    // Save order
    await trx('orders').insert(orderRow);

    // Save event to outbox table (cùng transaction)
    await trx('outbox_events').insert({
      id: crypto.randomUUID(),
      type: 'order.placed',
      payload: JSON.stringify(eventData),
      created_at: new Date(),
      published: false,
    });
  });
  // Transaction succeeds → cả order và event đều persisted
}

// Background worker: poll outbox, publish to broker
class OutboxPublisher {
  constructor(
    private readonly db: Knex,
    private readonly producer: KafkaProducer,
  ) {}

  async processOutbox(): Promise<void> {
    const events = await this.db('outbox_events')
      .where('published', false)
      .orderBy('created_at')
      .limit(100);

    for (const event of events) {
      try {
        await this.producer.send({
          topic: this.topicFor(event.type),
          messages: [{ key: event.aggregate_id, value: event.payload }],
        });

        await this.db('outbox_events')
          .where('id', event.id)
          .update({ published: true, published_at: new Date() });
      } catch (error) {
        console.error(`Failed to publish event ${event.id}:`, error);
        break; // Stop processing to maintain ordering
      }
    }
  }
}

// Run every 1-5 seconds
setInterval(() => outboxPublisher.processOutbox(), 2000);
```

## Choreography vs Orchestration

### Choreography — Decentralized, event-driven

```
Order Service: OrderPlaced →
  Payment Service: listens → charges → PaymentCompleted →
    Inventory Service: listens → reserves → InventoryReserved →
      Shipping Service: listens → creates shipment → ShipmentCreated →
        Notification Service: listens → sends email

Mỗi service tự quyết định react to events nào.
Không có central coordinator.
```

### Orchestration — Centralized coordinator

```
Order Saga Orchestrator:
  1. → Payment Service: charge
  2. ← Payment Service: success
  3. → Inventory Service: reserve
  4. ← Inventory Service: success
  5. → Shipping Service: create shipment
  6. ← Shipping Service: success
  7. → Notification Service: send email
  8. Done

Orchestrator biết toàn bộ flow, quyết định next step.
```

| | Choreography | Orchestration |
|--|-------------|---------------|
| Coupling | Very loose | Coordinator coupled to all services |
| Visibility | Hard to see full flow | Clear, sequential flow |
| Adding steps | Thêm consumer, existing services unchanged | Modify orchestrator |
| Error handling | Each service handles its own | Centralized compensation |
| Debugging | Need distributed tracing | Look at orchestrator state |
| Best for | Simple flows, notifications | Complex business workflows |

## Idempotency & Exactly-Once

### Vấn đề: Duplicate messages

```
Producer → Broker → Consumer
                      │
                      ▼
              Process message
              Save to DB         ← Succeeds
              Acknowledge         ← Network error! Broker thinks message not processed
              Broker resends      ← Consumer processes AGAIN → duplicate!
```

### Giải pháp: Idempotent Consumers

```typescript
// Mỗi event có unique ID → track which events already processed

class IdempotentEventHandler<T> {
  constructor(
    private readonly handler: (event: T) => Promise<void>,
    private readonly processedStore: ProcessedEventStore,
  ) {}

  async handle(eventId: string, event: T): Promise<void> {
    // Check if already processed
    if (await this.processedStore.isProcessed(eventId)) {
      console.log(`Event ${eventId} already processed, skipping`);
      return;
    }

    // Process
    await this.handler(event);

    // Mark as processed (ideally trong cùng transaction với handler)
    await this.processedStore.markProcessed(eventId);
  }
}

// Processed event store (Redis hoặc DB table)
class RedisProcessedEventStore implements ProcessedEventStore {
  constructor(private readonly redis: Redis) {}

  async isProcessed(eventId: string): Promise<boolean> {
    return (await this.redis.exists(`processed:${eventId}`)) === 1;
  }

  async markProcessed(eventId: string): Promise<void> {
    // TTL: keep for 7 days then auto-cleanup
    await this.redis.set(`processed:${eventId}`, '1', 'EX', 7 * 24 * 3600);
  }
}

// Usage
const handler = new IdempotentEventHandler(
  async (event: OrderPlacedEvent) => {
    await inventoryService.reserve(event.items);
  },
  redisProcessedStore,
);

// Nếu cùng event đến 2 lần → chỉ process 1 lần
await handler.handle(event.eventId, event);
await handler.handle(event.eventId, event); // Skipped
```

### Idempotent Operations by Design

```typescript
// Thiết kế operations để tự nhiên idempotent

// ❌ NOT idempotent: increment
await db.query('UPDATE accounts SET balance = balance + $1 WHERE id = $2', [100, accountId]);
// Chạy 2 lần = balance +200 (sai!)

// ✅ Idempotent: set absolute value with version check
await db.query(
  'UPDATE accounts SET balance = $1, version = $2 WHERE id = $3 AND version = $4',
  [newBalance, currentVersion + 1, accountId, currentVersion],
);
// Chạy 2 lần: lần 2 WHERE version fails → no-op (đúng!)

// ✅ Idempotent: upsert with unique constraint
await db.query(
  'INSERT INTO processed_payments (payment_id, order_id, amount) VALUES ($1, $2, $3) ON CONFLICT (payment_id) DO NOTHING',
  [paymentId, orderId, amount],
);
// Duplicate insert → silently ignored
```

### Dead Letter Queue (DLQ)

```typescript
// Events that fail repeatedly → move to DLQ for manual inspection

class EventConsumerWithDLQ {
  private readonly maxRetries = 3;

  async processMessage(message: KafkaMessage): Promise<void> {
    const retryCount = Number(message.headers?.['retry-count'] ?? 0);

    try {
      await this.handler(message);
    } catch (error) {
      if (retryCount >= this.maxRetries) {
        // Move to DLQ
        await this.dlqProducer.send({
          topic: `${message.topic}.dlq`,
          messages: [{
            key: message.key,
            value: message.value,
            headers: {
              ...message.headers,
              'error-message': String(error),
              'retry-count': String(retryCount),
              'moved-to-dlq-at': new Date().toISOString(),
            },
          }],
        });
        console.error(`Message moved to DLQ after ${retryCount} retries`);
      } else {
        // Retry with backoff
        await this.retryProducer.send({
          topic: `${message.topic}.retry`,
          messages: [{
            key: message.key,
            value: message.value,
            headers: {
              ...message.headers,
              'retry-count': String(retryCount + 1),
              'retry-after': new Date(Date.now() + 2 ** retryCount * 1000).toISOString(),
            },
          }],
        });
      }
    }
  }
}
```
