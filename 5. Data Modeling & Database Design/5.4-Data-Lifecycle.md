## 1. Migration Strategies

### Core principle: Zero-Downtime Migrations

Production database KHÔNG được phép downtime cho schema changes. Mọi migration phải backward-compatible.

```sql
-- ❌ Rename column trực tiếp → code cũ crash ngay lập tức
ALTER TABLE users RENAME COLUMN name TO display_name;

-- ❌ Drop column đang dùng
ALTER TABLE users DROP COLUMN legacy_field;

-- ❌ Add NOT NULL column không có default
ALTER TABLE users ADD COLUMN phone TEXT NOT NULL;  -- existing rows fail
```

### Expand-Contract Pattern (Multi-phase migration)

**Phase 1: Expand** — Thêm mới, không bỏ cũ.

```sql
ALTER TABLE users ADD COLUMN display_name TEXT;  -- nullable
```

```typescript
// Code v1: Write CẢ HAI columns, read từ mới (fallback cũ)
await db.query('UPDATE users SET name = $1, display_name = $1 WHERE id = $2', [name, id]);
const user = await db.query(
  'SELECT *, COALESCE(display_name, name) AS resolved_name FROM users WHERE id = $1', [id]
);
```

**Phase 2: Migrate** — Backfill data, chuyển code sang column mới.

```typescript
// ✅ Batch backfill cho large tables (KHÔNG update toàn bộ 1 lần)
async function batchMigrate(db: Pool, batchSize = 10_000) {
  let total = 0;
  while (true) {
    const { rowCount } = await db.query(`
      UPDATE users SET display_name = name
      WHERE id IN (SELECT id FROM users WHERE display_name IS NULL LIMIT $1)
    `, [batchSize]);
    total += rowCount ?? 0;
    if (!rowCount || rowCount < batchSize) break;
    await new Promise(r => setTimeout(r, 100));  // Pause để không overload
  }
  console.log(`Migrated ${total} rows`);
}
```

**Phase 3: Contract** — Xoá column cũ sau khi chắc chắn không còn code nào dùng.

```sql
ALTER TABLE users ALTER COLUMN display_name SET NOT NULL;
ALTER TABLE users DROP COLUMN name;
```

### Safe DDL Operations (PostgreSQL)

```sql
-- ✅ KHÔNG lock table:
ALTER TABLE users ADD COLUMN new_col TEXT;                        -- Instant (metadata only)
ALTER TABLE users ALTER COLUMN bio SET DEFAULT 'hello';           -- Instant
ALTER TABLE users ADD COLUMN calc INT DEFAULT 0;                  -- Instant (PG 11+)

-- ⚠️ CÓ THỂ lock:
ALTER TABLE users ALTER COLUMN bio SET NOT NULL;                  -- Full table scan

-- ✅ Safe NOT NULL: 2-step
ALTER TABLE users ADD CONSTRAINT users_name_nn CHECK (display_name IS NOT NULL) NOT VALID;
ALTER TABLE users VALIDATE CONSTRAINT users_name_nn;  -- Concurrent, no lock

-- ✅ Safe index creation
CREATE INDEX CONCURRENTLY idx_users_name ON users(display_name);
```

### Migration files

```typescript
// 20240115_001_add_display_name.ts
import { MigrationBuilder } from 'node-pg-migrate';

export async function up(pgm: MigrationBuilder): Promise<void> {
  pgm.addColumn('users', { display_name: { type: 'text', notNull: false } });
}
export async function down(pgm: MigrationBuilder): Promise<void> {
  pgm.dropColumn('users', 'display_name');
}
```

## 2. Data Versioning & Schema Evolution

### Schema Registry Pattern

```typescript
// ✅ Version events explicitly
interface OrderCreatedV1 {
  version: 1;
  orderId: string;
  customerName: string;     // V1: embedded name
  items: Array<{ productId: string; quantity: number }>;
}

interface OrderCreatedV2 {
  version: 2;
  orderId: string;
  customerId: string;       // V2: reference instead
  items: Array<{ productId: string; quantity: number; unitPrice: number }>;
}

type OrderCreatedEvent = OrderCreatedV1 | OrderCreatedV2;

// Handler supports all versions
function handleOrderCreated(event: OrderCreatedEvent) {
  switch (event.version) {
    case 1: return processOrder({
      orderId: event.orderId,
      customerId: lookupByName(event.customerName),
      items: event.items.map(i => ({ ...i, unitPrice: 0 })),
    });
    case 2: return processOrder(event);
  }
}
```

### Compatibility Rules

```typescript
// ✅ Safe (backward compatible):
// - Add optional field        - Add new enum value at END
// - Widen type (int32→int64)  - Add new table/collection

// ❌ Breaking (NOT backward compatible):
// - Remove field              - Rename field
// - Change field type         - Add required field without default

// ✅ Strategy: Never remove, only deprecate
interface UserProfileV3 {
  displayName: string;
  /** @deprecated Use displayName. Removed in V5. */
  name?: string;
}
```

### Schema Version Tracking

```sql
CREATE TABLE schema_migrations (

  name TEXT NOT NULL,
  applied_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  execution_time_ms INTEGER,
  checksum TEXT
```

## 3. Backup & Recovery

### ❌ Backup anti-patterns

```
- Có backup nhưng CHƯA BAO GIỜ test restore = chỉ là hy vọng
- Chỉ backup data, quên schema
- Backup trên cùng server → server chết = backup mất
```

### ✅ 3-2-1 Rule: 3 copies, 2 media, 1 offsite

```typescript
interface BackupStrategy {
  continuousArchiving: {

  };

    schedule: 'Sunday 03:00 UTC';
    retention: '90 days';

  restoreTest: {
    schedule: 'First Monday of each month';

```

```typescript
// ✅ Automated backup + upload + verify
async function performBackup(): Promise<void> {
  const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
  const filepath = `/tmp/backup-${timestamp}.dump`;

  await execPromise(`pg_dump --format=custom --compress=9 --file=${filepath} ${process.env.DATABASE_URL}`);

  await s3.send(new PutObjectCommand({
    Bucket: process.env.BACKUP_BUCKET!,
    Key: `postgres/daily/backup-${timestamp}.dump`,
    Body: createReadStream(filepath),
    ServerSideEncryption: 'aws:kms',
  }));

  await execPromise(`pg_restore --list ${filepath}`);  // Verify integrity
  await execPromise(`rm ${filepath}`);
  await notifySlack(`Backup completed: ${timestamp}`);
}

// ✅ Monthly restore test
async function testRestore(): Promise<void> {
  const backup = await getLatestBackupFromS3();
  await execPromise(`pg_restore --dbname=${process.env.TEST_DB_URL} --clean --if-exists ${backup}`);
  const checks = await runIntegrityChecks(process.env.TEST_DB_URL!);
  if (!checks.passed) await notifyPagerDuty('Restore test FAILED', checks.errors);
}
```

### Table Partitioning cho archival

```sql
-- ✅ Partition by time → easy archive/drop old data
CREATE TABLE events (
  id BIGSERIAL, user_id UUID NOT NULL, event_type TEXT NOT NULL,
  data JSONB, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
) PARTITION BY RANGE (created_at);

CREATE TABLE events_2024_01 PARTITION OF events FOR VALUES FROM ('2024-01-01') TO ('2024-02-01');
CREATE TABLE events_2024_02 PARTITION OF events FOR VALUES FROM ('2024-02-01') TO ('2024-03-01');

-- Archive: detach partition (instant), export to S3, drop table (free disk)
ALTER TABLE events DETACH PARTITION events_2023_01;
DROP TABLE events_2023_01;
```

```typescript
// ✅ Auto-create future partitions
async function ensurePartitions(months = 3) {
  for (let i = 0; i < months; i++) {
    const start = new Date(); start.setMonth(start.getMonth() + i, 1); start.setHours(0, 0, 0, 0);
    const end = new Date(start); end.setMonth(end.getMonth() + 1);
    const name = `events_${start.getFullYear()}_${String(start.getMonth() + 1).padStart(2, '0')}`;
    await db.query(`
      CREATE TABLE IF NOT EXISTS ${name} PARTITION OF events
      FOR VALUES FROM ('${start.toISOString()}') TO ('${end.toISOString()}')
    `).catch(e => { if (!e.message.includes('already exists')) throw e; });
  }
}
```

## 5. GDPR Compliance

### Right to Deletion (Article 17)

```typescript
// ❌ Soft delete KHÔNG đủ cho GDPR — PII vẫn queryable
await db.query('UPDATE users SET deleted_at = NOW() WHERE id = $1', [userId]);

// ✅ Hard delete PII, anonymize references, keep business data
class GdprDeletionService {
  async deleteUser(userId: string): Promise<DeletionReport> {
    const report: DeletionReport = { userId, deletedAt: new Date(), actions: [] };

    await this.db.transaction(async (tx) => {
      // 1. Anonymize orders (keep for financial reporting)
      await tx.query(`UPDATE orders SET customer_name='DELETED', customer_email=NULL,
        shipping_address_id=NULL WHERE customer_id=$1`, [userId]);
      report.actions.push('orders: anonymized');

      // 2. Delete PII tables
      await tx.query('DELETE FROM user_addresses WHERE user_id = $1', [userId]);
      await tx.query('DELETE FROM user_payment_methods WHERE user_id = $1', [userId]);
      report.actions.push('addresses, payment_methods: deleted');

      // 3. Anonymize audit logs
      await tx.query(`UPDATE audit_logs SET actor_id='DELETED', actor_ip=NULL WHERE actor_id=$1`, [userId]);

      // 4. Delete user record
      await tx.query('DELETE FROM users WHERE id = $1', [userId]);

      // 5. Log for compliance audit
      await tx.query(`INSERT INTO gdpr_deletion_log (user_id, report, completed_at)
        VALUES ($1, $2, NOW())`, [userId, JSON.stringify(report)]);
    });

    // 6. Delete from external systems
    await this.deleteFromElasticsearch(userId);
    await this.deleteFromRedisCache(userId);
    // Backups: expire naturally within retention period (30 days)
    return report;
  }
}
```

## 6. Data Seeding

### ❌ Random, non-deterministic

```typescript
// ❌ Random mỗi lần → tests không reproducible
for (let i = 0; i < 100; i++)
  db.query('INSERT INTO users (name, email) VALUES ($1, $2)',
    [faker.person.fullName(), faker.internet.email()]);  // Khác mỗi lần
```

### ✅ Layered, deterministic, idempotent

```typescript
import { faker } from '@faker-js/faker';

// Layer 1: Reference data (same across ALL environments)
async function seedReferenceData(db: Pool) {
  await db.query(`INSERT INTO countries (code, name) VALUES
    ('VN','Vietnam'), ('US','United States'), ('JP','Japan')
    ON CONFLICT (code) DO NOTHING`);
}

// Layer 2: Dev data (deterministic seed → same data every time)
async function seedDevData(db: Pool) {
  faker.seed(42);  // Deterministic!

  await db.query(`INSERT INTO users (id, email, display_name, password_hash, is_email_verified)
    VALUES ('00000000-0000-0000-0000-000000000001', 'admin@dev.local', 'Dev Admin',
      '$2b$10$fixedhash...', true) ON CONFLICT DO NOTHING`);

  for (let i = 0; i < 50; i++) {
    await db.query(`INSERT INTO users (id, email, display_name, password_hash, is_email_verified)
      VALUES ($1, $2, $3, '$2b$10$fixedhash...', true) ON CONFLICT DO NOTHING`,
      [`00000000-0000-0000-0000-${String(i + 100).padStart(12, '0')}`,
       faker.internet.email(), faker.person.fullName()]);
  }
}

// Layer 3: Load test data (high volume)
async function seedLoadTestData(db: Pool) { /* 100K orders in batches */ }

// ✅ Run layers based on environment
async function seed(env: string) {
  await seedReferenceData(db);
  if (env === 'development' || env === 'test') await seedDevData(db);
  if (env === 'staging') { await seedDevData(db); await seedLoadTestData(db); }
  // ⚠️ NEVER seed test data in production
}
```

## 7. Database Branching

Database branching = **isolated copy** cho mỗi feature branch. Giống git branch cho data.

**Dùng khi:** Test migrations trước production, mỗi PR cần DB riêng, preview environments.
**KHÔNG dùng khi:** DB > 100GB (cost cao), team nhỏ (<3 devs), schema ít thay đổi.

### Neon — PostgreSQL with Copy-on-Write branching

```typescript
import { NeonClient } from '@neondatabase/api-client';

// CI/CD: tạo branch DB cho mỗi PR (instant, near-zero storage)
async function createBranchForPR(prNumber: number): Promise<string> {
  const branch = await neon.createProjectBranch(projectId, {
    branch: { name: `pr-${prNumber}`, parent_id: 'main' },
    endpoints: [{ type: 'read_write', autoscaling_limit_min_cu: 0.25, autoscaling_limit_max_cu: 1 }],
  });
  return branch.connection_uris?.[0]?.connection_uri!;
}

async function deleteBranchForPR(prNumber: number) {
  const branches = await neon.listProjectBranches(projectId);
  const branch = branches.branches.find(b => b.name === `pr-${prNumber}`);
  if (branch) await neon.deleteProjectBranch(projectId, branch.id);
}
```

### Migration Testing Workflow

```typescript
// ✅ CI: Test migrations against real production schema
async function testMigrations() {
  const branchUrl = await createBranchForPR(parseInt(process.env.PR_NUMBER!));
  try {
    await execPromise(`DATABASE_URL="${branchUrl}" npx prisma migrate deploy`);
    const errors = await validateSchema(branchUrl);
    if (errors.length > 0) throw new Error(`Validation failed: ${errors}`);
    await execPromise(`DATABASE_URL="${branchUrl}" npm run test:integration`);
    await execPromise(`DATABASE_URL="${branchUrl}" npx prisma migrate reset --force`);
    console.log('Migration test passed');
  } finally {
    await deleteBranchForPR(parseInt(process.env.PR_NUMBER!));
  }
}
```
